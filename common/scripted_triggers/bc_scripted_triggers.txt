# Battle Chronicles - shared situation tests.
#
# Every gate lives here rather than inline in the events, so "what counts as
# losing" is one decision instead of seven.
#
# All of these assume they are evaluated from a combat phase event, where
# `scope:combat_side` exists. They are not safe anywhere else - EXCEPT
# bc_battle_is_notable_trigger at the bottom of the file, which takes the
# combat side as ROOT and is called from the combat-end events, where
# `scope:combat_side` does not exist. Check which one you are writing.

# --- how the battle is going -----------------------------------------------
# Proven in game (probe 5).

bc_side_is_losing_trigger = {
	scope:combat_side.side_strength < scope:combat_side.enemy_side.side_strength
}

bc_side_is_winning_trigger = {
	scope:combat_side.side_strength > scope:combat_side.enemy_side.side_strength
}

# Half again as many. See the warning in bc_script_values.txt.
bc_side_is_outnumbered_trigger = {
	scope:combat_side.enemy_side.side_strength > bc_outnumbered_threshold_value
}

bc_side_is_outnumbering_trigger = {
	scope:combat_side.side_strength > bc_outnumbering_threshold_value
}

# --- ground -----------------------------------------------------------------
# scope:combat_side.combat.location is a province scope, so a terrain check works
# there directly. Proven in game (probe 4).
#
# NEVER `terrain = X`. ALWAYS `agot_is_X_terrain = yes`. AGOT's
# common/terrain_types/01_agot_terrains.txt defines road variants as DISTINCT
# terrain types - `majorroad_hills` is not `hills` - and adds seven bases vanilla
# has no key for. These two triggers shipped with bare keys and so missed 424 and
# 51 provinces respectively, silently, from the first live run onward. AGOT's 23
# helpers in common/scripted_triggers/00_agot_terrain_type_triggers.txt each fold
# a base key together with its two road variants; that is the only correct way to
# ask what ground a battle is on.
#
# The helpers are base AGOT, not Seasons of Ice and Fire. An earlier version of
# this comment justified the bare keys as avoiding a hard dependency on a submod.
# That was the wrong file: 2962333032 is `name="A Game of Thrones"` itself, and
# descriptor.mod already declares `dependencies = { "A Game of Thrones" }`.
#
# `mountains` used to appear in BOTH lists, so in the mountains the ground
# simultaneously favoured you and bogged you down and the two events fought over
# the same roll. It belongs to the bogging list: hills and forest already carry
# "good defensive ground".
#
# the_bog and taiga_bog are new to the bogging list and are the point of it. The
# old comment argued that wetlands and jungle alone are "close to nonexistent in
# Westeros - the Neck and nothing else". That was right about the Neck and wrong
# about the key: AGOT models the Neck as `the_bog`, 139 provinces including road
# variants, which the old trigger did not match at all. highlands and
# terraced_hills join the favouring list for the same reason - they are AGOT's own
# variations on hills. terraced_hills currently has zero provinces on the map; it
# is listed anyway so the pair stays exhaustive over what "hill-like" means if
# AGOT ever places one.

bc_terrain_favours_us_trigger = {
	scope:combat_side.combat.location = {
		OR = {
			agot_is_hills_terrain = yes
			agot_is_forest_terrain = yes
			agot_is_highlands_terrain = yes
			agot_is_terraced_hills_terrain = yes
		}
	}
}

bc_terrain_bogs_us_down_trigger = {
	scope:combat_side.combat.location = {
		OR = {
			agot_is_wetlands_terrain = yes
			agot_is_jungle_terrain = yes
			agot_is_mountains_terrain = yes
			agot_is_the_bog_terrain = yes
			agot_is_taiga_bog_terrain = yes
		}
	}
}

# --- the field, by band -----------------------------------------------------
#
# COMBAT-SIDE WRAPPERS around the six band triggers in bc_ground_triggers.txt.
# They exist because a band trigger is PROVINCE-scoped - bc_write_ground_effect
# calls them from inside `scope:combat_side.combat.location = { }` - while an
# is_valid block is rooted on the commander and has to walk there itself.
#
# THESE ARE NOT A SECOND TERRAIN LADDER. They define no membership; each one
# delegates to the band trigger of the same name. That distinction is what keeps
# tools/ck3.py validate's partition check meaningful: it reads
# bc_ground_triggers.txt only, so a wrapper that inlined its own `agot_is_*`
# list here would drift from AGOT uncaught. If a band's membership is ever
# wrong, it is wrong in one file for prose, gates and streaks alike.
#
# EXISTS-GUARDED AND NESTED, because combat.location does not always resolve -
# bc_write_ground_effect carries a `BC: ground missing` marker for exactly that
# case - and CK3 does not short-circuit among siblings.
#
# ALL SIX BANDS ARE WRAPPED SINCE 2026-08-17. Four were wrapped before, with a
# note here to add the others when something needed them; the rough and jungle
# gates below are that something. The set is now closed and matches the six in
# bc_ground_triggers.txt one for one, which is the state a reader should expect:
# any band that prose can name, a gate can name too.
#
# These are rooted on a CHARACTER but never touch that character, so a knight
# event may use them exactly as a commander event does: `scope:combat_side` is
# set for both event types.

bc_field_is_winter_trigger = {
	exists = scope:combat_side.combat.location
	trigger_if = {
		limit = { exists = scope:combat_side.combat.location }
		scope:combat_side.combat.location = { bc_ground_is_winter_trigger = yes }
	}
}

# --- the shape of the battle ------------------------------------------------
#
# Read live off the combat, NOT off a bc_ scratch variable. Every bc_battle_*
# variable is written in the EFFECT of a phase event and weights are evaluated
# BEFORE the effect, so a variable read in an is_valid block would be absent on
# the first roll of every battle and stale-or-absent thereafter. The combat
# itself is always current.

# Fifteen thousand men on one field. num_total_troops counts BOTH sides, which
# is what makes this a statement about the battle rather than about our host -
# the same reason bc_battle_is_notable_trigger uses it.
bc_host_is_vast_trigger = {
	scope:combat_side.combat = { num_total_troops >= 15000 }
}

# Two fifths of the enemy already dead. The notability trigger's blood row sits
# at 50 and is measured against COMPLETED battles; this one is asked mid-fight,
# when the number is still climbing, so it is set below that bar on purpose.
bc_field_is_bloody_trigger = {
	scope:combat_side = { percent_enemies_killed >= 40 }
}

# Two years of war. combat_war is a combat -> war link and a combat does not
# always have one, so it is guarded exactly like the location walks above.
bc_war_is_long_trigger = {
	exists = scope:combat_side.combat.combat_war
	trigger_if = {
		limit = { exists = scope:combat_side.combat.combat_war }
		scope:combat_side.combat.combat_war = { war_days >= 730 }
	}
}

# --- the man across the field, and the man behind us ------------------------

# Close family: parents, children, siblings, grandparents, grandchildren.
#
# THE SAME TEST THE BOND LADDER USES, deliberately - bc_write_bond_effect's
# `kin` arm is `is_close_family_of` too. If these two ever disagree, a battle
# would be gated as a kinslaying and then have its toast call the man a
# stranger. Widening one means widening both.
#
# CALLED ON ROOT WITH THE FOE AS THE TARGET, rather than entering the foe's
# scope. `is_close_family_of` is mutual, so the direction does not matter, and
# staying out of the foe's scope keeps ROOT unambiguous for any caller.
bc_foe_is_kin_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	trigger_if = {
		limit = { exists = scope:combat_side.enemy_side.side_commander }
		is_close_family_of = scope:combat_side.enemy_side.side_commander
	}
}

# The other two answers the bond ladder gives, in the same shape and reading the
# same two properties it reads.
#
# THE LADDER IS THE SPECIFICATION FOR BOTH. bc_write_bond_effect tests culture
# BEFORE faith and each of these tests only its own property, so THESE TWO ARE
# NOT MUTUALLY EXCLUSIVE the way the ladder's arms are: a man of the same
# culture and a different faith answers yes to both here and is recorded as
# `countryman` there. That is correct for a gate - a phase event asks one
# question - and it is the reason a template must not assume the clause under
# its own toast will agree with its gate. Nothing in the mod requires them to.
#
# WRITTEN LIKE THE LADDER, foe's scope entered and root's property read from
# outside it, so the two places can be diffed by eye.
bc_foe_is_countryman_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	trigger_if = {
		limit = { exists = scope:combat_side.enemy_side.side_commander }
		scope:combat_side.enemy_side.side_commander = { culture = root.culture }
	}
}

bc_foe_is_heathen_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	trigger_if = {
		limit = { exists = scope:combat_side.enemy_side.side_commander }
		scope:combat_side.enemy_side.side_commander = { NOT = { faith = root.faith } }
	}
}

# The same name on both sides, and NOT the same blood.
#
# THIS IS NOT bc_foe_is_kin_trigger WIDENED - the two are DISJOINT by
# construction, and the final NOT is what makes them so. `is_close_family_of`
# is parent, child, sibling; a house is everything else that shares the name.
# The distinction is the whole template: a man across the field from his
# brother is a tragedy the mod already writes, and a man across the field from
# a third cousin whose grandfather took the other side of a succession is a
# different sentence entirely, and the commoner one in Westeros.
#
# HOUSE, NOT DYNASTY. In AGOT the unit that means anything on a shield is the
# house - the cadet branches ARE the story - and both comparisons have AGOT
# precedent (`house = scope:attacker.house` at
# common/casus_belli_types/05_fp3_wars.txt:1282, `dynasty = scope:attacker.dynasty`
# at common/casus_belli_types/00_agot_adventurer_wars.txt:384). Widening this
# to dynasty would put every branch of a great house in one bucket, which is
# the opposite of the case being written.
#
# BOTH HOUSES ARE `exists`-GUARDED, AND THAT GUARD IS LEAD-LINED. A lowborn
# character has no house at all, and CK3 does not short-circuit, so an
# unguarded `house = root.house` between two lowborn commanders is a
# comparison between two absent links - which this mod may not gamble on
# resolving to false. bc_knight_no_house_behind_him gates on exactly those men
# existing, so they are not hypothetical. Vanilla's bare `exists = house` is at
# common/casus_belli_types/01_ep1_wars.txt:81.
#
# `root`, NOT `prev`, inside the entered scope, for the reason recorded on
# bc_foe_is_countryman_trigger.
bc_foe_is_house_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	trigger_if = {
		limit = { exists = scope:combat_side.enemy_side.side_commander }
		exists = house
		trigger_if = {
			limit = { exists = house }
			scope:combat_side.enemy_side.side_commander = {
				exists = house
				trigger_if = {
					limit = { exists = house }
					house = root.house
				}
			}
		}
		NOT = { is_close_family_of = scope:combat_side.enemy_side.side_commander }
	}
}

# How good the man on the other side is at this.
#
# THE MOD HAS NEVER ASKED. Every foe trigger above reads the enemy commander's
# IDENTITY - his blood, his house, his faith, his tongue - and the battle
# triggers read the two hosts as numbers. Nothing anywhere has ever asked
# whether the man giving the other side's orders is any good, which means
# seventy-plus templates are written as though a commander only ever plays
# against the ground and the odds.
#
# `martial`, NOT `prowess`, AND THE CHOICE IS THE WHOLE POINT. Prowess is how
# well he personally fights, and what he personally does with a sword has no
# bearing on whether our orders are being answered before they arrive. martial
# is the number about command, and command is what is being felt from over
# here.
#
# NO `exists = house`-STYLE INNER GUARD IS NEEDED. Unlike house and dynasty, a
# skill is not a link - every character has a martial value - so the single
# `exists` on the commander is the whole guard.
#
# 16 AND 8 ARE THE SAME BARS THE COMMANDER FILE ALREADY USES on its own
# `martial` rows, so the two triggers are calibrated against a scale this mod
# has been reading since plan 1 rather than a new one invented here.
#
# THE TWO ARE MUTUALLY EXCLUSIVE and there is a wide band between them that
# matches neither. That gap is deliberate: an ordinary competent opponent is
# not a moment, and widening either bar towards the middle would make one of
# them fire in most battles in Westeros.
bc_foe_is_formidable_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	trigger_if = {
		limit = { exists = scope:combat_side.enemy_side.side_commander }
		scope:combat_side.enemy_side.side_commander = { martial >= 16 }
	}
}

bc_foe_is_outmatched_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	trigger_if = {
		limit = { exists = scope:combat_side.enemy_side.side_commander }
		scope:combat_side.enemy_side.side_commander = { martial <= 8 }
	}
}

# He is not the man whose war this is.
#
# side_primary_participant is the WAR LEADER of this side, and this trigger
# makes NO claim that he is standing on the field - a war leader can be a
# kingdom away. It says only that the commander is fighting someone else's
# quarrel, which is what the prose it gates is about. Do not rewrite the prose
# to put the liege at the battle; nothing here can support that.
bc_fighting_his_lords_war_trigger = {
	exists = scope:combat_side.side_primary_participant
	trigger_if = {
		limit = { exists = scope:combat_side.side_primary_participant }
		NOT = { this = scope:combat_side.side_primary_participant }
	}
}

# The man whose war this is holds something over him.
#
# `has_hook` APPEARED NOWHERE IN THE MOD BEFORE THIS. It is the one trigger
# available here that answers WHY a man is standing on a field he did not
# choose, and the mod has been writing about obligation - bc_his_lords_war,
# bc_nothing_but_the_horse, the whole station batch - without ever asking the
# question the game can actually answer.
#
# THE DIRECTION IS THE WAR LEADER ON HIM, and the reverse trigger below is a
# different sentence entirely, not a formatting variant. Keep them apart.
#
# `root`, NOT `this`, INSIDE THE ENTERED SCOPE. `has_hook = root` is read from
# the war leader's scope and asks about the character the phase event is
# rolling for. The same rule and the same reason as bc_side_has_kin_knight_trigger
# below: once a scope has been entered, `this` is the man whose scope it is.
#
# THE `NOT = { this = ... }` ROW IS NOT REDUNDANT DEFENSIVENESS. A man cannot
# hold a hook on himself, so the row cannot change the answer - it is there
# because the prose these gate is about somebody ELSE having leverage, and a
# reader who deletes the row has to notice that first.
#
# ROLE-AGNOSTIC. Nothing here reads a property only a commander has, so the
# knight file may call it, and a knight is never his own side's war leader.
bc_his_lord_holds_a_hook_trigger = {
	exists = scope:combat_side.side_primary_participant
	trigger_if = {
		limit = { exists = scope:combat_side.side_primary_participant }
		NOT = { this = scope:combat_side.side_primary_participant }
		scope:combat_side.side_primary_participant = { has_hook = root }
	}
}

# He holds something over the man whose war this is.
#
# THE SAME TWO CHARACTERS, THE OTHER WAY ROUND, AND A DIFFERENT AFTERNOON. The
# trigger above is about a man who was made to come; this one is about a man
# who came anyway while holding the leash. Both can be true at once, which is
# a knot the mod does not try to describe - if both templates are ever gated
# so they can fire in one battle, that is the case to think about.
#
# NO ENTERED SCOPE NEEDED. `has_hook` is read from the caller's own scope
# against a target, so this half needs no `root` and no inner scope.
bc_he_holds_his_lords_hook_trigger = {
	exists = scope:combat_side.side_primary_participant
	trigger_if = {
		limit = { exists = scope:combat_side.side_primary_participant }
		NOT = { this = scope:combat_side.side_primary_participant }
		has_hook = scope:combat_side.side_primary_participant
	}
}

# The war is his own.
#
# NOT THE `NOT` OF bc_fighting_his_lords_war_trigger, AND IT MATTERS. Wrapping
# bc_fighting_his_lords_war_trigger in a NOT would answer YES on a side with no
# war leader at all, because that trigger's `exists` makes it false there and
# the NOT flips it. This one has to be written affirmatively so a leaderless
# side answers no to both. Do not "simplify" it into the negation.
#
# `this`, NOT `root`. Written for the commander file, where the phase event's
# root is the commander being rolled for and `this` is the same character. A
# knight template must not call it - a knight is by definition not the war
# leader of his side, so it would be a gate that is always false. The knight
# side's version of this question is the foe trigger below, which reads
# nothing about the man calling it.
bc_this_is_his_own_war_trigger = {
	exists = scope:combat_side.side_primary_participant
	trigger_if = {
		limit = { exists = scope:combat_side.side_primary_participant }
		this = scope:combat_side.side_primary_participant
	}
}

# The man whose war this is, is commanding the other side in person.
#
# THIS IS THE FIRST TRIGGER IN THE MOD THAT COMPARES TWO MEMBERS OF THE SAME
# SIDE. Every other reads one property off one character. It says the enemy's
# war leader and the enemy's commander are the same man, which is the one thing
# available here that distinguishes a lord's host from a lord's deputy's host.
#
# TWO `exists` AND ONE `limit` CARRYING BOTH. Either link can be absent - a
# side can be leaderless and a combat can have no war - and CK3 does not
# short-circuit, so both guards are required and the trigger_if's limit has to
# repeat both. The comparison is `this = ...` from inside the commander's
# scope, the same idiom bc_fighting_his_lords_war_trigger uses.
#
# ROLE-AGNOSTIC BY CONSTRUCTION. It reads no property of the character calling
# it, so it answers the same question from a commander root and a knight root.
# That is deliberate: it is the one enemy-side question a knight is as
# entitled to ask as his commander, because it is answered by a banner.
bc_foe_leader_commands_in_person_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	exists = scope:combat_side.enemy_side.side_primary_participant
	trigger_if = {
		limit = {
			exists = scope:combat_side.enemy_side.side_commander
			exists = scope:combat_side.enemy_side.side_primary_participant
		}
		scope:combat_side.enemy_side.side_commander = {
			this = scope:combat_side.enemy_side.side_primary_participant
		}
	}
}

# His own blood, on his own side, in the knight list.
#
# THE SAME is_close_family_of AS bc_foe_is_kin_trigger, pointed the other way.
# Both are the bond ladder's `kin` test and both must be widened together if
# either ever is - the note on that trigger applies here word for word.
#
# NO `exists` GUARD, AND NONE IS NEEDED. Every other trigger in this section
# walks a link that can be absent (a side can be leaderless, a combat can have
# no war) and pays for it with a trigger_if. `any_side_knight` is an ITERATOR,
# not a link: a side with no knights iterates nothing and the trigger is simply
# false, which is the answer wanted.
#
# `root`, NOT `prev` OR `this`. Inside `any_side_knight` the current scope is
# the knight; root is still the commander the phase event is rolling for. The
# two foe triggers above read `root.culture` and `root.faith` from inside an
# entered scope for the same reason.
#
# THE COMMANDER MAY BE IN HIS OWN KNIGHT LIST - nothing in the API settles it,
# and bc_chronicles.0091 guards against it when handing out memories. It does
# not matter here: a man is not close family of himself, so an unguarded
# iteration cannot match on him.
bc_side_has_kin_knight_trigger = {
	scope:combat_side = {
		any_side_knight = {
			is_close_family_of = root
		}
	}
}

# --- pacing -----------------------------------------------------------------

bc_chronicle_ready_trigger = {
	NOT = { has_character_flag = bc_chronicle_recent }
}

# The COMMANDER variant. A separate trigger, NOT an edit of the one above.
#
# The shared one is still used by the knight events, and it must stay exactly as
# it is for them: no bc_ variable is ever written onto a knight (spec Section 1),
# so `NOT = { exists = var:bc_battle_moments }` is ALWAYS TRUE on a knight.
# Rewriting the shared trigger in place would therefore disable the knight
# cooldown entirely - every knight event would pass the OR on its second arm
# forever.
#
# The second arm is the FLOOR: a commander who has recorded nothing yet gets one
# moment immediately, no matter how long the size-scaled flag has left to run.
# Without it a size-1 skirmish sets a 40-day flag on the first roll that misses,
# and a short battle ends with no chronicle at all.
#
# It is an `exists` arm, not `var:bc_battle_moments = 0`. The variable is written
# on the FIRST recorded moment, so before that it does not exist - and comparing
# a missing variable is the read CK3 logs about. Once it exists it is at least 1
# and this arm is false forever after, which is what makes it a one-shot floor.
bc_chronicle_ready_commander_trigger = {
	OR = {
		NOT = { has_character_flag = bc_chronicle_recent }
		NOT = { exists = var:bc_battle_moments }
	}
}

bc_can_interrupt_player_trigger = {

}

# --- was this battle worth remembering --------------------------------------
#
# ROOT is the combat side. This is the one trigger in the file that does not
# use scope:combat_side - see the header.
#
# calc_true_if is the engine's own points system and is what vanilla uses for
# this exact question, twice: combat_event.3000 for the map point-of-interest
# and historical_artifacts.0100 for naming an artifact after a battle. Both
# count direct children, so DO NOT wrap any of these rows in a nested scripted
# trigger - a nested trigger counts as one row no matter how many tests it
# holds.
#
# Six substantive rows plus the game rule. The two size rows nest, so the bar
# actually reads "huge, or big plus one other thing". Deliberate: thirty
# thousand men on a field is notable whatever else happened. It also means the
# blood, odds and warscore rows never decide the largest battles, so read their
# calibration ladders against the sub-huge population.
#
# THE SIXTH ROW - bc_was_wipe_trigger - WAS ADDED AFTER THE 2026-08-13
# CALIBRATION AND IS NOT IN THAT SAMPLE. The 39 battles below were measured
# before scope:wipe was read at all, so the ladders describe a five-row trigger.
# A wipe is worth one point of the two, so the row cannot raise the pass rate on
# its own; it can only promote a battle that already had one other row. Re-run
# the calibration if that assumption is ever questioned - do not adjust the
# thresholds to compensate for it.
#
# The bar must be SYMMETRIC - both sides must reach the same verdict, or one
# side's knights get a memory of a battle the other side's knights do not.
# num_total_troops and warscore_value are combat-scoped and symmetric for free.
# Blood is made symmetric by the explicit OR across enemy_side. The odds row is
# symmetric because it asks "was either host much larger", not "were we
# outnumbered". The game rule is the same on both sides.
#
# CALIBRATED 2026-08-13 against 39 combat ends, both sides logged separately.
# Vanilla's numbers turned out to be the wrong shape entirely, not merely the
# wrong size: THREE of the five rows never fired once in 39 battles. What the
# ladders showed, and what each number is now:
#
#   size   under 5k 26 / 5-10k 8 / 10-20k 5 / 20k+ NONE
#          Vanilla's 20k and 30k were dead rows. 10k/20k instead. 10k is the
#          top 13% of this sample; 20k fires on nothing yet and is deliberately
#          reserved for late-campaign hosts, which are much larger than the
#          early-game armies this run measured. Do not lower it to fit a
#          bookmark start - it is the one row meant to age into relevance.
#   blood  under 25 3 / 25-50 29 / 50-75 7 / 75+ NONE
#          75 was a dead row. 50 is the top 18%.
#   odds   even 9 / 1.2x 9 / 1.5x 11 / 2x 10
#          1.5x fired on 21 of 39 - FIFTY-FOUR PERCENT. It was carrying the
#          whole trigger and calling half of all battles remarkable. 2x is the
#          top 26%.
#   score  under 10 30 / 10-25 3 / 25-50 2 / 50+ 4
#          25 already sat right at the top 15%. The only threshold kept.
#
# The three ratio rows - blood, odds, warscore - are scale free and will hold
# as hosts grow. Size is the only absolute, which is why its two rungs are set
# high rather than fitted to this sample.
#
# Re-derive by restoring the ladders from git history, not by guessing.

bc_battle_is_notable_trigger = {
	calc_true_if = {
		amount >= 2

		combat = { num_total_troops >= 10000 }
		combat = { num_total_troops >= 20000 }
		OR = {
			percent_enemies_killed >= 50
			enemy_side.percent_enemies_killed >= 50
		}
		OR = {
			enemy_side.side_max_army_size >= bc_our_host_gap_value
			side_max_army_size >= bc_their_host_gap_value
		}
		combat = { warscore_value >= 25 }
		bc_was_wipe_trigger = yes
		has_game_rule = bc_aftermath_generous
	}
}

# Is the man across the field the man who has beaten me before?
#
# CALLED ON THE COMMANDER, in TRIGGER contexts only - weight multipliers and
# is_valid blocks. It must therefore be safe to evaluate on every commander in
# the world on every phase event roll, most of whom have no nemesis at all.
#
# THE TWO `exists` ROWS ARE SAFE AS SIBLINGS - `exists` on a missing variable is
# safe, and only a READ of a missing variable logs an error. The trigger_if
# repeats them and does the reading, which is the shipped pattern this mod
# already uses in the graders' every_side_knight limit.
#
# IT MISSES THE FIRST ROLL OF EVERY BATTLE, deliberately. bc_battle_foe is
# written in the EFFECT of a phase event and weights are evaluated BEFORE the
# effect, so the first roll of a battle sees nothing and every roll after sees
# this battle's foe. Reading the combat scope directly instead would fix that at
# the cost of an unverified scope path evaluated on every roll of every battle in
# the world.
#
# SINCE bc_write_bond_effect, ONE CALLER IS NOT A WEIGHT BLOCK. That effect - in
# bc_context_effects.txt, called from bc_battle_open_effect - asks this in the
# `limit` of its first `if`. Still a trigger context, so the safety argument
# above holds unchanged, but two things above are now local to the weight call
# sites and not to this trigger:
#   - "TRIGGER contexts only - weight multipliers and is_valid blocks" now also
#     means the limit of an if in an effect.
#   - "IT MISSES THE FIRST ROLL OF EVERY BATTLE" is true of weights and FALSE
#     from bc_write_bond_effect, which runs after the foe block in the same
#     effect and therefore sees bc_battle_foe on the very first moment.
# The second one matters: do not "fix" the first-roll gap by moving the foe
# write, because the bond ladder already depends on the write being upstream.
bc_facing_nemesis_trigger = {
	exists = var:bc_nemesis
	exists = var:bc_battle_foe
	trigger_if = {
		limit = {
			exists = var:bc_nemesis
			exists = var:bc_battle_foe
		}
		var:bc_nemesis = var:bc_battle_foe
		var:bc_nemesis = { is_alive = yes }
	}
}

# --- was this an annihilation ------------------------------------------------
#
# READS scope:wipe, WHICH THE ENGINE PASSES TO BOTH COMBAT-END HOOKS. AGOT
# combat_on_actions.txt:13 documents it above on_combat_end_winner and :527 above
# on_combat_end_loser, so both graders see the same value and any bar built on it
# is SYMMETRIC for free - which bc_battle_is_notable_trigger's header requires.
#
# IT SURVIVES INTO AN EVENT RAISED FROM THE `events = { }` LIST. Proven by
# vanilla: combat_event.1001 is raised from on_combat_end_winner's list
# (AGOT combat_on_actions.txt:17) and reads `scope:wipe = yes` at AGOT
# events/war_events/combat_events.txt:781 and :947. Nothing has to thread it.
#
# SCOPE-INDEPENDENT. A saved scope is readable from any scope in the event, so
# this trigger is equally valid on the combat side (the notability row) and on
# the commander inside side_commander ?= { } (bc_style_step_value).
#
# TRIGGER_IF, NOT THE SPEC'S `AND = { exists = scope:wipe  scope:wipe = yes }`.
# The conjunction is right and the sibling shape is wrong. Shipped script uses
# BOTH idioms - `exists` at combat_on_actions.txt:100, `= yes` at :758 - which
# means it is unsettled whether the scope always exists carrying a boolean or
# exists only when true. Under the second reading the spec's AND compares against
# a missing scope on EVERY non-wipe combat end in the world, which is nearly all
# of them, and CK3 does not short-circuit and guarantees no evaluation order among
# siblings. `exists` is safe on a missing scope; the COMPARISON is what has to be
# nested. Same rule, same reason, as every variable guard in plans 3 and 4.
bc_was_wipe_trigger = {
	exists = scope:wipe
	trigger_if = {
		limit = { exists = scope:wipe }
		scope:wipe = yes
	}
}

# --- is the man across the field the one who has beaten me before -----------
#
# THE OPENING'S read-back, and it is NOT bc_facing_nemesis_trigger. That one reads
# var:bc_battle_foe live, which is right for an effect that runs inside a moment.
# This one is read from an EVENT DESC, and a desc is rendered when the player gets
# round to the pop-up - possibly after the battle has ended and
# bc_clear_scratch_effect has removed the variable. A saved scope is not cleared,
# so scope:bc_eve_foe still names the right man; var:bc_battle_foe might name
# nobody at all.
#
# THE SCOPE IS WRITTEN FROM THE VARIABLE, one line before the raise, in
# bc_eve_offer_effect. So it is this battle's opposing commander, not a stale one -
# the freshness comes from the variable, the durability from the scope.
#
# CALLED ON THE COMMANDER, in a TRIGGER context - an event desc branch. It must be
# safe for a commander who has no nemesis, which is nearly all of them, and safe
# when the other side is leaderless and nothing saved the scope.
#
# THE TWO `exists` ROWS ARE SAFE AS SIBLINGS. `exists` on a missing variable or a
# missing scope is safe; only a READ has to be nested, which is what the
# trigger_if does.
#
# NO is_alive ROW, unlike bc_facing_nemesis_trigger. The eve's foe is commanding a
# battle that is starting, so he is alive by construction; that trigger reads a
# variable which can outlive its subject.
bc_eve_facing_nemesis_trigger = {
	exists = var:bc_nemesis
	exists = scope:bc_eve_foe
	trigger_if = {
		limit = {
			exists = var:bc_nemesis
			exists = scope:bc_eve_foe
		}
		var:bc_nemesis = scope:bc_eve_foe
	}
}

# He wears a white cloak.
#
# SEVEN ROWS BECAUSE AGOT MODELS THE KINGSGUARD AS SEVEN SEPARATE COUNCIL
# POSITIONS, not as one with seven holders: `kingsguard_lord_commander` plus
# `kingsguard_1` through `kingsguard_6`, all defined in
# common/council_positions/00_agot_council_kingsguard.txt. There is no single
# key that means "is in the Kingsguard", so the OR is the whole of the
# available answer and dropping any row silently loses a seventh of the order.
#
# NOT A TRAIT AND NOT A TITLE. The temptation is to look for a `kingsguard`
# trait; there is none. The temptation after that is to test for a knighthood
# or a vow trait, which would collect a great many men who are not in the
# order at all.
#
# NO SCOPE WALK AND NO GUARD. `has_council_position` reads off the character
# in scope and answers no for a character who holds none, so this is one of the
# few triggers in this file that needs no `exists` and no trigger_if. It is
# safe from any character scope, including one entered from another.
#
# THE LORD COMMANDER IS NOT SEPARATED OUT HERE. If a template ever wants him
# alone, gate on `kingsguard_lord_commander` directly rather than adding a
# second trigger - one row is not worth a name.
bc_is_kingsguard_trigger = {
	OR = {
		has_council_position = kingsguard_lord_commander
		has_council_position = kingsguard_1
		has_council_position = kingsguard_2
		has_council_position = kingsguard_3
		has_council_position = kingsguard_4
		has_council_position = kingsguard_5
		has_council_position = kingsguard_6
	}
}
