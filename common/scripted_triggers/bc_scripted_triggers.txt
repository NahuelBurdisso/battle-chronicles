# Battle Chronicles - shared situation tests.
#
# Every gate lives here rather than inline in the events, so "what counts as
# losing" is one decision instead of seven.
#
# All of these assume they are evaluated from a combat phase event, where
# `scope:combat_side` exists. They are not safe anywhere else - EXCEPT
# bc_battle_is_notable_trigger at the bottom of the file, which takes the
# combat side as ROOT and is called from the combat-end events, where
# `scope:combat_side` does not exist. Check which one you are writing.

# --- how the battle is going -----------------------------------------------
# Proven in game (probe 5).

bc_side_is_losing_trigger = {
	scope:combat_side.side_strength < scope:combat_side.enemy_side.side_strength
}

bc_side_is_winning_trigger = {
	scope:combat_side.side_strength > scope:combat_side.enemy_side.side_strength
}

# Half again as many. See the warning in bc_script_values.txt.
bc_side_is_outnumbered_trigger = {
	scope:combat_side.enemy_side.side_strength > bc_outnumbered_threshold_value
}

bc_side_is_outnumbering_trigger = {
	scope:combat_side.side_strength > bc_outnumbering_threshold_value
}

# --- ground -----------------------------------------------------------------
# scope:combat_side.combat.location is a province scope, so a terrain check works
# there directly. Proven in game (probe 4).
#
# NEVER `terrain = X`. ALWAYS `agot_is_X_terrain = yes`. AGOT's
# common/terrain_types/01_agot_terrains.txt defines road variants as DISTINCT
# terrain types - `majorroad_hills` is not `hills` - and adds seven bases vanilla
# has no key for. These two triggers shipped with bare keys and so missed 424 and
# 51 provinces respectively, silently, from the first live run onward. AGOT's 23
# helpers in common/scripted_triggers/00_agot_terrain_type_triggers.txt each fold
# a base key together with its two road variants; that is the only correct way to
# ask what ground a battle is on.
#
# The helpers are base AGOT, not Seasons of Ice and Fire. An earlier version of
# this comment justified the bare keys as avoiding a hard dependency on a submod.
# That was the wrong file: 2962333032 is `name="A Game of Thrones"` itself, and
# descriptor.mod already declares `dependencies = { "A Game of Thrones" }`.
#
# `mountains` used to appear in BOTH lists, so in the mountains the ground
# simultaneously favoured you and bogged you down and the two events fought over
# the same roll. It belongs to the bogging list: hills and forest already carry
# "good defensive ground".
#
# the_bog and taiga_bog are new to the bogging list and are the point of it. The
# old comment argued that wetlands and jungle alone are "close to nonexistent in
# Westeros - the Neck and nothing else". That was right about the Neck and wrong
# about the key: AGOT models the Neck as `the_bog`, 139 provinces including road
# variants, which the old trigger did not match at all. highlands and
# terraced_hills join the favouring list for the same reason - they are AGOT's own
# variations on hills. terraced_hills currently has zero provinces on the map; it
# is listed anyway so the pair stays exhaustive over what "hill-like" means if
# AGOT ever places one.

bc_terrain_favours_us_trigger = {
	scope:combat_side.combat.location = {
		OR = {
			agot_is_hills_terrain = yes
			agot_is_forest_terrain = yes
			agot_is_highlands_terrain = yes
			agot_is_terraced_hills_terrain = yes
		}
	}
}

bc_terrain_bogs_us_down_trigger = {
	scope:combat_side.combat.location = {
		OR = {
			agot_is_wetlands_terrain = yes
			agot_is_jungle_terrain = yes
			agot_is_mountains_terrain = yes
			agot_is_the_bog_terrain = yes
			agot_is_taiga_bog_terrain = yes
		}
	}
}

# --- the field, by band -----------------------------------------------------
#
# COMBAT-SIDE WRAPPERS around the six band triggers in bc_ground_triggers.txt.
# They exist because a band trigger is PROVINCE-scoped - bc_write_ground_effect
# calls them from inside `scope:combat_side.combat.location = { }` - while an
# is_valid block is rooted on the commander and has to walk there itself.
#
# THESE ARE NOT A SECOND TERRAIN LADDER. They define no membership; each one
# delegates to the band trigger of the same name. That distinction is what keeps
# tools/ck3.py validate's partition check meaningful: it reads
# bc_ground_triggers.txt only, so a wrapper that inlined its own `agot_is_*`
# list here would drift from AGOT uncaught. If a band's membership is ever
# wrong, it is wrong in one file for prose, gates and streaks alike.
#
# EXISTS-GUARDED AND NESTED, because combat.location does not always resolve -
# bc_write_ground_effect carries a `BC: ground missing` marker for exactly that
# case - and CK3 does not short-circuit among siblings.
#
# ALL SIX BANDS ARE WRAPPED SINCE 2026-08-17. Four were wrapped before, with a
# note here to add the others when something needed them; the rough and jungle
# gates below are that something. The set is now closed and matches the six in
# bc_ground_triggers.txt one for one, which is the state a reader should expect:
# any band that prose can name, a gate can name too.
#
# These are rooted on a CHARACTER but never touch that character, so a knight
# event may use them exactly as a commander event does: `scope:combat_side` is
# set for both event types.

bc_field_is_winter_trigger = {
	exists = scope:combat_side.combat.location
	trigger_if = {
		limit = { exists = scope:combat_side.combat.location }
		scope:combat_side.combat.location = { bc_ground_is_winter_trigger = yes }
	}
}

# --- the shape of the battle ------------------------------------------------
#
# Read live off the combat, NOT off a bc_ scratch variable. Every bc_battle_*
# variable is written in the EFFECT of a phase event and weights are evaluated
# BEFORE the effect, so a variable read in an is_valid block would be absent on
# the first roll of every battle and stale-or-absent thereafter. The combat
# itself is always current.

# Fifteen thousand men on one field. num_total_troops counts BOTH sides, which
# is what makes this a statement about the battle rather than about our host -
# the same reason bc_battle_is_notable_trigger uses it.
bc_host_is_vast_trigger = {
	scope:combat_side.combat = { num_total_troops >= 15000 }
}

# Two fifths of the enemy already dead. The notability trigger's blood row sits
# at 50 and is measured against COMPLETED battles; this one is asked mid-fight,
# when the number is still climbing, so it is set below that bar on purpose.
bc_field_is_bloody_trigger = {
	scope:combat_side = { percent_enemies_killed >= 40 }
}

# Two years of war. combat_war is a combat -> war link and a combat does not
# always have one, so it is guarded exactly like the location walks above.
bc_war_is_long_trigger = {
	exists = scope:combat_side.combat.combat_war
	trigger_if = {
		limit = { exists = scope:combat_side.combat.combat_war }
		scope:combat_side.combat.combat_war = { war_days >= 730 }
	}
}

# --- the man across the field, and the man behind us ------------------------

# Close family: parents, children, siblings, grandparents, grandchildren.
#
# THE SAME TEST THE BOND LADDER USES, deliberately - bc_write_bond_effect's
# `kin` arm is `is_close_family_of` too. If these two ever disagree, a battle
# would be gated as a kinslaying and then have its toast call the man a
# stranger. Widening one means widening both.
#
# CALLED ON ROOT WITH THE FOE AS THE TARGET, rather than entering the foe's
# scope. `is_close_family_of` is mutual, so the direction does not matter, and
# staying out of the foe's scope keeps ROOT unambiguous for any caller.
bc_foe_is_kin_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	trigger_if = {
		limit = { exists = scope:combat_side.enemy_side.side_commander }
		is_close_family_of = scope:combat_side.enemy_side.side_commander
	}
}

# The other two answers the bond ladder gives, in the same shape and reading the
# same two properties it reads.
#
# THE LADDER IS THE SPECIFICATION FOR BOTH. bc_write_bond_effect tests culture
# BEFORE faith and each of these tests only its own property, so THESE TWO ARE
# NOT MUTUALLY EXCLUSIVE the way the ladder's arms are: a man of the same
# culture and a different faith answers yes to both here and is recorded as
# `countryman` there. That is correct for a gate - a phase event asks one
# question - and it is the reason a template must not assume the clause under
# its own toast will agree with its gate. Nothing in the mod requires them to.
#
# WRITTEN LIKE THE LADDER, foe's scope entered and root's property read from
# outside it, so the two places can be diffed by eye.
bc_foe_is_countryman_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	trigger_if = {
		limit = { exists = scope:combat_side.enemy_side.side_commander }
		scope:combat_side.enemy_side.side_commander = { culture = root.culture }
	}
}

bc_foe_is_heathen_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	trigger_if = {
		limit = { exists = scope:combat_side.enemy_side.side_commander }
		scope:combat_side.enemy_side.side_commander = { NOT = { faith = root.faith } }
	}
}

# The same name on both sides, and NOT the same blood.
#
# THIS IS NOT bc_foe_is_kin_trigger WIDENED - the two are DISJOINT by
# construction, and the final NOT is what makes them so. `is_close_family_of`
# is parent, child, sibling; a house is everything else that shares the name.
# The distinction is the whole template: a man across the field from his
# brother is a tragedy the mod already writes, and a man across the field from
# a third cousin whose grandfather took the other side of a succession is a
# different sentence entirely, and the commoner one in Westeros.
#
# HOUSE, NOT DYNASTY. In AGOT the unit that means anything on a shield is the
# house - the cadet branches ARE the story - and both comparisons have AGOT
# precedent (`house = scope:attacker.house` at
# common/casus_belli_types/05_fp3_wars.txt:1282, `dynasty = scope:attacker.dynasty`
# at common/casus_belli_types/00_agot_adventurer_wars.txt:384). Widening this
# to dynasty would put every branch of a great house in one bucket, which is
# the opposite of the case being written.
#
# BOTH HOUSES ARE `exists`-GUARDED, AND THAT GUARD IS LEAD-LINED. A lowborn
# character has no house at all, and CK3 does not short-circuit, so an
# unguarded `house = root.house` between two lowborn commanders is a
# comparison between two absent links - which this mod may not gamble on
# resolving to false. bc_knight_no_house_behind_him gates on exactly those men
# existing, so they are not hypothetical. Vanilla's bare `exists = house` is at
# common/casus_belli_types/01_ep1_wars.txt:81.
#
# `root`, NOT `prev`, inside the entered scope, for the reason recorded on
# bc_foe_is_countryman_trigger.
bc_foe_is_house_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	trigger_if = {
		limit = { exists = scope:combat_side.enemy_side.side_commander }
		exists = house
		trigger_if = {
			limit = { exists = house }
			scope:combat_side.enemy_side.side_commander = {
				exists = house
				trigger_if = {
					limit = { exists = house }
					house = root.house
				}
			}
		}
		NOT = { is_close_family_of = scope:combat_side.enemy_side.side_commander }
	}
}

# How good the man on the other side is at this.
#
# THE MOD HAS NEVER ASKED. Every foe trigger above reads the enemy commander's
# IDENTITY - his blood, his house, his faith, his tongue - and the battle
# triggers read the two hosts as numbers. Nothing anywhere has ever asked
# whether the man giving the other side's orders is any good, which means
# seventy-plus templates are written as though a commander only ever plays
# against the ground and the odds.
#
# `martial`, NOT `prowess`, AND THE CHOICE IS THE WHOLE POINT. Prowess is how
# well he personally fights, and what he personally does with a sword has no
# bearing on whether our orders are being answered before they arrive. martial
# is the number about command, and command is what is being felt from over
# here.
#
# NO `exists = house`-STYLE INNER GUARD IS NEEDED. Unlike house and dynasty, a
# skill is not a link - every character has a martial value - so the single
# `exists` on the commander is the whole guard.
#
# 16 AND 8 ARE THE SAME BARS THE COMMANDER FILE ALREADY USES on its own
# `martial` rows, so the two triggers are calibrated against a scale this mod
# has been reading since plan 1 rather than a new one invented here.
#
# THE TWO ARE MUTUALLY EXCLUSIVE and there is a wide band between them that
# matches neither. That gap is deliberate: an ordinary competent opponent is
# not a moment, and widening either bar towards the middle would make one of
# them fire in most battles in Westeros.
bc_foe_is_formidable_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	trigger_if = {
		limit = { exists = scope:combat_side.enemy_side.side_commander }
		scope:combat_side.enemy_side.side_commander = { martial >= 16 }
	}
}

bc_foe_is_outmatched_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	trigger_if = {
		limit = { exists = scope:combat_side.enemy_side.side_commander }
		scope:combat_side.enemy_side.side_commander = { martial <= 8 }
	}
}

# He is not the man whose war this is.
#
# side_primary_participant is the WAR LEADER of this side, and this trigger
# makes NO claim that he is standing on the field - a war leader can be a
# kingdom away. It says only that the commander is fighting someone else's
# quarrel, which is what the prose it gates is about. Do not rewrite the prose
# to put the liege at the battle; nothing here can support that.
bc_fighting_his_lords_war_trigger = {
	exists = scope:combat_side.side_primary_participant
	trigger_if = {
		limit = { exists = scope:combat_side.side_primary_participant }
		NOT = { this = scope:combat_side.side_primary_participant }
	}
}

# The man whose war this is holds something over him.
#
# `has_hook` APPEARED NOWHERE IN THE MOD BEFORE THIS. It is the one trigger
# available here that answers WHY a man is standing on a field he did not
# choose, and the mod has been writing about obligation - bc_his_lords_war,
# bc_nothing_but_the_horse, the whole station batch - without ever asking the
# question the game can actually answer.
#
# THE DIRECTION IS THE WAR LEADER ON HIM, and the reverse trigger below is a
# different sentence entirely, not a formatting variant. Keep them apart.
#
# `root`, NOT `this`, INSIDE THE ENTERED SCOPE. `has_hook = root` is read from
# the war leader's scope and asks about the character the phase event is
# rolling for. The same rule and the same reason as bc_side_has_kin_knight_trigger
# below: once a scope has been entered, `this` is the man whose scope it is.
#
# THE `NOT = { this = ... }` ROW IS NOT REDUNDANT DEFENSIVENESS. A man cannot
# hold a hook on himself, so the row cannot change the answer - it is there
# because the prose these gate is about somebody ELSE having leverage, and a
# reader who deletes the row has to notice that first.
#
# ROLE-AGNOSTIC. Nothing here reads a property only a commander has, so the
# knight file may call it, and a knight is never his own side's war leader.
bc_his_lord_holds_a_hook_trigger = {
	exists = scope:combat_side.side_primary_participant
	trigger_if = {
		limit = { exists = scope:combat_side.side_primary_participant }
		NOT = { this = scope:combat_side.side_primary_participant }
		scope:combat_side.side_primary_participant = { has_hook = root }
	}
}

# He holds something over the man whose war this is.
#
# THE SAME TWO CHARACTERS, THE OTHER WAY ROUND, AND A DIFFERENT AFTERNOON. The
# trigger above is about a man who was made to come; this one is about a man
# who came anyway while holding the leash. Both can be true at once, which is
# a knot the mod does not try to describe - if both templates are ever gated
# so they can fire in one battle, that is the case to think about.
#
# NO ENTERED SCOPE NEEDED. `has_hook` is read from the caller's own scope
# against a target, so this half needs no `root` and no inner scope.
bc_he_holds_his_lords_hook_trigger = {
	exists = scope:combat_side.side_primary_participant
	trigger_if = {
		limit = { exists = scope:combat_side.side_primary_participant }
		NOT = { this = scope:combat_side.side_primary_participant }
		has_hook = scope:combat_side.side_primary_participant
	}
}

# The war is his own.
#
# NOT THE `NOT` OF bc_fighting_his_lords_war_trigger, AND IT MATTERS. Wrapping
# bc_fighting_his_lords_war_trigger in a NOT would answer YES on a side with no
# war leader at all, because that trigger's `exists` makes it false there and
# the NOT flips it. This one has to be written affirmatively so a leaderless
# side answers no to both. Do not "simplify" it into the negation.
#
# `this`, NOT `root`. Written for the commander file, where the phase event's
# root is the commander being rolled for and `this` is the same character. A
# knight template must not call it - a knight is by definition not the war
# leader of his side, so it would be a gate that is always false. The knight
# side's version of this question is the foe trigger below, which reads
# nothing about the man calling it.
bc_this_is_his_own_war_trigger = {
	exists = scope:combat_side.side_primary_participant
	trigger_if = {
		limit = { exists = scope:combat_side.side_primary_participant }
		this = scope:combat_side.side_primary_participant
	}
}

# The man whose war this is, is commanding the other side in person.
#
# THIS IS THE FIRST TRIGGER IN THE MOD THAT COMPARES TWO MEMBERS OF THE SAME
# SIDE. Every other reads one property off one character. It says the enemy's
# war leader and the enemy's commander are the same man, which is the one thing
# available here that distinguishes a lord's host from a lord's deputy's host.
#
# TWO `exists` AND ONE `limit` CARRYING BOTH. Either link can be absent - a
# side can be leaderless and a combat can have no war - and CK3 does not
# short-circuit, so both guards are required and the trigger_if's limit has to
# repeat both. The comparison is `this = ...` from inside the commander's
# scope, the same idiom bc_fighting_his_lords_war_trigger uses.
#
# ROLE-AGNOSTIC BY CONSTRUCTION. It reads no property of the character calling
# it, so it answers the same question from a commander root and a knight root.
# That is deliberate: it is the one enemy-side question a knight is as
# entitled to ask as his commander, because it is answered by a banner.
bc_foe_leader_commands_in_person_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	exists = scope:combat_side.enemy_side.side_primary_participant
	trigger_if = {
		limit = {
			exists = scope:combat_side.enemy_side.side_commander
			exists = scope:combat_side.enemy_side.side_primary_participant
		}
		scope:combat_side.enemy_side.side_commander = {
			this = scope:combat_side.enemy_side.side_primary_participant
		}
	}
}

# His own blood, on his own side, in the knight list.
#
# THE SAME is_close_family_of AS bc_foe_is_kin_trigger, pointed the other way.
# Both are the bond ladder's `kin` test and both must be widened together if
# either ever is - the note on that trigger applies here word for word.
#
# NO `exists` GUARD, AND NONE IS NEEDED. Every other trigger in this section
# walks a link that can be absent (a side can be leaderless, a combat can have
# no war) and pays for it with a trigger_if. `any_side_knight` is an ITERATOR,
# not a link: a side with no knights iterates nothing and the trigger is simply
# false, which is the answer wanted.
#
# `root`, NOT `prev` OR `this`. Inside `any_side_knight` the current scope is
# the knight; root is still the commander the phase event is rolling for. The
# two foe triggers above read `root.culture` and `root.faith` from inside an
# entered scope for the same reason.
#
# THE COMMANDER MAY BE IN HIS OWN KNIGHT LIST - nothing in the API settles it,
# and bc_chronicles.0091 guards against it when handing out memories. It does
# not matter here: a man is not close family of himself, so an unguarded
# iteration cannot match on him.
bc_side_has_kin_knight_trigger = {
	scope:combat_side = {
		any_side_knight = {
			is_close_family_of = root
		}
	}
}

# --- pacing -----------------------------------------------------------------

bc_chronicle_ready_trigger = {
	NOT = { has_character_flag = bc_chronicle_recent }
}

# The COMMANDER variant. A separate trigger, NOT an edit of the one above.
#
# The shared one is still used by the knight events, and it must stay exactly as
# it is for them: no bc_ variable is ever written onto a knight (spec Section 1),
# so `NOT = { exists = var:bc_battle_moments }` is ALWAYS TRUE on a knight.
# Rewriting the shared trigger in place would therefore disable the knight
# cooldown entirely - every knight event would pass the OR on its second arm
# forever.
#
# The second arm is the FLOOR: a commander who has recorded nothing yet gets one
# moment immediately, no matter how long the size-scaled flag has left to run.
# Without it a size-1 skirmish sets a 40-day flag on the first roll that misses,
# and a short battle ends with no chronicle at all.
#
# It is an `exists` arm, not `var:bc_battle_moments = 0`. The variable is written
# on the FIRST recorded moment, so before that it does not exist - and comparing
# a missing variable is the read CK3 logs about. Once it exists it is at least 1
# and this arm is false forever after, which is what makes it a one-shot floor.
bc_chronicle_ready_commander_trigger = {
	OR = {
		NOT = { has_character_flag = bc_chronicle_recent }
		NOT = { exists = var:bc_battle_moments }
	}
}

bc_can_interrupt_player_trigger = {
	is_ai = no
	NOT = { has_character_flag = bc_popup_cooldown }
}

# --- was this battle worth remembering --------------------------------------
#
# ROOT is the combat side. This is the one trigger in the file that does not
# use scope:combat_side - see the header.
#
# calc_true_if is the engine's own points system and is what vanilla uses for
# this exact question, twice: combat_event.3000 for the map point-of-interest
# and historical_artifacts.0100 for naming an artifact after a battle. Both
# count direct children, so DO NOT wrap any of these rows in a nested scripted
# trigger - a nested trigger counts as one row no matter how many tests it
# holds.
#
# Six substantive rows plus the game rule. The two size rows nest, so the bar
# actually reads "huge, or big plus one other thing". Deliberate: thirty
# thousand men on a field is notable whatever else happened. It also means the
# blood, odds and warscore rows never decide the largest battles, so read their
# calibration ladders against the sub-huge population.
#
# THE SIXTH ROW - bc_was_wipe_trigger - WAS ADDED AFTER THE 2026-08-13
# CALIBRATION AND IS NOT IN THAT SAMPLE. The 39 battles below were measured
# before scope:wipe was read at all, so the ladders describe a five-row trigger.
# A wipe is worth one point of the two, so the row cannot raise the pass rate on
# its own; it can only promote a battle that already had one other row. Re-run
# the calibration if that assumption is ever questioned - do not adjust the
# thresholds to compensate for it.
#
# The bar must be SYMMETRIC - both sides must reach the same verdict, or one
# side's knights get a memory of a battle the other side's knights do not.
# num_total_troops and warscore_value are combat-scoped and symmetric for free.
# Blood is made symmetric by the explicit OR across enemy_side. The odds row is
# symmetric because it asks "was either host much larger", not "were we
# outnumbered". The game rule is the same on both sides.
#
# CALIBRATED 2026-08-13 against 39 combat ends, both sides logged separately.
# Vanilla's numbers turned out to be the wrong shape entirely, not merely the
# wrong size: THREE of the five rows never fired once in 39 battles. What the
# ladders showed, and what each number is now:
#
#   size   under 5k 26 / 5-10k 8 / 10-20k 5 / 20k+ NONE
#          Vanilla's 20k and 30k were dead rows. 10k/20k instead. 10k is the
#          top 13% of this sample; 20k fires on nothing yet and is deliberately
#          reserved for late-campaign hosts, which are much larger than the
#          early-game armies this run measured. Do not lower it to fit a
#          bookmark start - it is the one row meant to age into relevance.
#   blood  under 25 3 / 25-50 29 / 50-75 7 / 75+ NONE
#          75 was a dead row. 50 is the top 18%.
#   odds   even 9 / 1.2x 9 / 1.5x 11 / 2x 10
#          1.5x fired on 21 of 39 - FIFTY-FOUR PERCENT. It was carrying the
#          whole trigger and calling half of all battles remarkable. 2x is the
#          top 26%.
#   score  under 10 30 / 10-25 3 / 25-50 2 / 50+ 4
#          25 already sat right at the top 15%. The only threshold kept.
#
# The three ratio rows - blood, odds, warscore - are scale free and will hold
# as hosts grow. Size is the only absolute, which is why its two rungs are set
# high rather than fitted to this sample.
#
# Re-derive by restoring the ladders from git history, not by guessing.

bc_battle_is_notable_trigger = {
	calc_true_if = {
		amount >= 2

		combat = { num_total_troops >= 10000 }
		combat = { num_total_troops >= 20000 }
		OR = {
			percent_enemies_killed >= 50
			enemy_side.percent_enemies_killed >= 50
		}
		OR = {
			enemy_side.side_max_army_size >= bc_our_host_gap_value
			side_max_army_size >= bc_their_host_gap_value
		}
		combat = { warscore_value >= 25 }
		bc_was_wipe_trigger = yes
		has_game_rule = bc_aftermath_generous
	}
}

# Is the man across the field the man who has beaten me before?
#
# CALLED ON THE COMMANDER, in TRIGGER contexts only - weight multipliers and
# is_valid blocks. It must therefore be safe to evaluate on every commander in
# the world on every phase event roll, most of whom have no nemesis at all.
#
# THE TWO `exists` ROWS ARE SAFE AS SIBLINGS - `exists` on a missing variable is
# safe, and only a READ of a missing variable logs an error. The trigger_if
# repeats them and does the reading, which is the shipped pattern this mod
# already uses in the graders' every_side_knight limit.
#
# IT MISSES THE FIRST ROLL OF EVERY BATTLE, deliberately. bc_battle_foe is
# written in the EFFECT of a phase event and weights are evaluated BEFORE the
# effect, so the first roll of a battle sees nothing and every roll after sees
# this battle's foe. Reading the combat scope directly instead would fix that at
# the cost of an unverified scope path evaluated on every roll of every battle in
# the world.
#
# SINCE bc_write_bond_effect, ONE CALLER IS NOT A WEIGHT BLOCK. That effect - in
# bc_context_effects.txt, called from bc_battle_open_effect - asks this in the
# `limit` of its first `if`. Still a trigger context, so the safety argument
# above holds unchanged, but two things above are now local to the weight call
# sites and not to this trigger:
#   - "TRIGGER contexts only - weight multipliers and is_valid blocks" now also
#     means the limit of an if in an effect.
#   - "IT MISSES THE FIRST ROLL OF EVERY BATTLE" is true of weights and FALSE
#     from bc_write_bond_effect, which runs after the foe block in the same
#     effect and therefore sees bc_battle_foe on the very first moment.
# The second one matters: do not "fix" the first-roll gap by moving the foe
# write, because the bond ladder already depends on the write being upstream.
bc_facing_nemesis_trigger = {
	exists = var:bc_nemesis
	exists = var:bc_battle_foe
	trigger_if = {
		limit = {
			exists = var:bc_nemesis
			exists = var:bc_battle_foe
		}
		var:bc_nemesis = var:bc_battle_foe
		var:bc_nemesis = { is_alive = yes }
	}
}

# --- was this an annihilation ------------------------------------------------
#
# READS scope:wipe, WHICH THE ENGINE PASSES TO BOTH COMBAT-END HOOKS. AGOT
# combat_on_actions.txt:13 documents it above on_combat_end_winner and :527 above
# on_combat_end_loser, so both graders see the same value and any bar built on it
# is SYMMETRIC for free - which bc_battle_is_notable_trigger's header requires.
#
# IT SURVIVES INTO AN EVENT RAISED FROM THE `events = { }` LIST. Proven by
# vanilla: combat_event.1001 is raised from on_combat_end_winner's list
# (AGOT combat_on_actions.txt:17) and reads `scope:wipe = yes` at AGOT
# events/war_events/combat_events.txt:781 and :947. Nothing has to thread it.
#
# SCOPE-INDEPENDENT. A saved scope is readable from any scope in the event, so
# this trigger is equally valid on the combat side (the notability row) and on
# the commander inside side_commander ?= { } (bc_style_step_value).
#
# TRIGGER_IF, NOT THE SPEC'S `AND = { exists = scope:wipe  scope:wipe = yes }`.
# The conjunction is right and the sibling shape is wrong. Shipped script uses
# BOTH idioms - `exists` at combat_on_actions.txt:100, `= yes` at :758 - which
# means it is unsettled whether the scope always exists carrying a boolean or
# exists only when true. Under the second reading the spec's AND compares against
# a missing scope on EVERY non-wipe combat end in the world, which is nearly all
# of them, and CK3 does not short-circuit and guarantees no evaluation order among
# siblings. `exists` is safe on a missing scope; the COMPARISON is what has to be
# nested. Same rule, same reason, as every variable guard in plans 3 and 4.
bc_was_wipe_trigger = {
	exists = scope:wipe
	trigger_if = {
		limit = { exists = scope:wipe }
		scope:wipe = yes
	}
}

# --- is the man across the field the one who has beaten me before -----------
#
# THE OPENING'S read-back, and it is NOT bc_facing_nemesis_trigger. That one reads
# var:bc_battle_foe live, which is right for an effect that runs inside a moment.
# This one is read from an EVENT DESC, and a desc is rendered when the player gets
# round to the pop-up - possibly after the battle has ended and
# bc_clear_scratch_effect has removed the variable. A saved scope is not cleared,
# so scope:bc_eve_foe still names the right man; var:bc_battle_foe might name
# nobody at all.
#
# THE SCOPE IS WRITTEN FROM THE VARIABLE, one line before the raise, in
# bc_eve_offer_effect. So it is this battle's opposing commander, not a stale one -
# the freshness comes from the variable, the durability from the scope.
#
# CALLED ON THE COMMANDER, in a TRIGGER context - an event desc branch. It must be
# safe for a commander who has no nemesis, which is nearly all of them, and safe
# when the other side is leaderless and nothing saved the scope.
#
# THE TWO `exists` ROWS ARE SAFE AS SIBLINGS. `exists` on a missing variable or a
# missing scope is safe; only a READ has to be nested, which is what the
# trigger_if does.
#
# NO is_alive ROW, unlike bc_facing_nemesis_trigger. The eve's foe is commanding a
# battle that is starting, so he is alive by construction; that trigger reads a
# variable which can outlive its subject.
bc_eve_facing_nemesis_trigger = {
	exists = var:bc_nemesis
	exists = scope:bc_eve_foe
	trigger_if = {
		limit = {
			exists = var:bc_nemesis
			exists = scope:bc_eve_foe
		}
		var:bc_nemesis = scope:bc_eve_foe
	}
}

# He wears a white cloak.
#
# SEVEN ROWS BECAUSE AGOT MODELS THE KINGSGUARD AS SEVEN SEPARATE COUNCIL
# POSITIONS, not as one with seven holders: `kingsguard_lord_commander` plus
# `kingsguard_1` through `kingsguard_6`, all defined in
# common/council_positions/00_agot_council_kingsguard.txt. There is no single
# key that means "is in the Kingsguard", so the OR is the whole of the
# available answer and dropping any row silently loses a seventh of the order.
#
# NOT A TRAIT AND NOT A TITLE. The temptation is to look for a `kingsguard`
# trait; there is none. The temptation after that is to test for a knighthood
# or a vow trait, which would collect a great many men who are not in the
# order at all.
#
# NO SCOPE WALK AND NO GUARD. `has_council_position` reads off the character
# in scope and answers no for a character who holds none, so this is one of the
# few triggers in this file that needs no `exists` and no trigger_if. It is
# safe from any character scope, including one entered from another.
#
# THE LORD COMMANDER IS NOT SEPARATED OUT HERE. If a template ever wants him
# alone, gate on `kingsguard_lord_commander` directly rather than adding a
# second trigger - one row is not worth a name.
bc_is_kingsguard_trigger = {
	OR = {
		has_council_position = kingsguard_lord_commander
		has_council_position = kingsguard_1
		has_council_position = kingsguard_2
		has_council_position = kingsguard_3
		has_council_position = kingsguard_4
		has_council_position = kingsguard_5
		has_council_position = kingsguard_6
	}
}

# --- has he been fighting the same battle over and over ---------------------
#
# THE MOD'S OWN MEMORY, READ BACK ONTO THE FIELD FOR THE FIRST TIME.
# `bc_style_streak` is written by bc_style_streak_effect in
# bc_reputation_effects.txt at the END of every notable battle, holds the number
# of consecutive notable battles this commander has finished in the same style,
# and is never cleared. Until this trigger, nothing in the mod ever read it
# except bc_reputation_effects.txt itself, which uses it to decide whether to
# award a commander trait at three. The mod has been keeping a record of how a
# man fights for the whole of a campaign and has never once let the record
# affect anything a player sees during a battle.
#
# `>= 2`, NOT `>= 3`. Three is the reward threshold, and a trigger that fired at
# three would be describing a man who is about to be handed a trait for it - the
# reward chain already speaks at that point. Two is the first number at which
# "again" is true, which is the whole of what this asks.
#
# THE VALUE CANNOT INCLUDE THE BATTLE BEING FOUGHT. bc_style_streak_effect runs
# from the combat-end chain, so mid-battle the variable holds the count as of the
# LAST finished battle. That is the correct reading for a template about a habit:
# the habit is what he brought with him this morning.
#
# `exists` AS A SIBLING, THE COMPARISON NESTED - the standing rule of this file
# and of every variable guard in the mod. `exists` is safe on a missing variable;
# a READ is not, and CK3 guarantees no evaluation order among siblings. Nearly
# every commander in the world has fought no notable battle at all, so the naive
# `AND = { exists = var:bc_style_streak  var:bc_style_streak >= 2 }` would log a
# missing-variable read for almost every commander on almost every phase.
#
# READ ON THE COMMANDER. bc_style_streak lives on the character the seam calls
# `side_commander`; a knight template that wants it must walk there first, the
# way bc_knight_he_has_lost_count walks to var:bc_battle_moments.
bc_fights_the_same_way_trigger = {
	exists = var:bc_style_streak
	trigger_if = {
		limit = { exists = var:bc_style_streak }
		var:bc_style_streak >= 2
	}
}


# --- has he won a great many of these ---------------------------------------
#
# `bc_notable_battles` IS A TALLY OF VICTORIES, NOT OF EXPERIENCE, and getting
# that wrong is the only way to misuse this trigger. bc_notable_battle_counter_effect
# lives in bc_reward_effects.txt and is called from ONE site - the winner's arm
# of the aftermath chain in bc_aftermath_events.txt:199. A man who has lost nine
# notable battles has no variable at all. So this asks "has he won five", never
# "has he fought five", and no template gated on it may put a defeat in his
# past on the strength of the number.
#
# THE COUNTER IS CUMULATIVE AND IS NEVER CLEARED, which is what separates it
# from vanilla's combat_event_0001_tracker - that one counts CONSECUTIVE wins
# and resets to 0 on any defeat. A defeat does not reduce this one, it simply
# fails to grow it, so the number is a career and not a run of form.
#
# FIVE, WHICH IS ABOVE EVERY OTHER THRESHOLD THE MOD READS OFF THIS VARIABLE.
# The nickname gate in bc_reputation_effects.txt uses `>= 3`; a trigger at
# three would be true of every commander the mod has already named, and the
# point of a gate is to select a smaller set than the one already spoken for.
#
# IT LIVES ON THE COMMANDER, written inside the aftermath's side_commander
# scope, so this trigger is only meaningful when read there.
#
# `exists` AS A SIBLING, THE COMPARISON NESTED - the standing rule. Most
# characters in the world have never commanded a notable battle and have no
# variable, so a sibling comparison would log a missing-variable read on nearly
# every phase of nearly every battle.
bc_won_many_before_trigger = {
	exists = var:bc_notable_battles
	trigger_if = {
		limit = { exists = var:bc_notable_battles }
		var:bc_notable_battles >= 5
	}
}

# --- how deep is the debt to the man across the field -----------------------
#
# READ THIS WITH bc_facing_nemesis_trigger, NEVER ALONE. `bc_nemesis_losses`
# counts the losses owed to whoever `bc_nemesis` names, and that man is not
# necessarily the man on the other side of this battle. Alone this trigger says
# only "he is carrying a deep grudge against somebody"; with the nemesis test
# beside it, it says the thing worth a moment.
#
# TWO, BECAUSE THREE IS WHERE THE RIVALRY LADDER ITSELF SPEAKS.
# bc_rivalry_set_relation_effect sets the `nemesis` relation at `>= 3` under
# bc_rivalry_nemesis and `rival` at `>= 2`, so a gate at three would be
# describing a man the relation ladder is already describing. Two is the first
# number at which "he has done this to me before" is more than once.
#
# THE COUNTER SKIPS VALUES. Section 5's wipe rule increments by 2, so a
# commander can go from 1 to 3 without ever holding 2 - which is exactly why
# this is a threshold and not an exact match, the same reasoning the rivalry
# ladder's own descending arms are built on.
#
# IT IS NOT SCRATCH. bc_nemesis and bc_nemesis_losses are the rivalry feature
# and bc_clear_scratch_effect must never learn about either, so unlike
# bc_battle_foe they are safe to read at any point in a battle.
#
# `exists` AS A SIBLING, THE COMPARISON NESTED. The invariant that lets
# bc_rivalry_set_relation_effect read this variable bare - its caller wrote it
# one line earlier - does not exist here, and a phase template runs on
# commanders who have never had a nemesis at all.
bc_nemesis_debt_deep_trigger = {
	exists = var:bc_nemesis_losses
	trigger_if = {
		limit = { exists = var:bc_nemesis_losses }
		var:bc_nemesis_losses >= 2
	}
}


# --- has the war kept him in the same country -------------------------------
#
# THE LAST OF THE FOUR COUNTERS TO REACH THE FIELD. `bc_terrain_streak` is
# written by bc_terrain_streak_effect in bc_reputation_effects.txt at the end of
# every notable battle and counts consecutive notable battles fought in the same
# TERRAIN BUCKET - not the same province and not the same terrain key.
# bc_terrain_last holds the bucket, and the effect's own header is explicit that
# it is a bucket "never the terrain key", which is why nothing gated on this may
# name a terrain in prose. Six buckets cover the whole map; a streak means the
# war has kept him in one kind of country, not on one hill.
#
# THREE, WHERE THE REPUTATION LADDER USES TWO. bc_terrain_streak_effect's own
# comment gives the reason for its threshold - "fighting the same ground twice
# is a weaker claim than fighting the same WAY three times, and terrain is not
# chosen" - and awards a terrain trait at two. A gate at two would therefore be
# true of every commander the mod has already paid for it. Three is the first
# number the reward chain has nothing to say about.
#
# TERRAIN IS NOT CHOSEN. A commander goes where the war goes, and no template
# gated on this may suggest he picked the ground or sought it out. What the
# number supports is familiarity, and nothing else.
#
# THE VALUE IS AS OF THE LAST FINISHED BATTLE, like every other counter in this
# family: the streak effect runs from the combat-end chain, so the battle being
# fought is not in it yet.
#
# `exists` AS A SIBLING, THE COMPARISON NESTED - the standing rule.
bc_same_ground_again_trigger = {
	exists = var:bc_terrain_streak
	trigger_if = {
		limit = { exists = var:bc_terrain_streak }
		var:bc_terrain_streak >= 3
	}
}


# --- what kind of war is this battle part of --------------------------------
#
# THE WAR SCOPE HAS BEEN REACHABLE SINCE PLAN 1 AND THE MOD READS ONE THING
# OFF IT. bc_war_is_long_trigger walks `scope:combat_side.combat.combat_war`
# and asks `war_days >= 730`; nothing else in the mod has ever looked at the
# war at all. The 1.19.0.6 dump lists `is_civil_war`, `is_war_leader`,
# `attacker_war_score`, `defender_war_score` and `days_since_max_war_score` on
# that scope, and a battle is never only a battle - it is an afternoon inside
# a quarrel that has a shape of its own.
#
# THE WALK IS bc_war_is_long_trigger'S, EXACTLY. `exists` on the chained scope
# as a sibling, the read nested in a trigger_if. A combat can exist without a
# war behind it - the engine's own scope is optional - so neither of these may
# read the war bare.
#
# `is_civil_war` IS AN ENGINE TRIGGER ON THE WAR, "Check if the scope war is a
# civil war or not", and AGOT overrides nothing about it. It is TRUE of a
# rebellion inside a realm and false of two realms at war, which is the
# distinction the templates below are built on.
bc_war_is_a_rebellion_trigger = {
	exists = scope:combat_side.combat.combat_war
	trigger_if = {
		limit = { exists = scope:combat_side.combat.combat_war }
		scope:combat_side.combat.combat_war = { is_civil_war = yes }
	}
}

# --- is this his own quarrel or somebody else's -----------------------------
#
# `is_war_leader` TAKES A CHARACTER TARGET AND IS READ ON THE WAR, not on the
# character - "is the target character leading one of the sides in the scope
# war?" in the dump. So the shape is a walk to the war and a comparison back
# against the man, not a test on him.
#
# `root` IS THE COMMANDER HERE, and that is safe rather than lucky: a combat
# phase event's is_valid roots on the character the template is about, and this
# file already reads `root.culture` and `root.faith` the same way inside
# bc_foe_is_countryman_trigger and bc_foe_is_heathen_trigger. Called from a
# knight template it would answer about the KNIGHT, which is nearly always no
# and is not what any knight template would mean - so this one is for the
# commander pool.
#
# IT IS NOT "IS HE THE TOP MAN". A war leader may be a rebel, a claimant or a
# minor lord with one county; what the trigger says is that the quarrel is
# HIS - that he is the man who will sign whatever gets signed - and that there
# is nobody further up to carry the blame for the afternoon.
bc_the_war_is_his_trigger = {
	exists = scope:combat_side.combat.combat_war
	trigger_if = {
		limit = { exists = scope:combat_side.combat.combat_war }
		scope:combat_side.combat.combat_war = { is_war_leader = root }
	}
}


# --- the body he arrived with ------------------------------------------------
#
# THE MOD HAS ASKED ABOUT WOUNDS SIX TIMES AND ABOUT ILLNESS NEVER. Three
# knight templates read the `wounded_*` ladder, bc_he_cannot_get_there reads it
# on the commander, and plan 6 writes a wound of its own. Every one of those is
# about damage the fighting did. Nobody in this mod has ever ridden out already
# sick, which in this setting is the likelier of the two ways a great captain
# stops being one.
#
# `has_trait_with_flag = illness` IS THE RIGHT GATE AND A TRAIT LIST IS NOT.
# The flag is carried by sixteen traits in AGOT's own 00_traits.txt - ill,
# pneumonic, great_pox, early_great_pox, lovers_pox, leper, gout_ridden,
# consumption, cancer, typhus, bubonic_plague, smallpox, measles, dysentery,
# ergotism and sickly - and a hand-written OR of sixteen keys would go stale the
# first time AGOT or a patch adds a seventeenth. Vanilla's own has_any_illness
# is written exactly this way in 20_health_triggers.txt:5.
#
# NONE OF THE SIXTEEN CARRIES `disables_combat_leadership`, WHICH IS WHY THIS
# GATE IS REACHABLE AT ALL AND THE WOUND GATE LARGELY IS NOT. _traits.info:75
# says that field means "this character is blocked from being a commander", and
# wounded_1, wounded_2, wounded_3 and infirm all set it. A sick man may be given
# an army; a wounded one may not. That asymmetry is checked, not assumed, and
# the note on bc_he_cannot_get_there carries the other half of it.
#
# NO VARIABLE, SO NO GUARD. This is a plain character trigger on the rooted
# character and needs none of the exists/trigger_if scaffolding the war and
# ledger reads need.
bc_he_is_unwell_trigger = {
	has_trait_with_flag = illness
}


# --- the two armies at the edge of the map -----------------------------------
#
# BOTH OF THESE WRAP AN AGOT TRIGGER RATHER THAN A TRAIT OR A CULTURE KEY, and
# both wrap it for the same reason: the thing being asked about is a set, and
# AGOT already maintains the set.
#
# agot_is_member_of_nights_watch (00_agot_nights_watch_triggers.txt:15) is an OR
# over `nightswatch`, `nightswatch_temp` and `nightswatch_historical`. Writing
# `has_trait = nightswatch` here instead would miss the temporary members - the
# men AGOT's own mega-war content puts on the Wall for a campaign - and every
# historical black brother in the bookmarks, which between them are most of the
# ones a player will ever fight beside.
#
# agot_is_wildling_culture (00_agot_cultural_triggers.txt:62) is
# `has_variable = wildling_culture` READ ON A CULTURE, which is why the wrapper
# below spells `culture = { ... }` around it. AGOT sets that variable on its
# wildling cultures at game start; a hand-written OR of culture keys would miss
# every culture the mod adds later and every one a player converts.
#
# NEITHER NEEDS THE exists/trigger_if SCAFFOLDING. `has_trait` and
# `has_variable` are safe on any character and any culture, and `culture` is
# not an optional link on a living character.
bc_sworn_to_the_wall_trigger = {
	agot_is_member_of_nights_watch = yes
}

bc_he_is_free_folk_trigger = {
	culture = { agot_is_wildling_culture = yes }
}


# --- whose ground this is, legally -------------------------------------------
#
# THE MOD KNOWS FIVE THINGS ABOUT THE GROUND AND NOTHING ABOUT THE OWNER. Every
# bc_field_* trigger walks `scope:combat_side.combat.location` and asks the
# province what it is made of - forest, winter, rough, jungle, open, bloody -
# and not one of them has ever asked whose county it is. A battle in Westeros
# is always in somebody's county and half the time it is in one of the two
# commanders'.
#
# THE WALK IS location -> county -> holder, AND BOTH LINKS ARE GUARDED. The
# `location` guard is the standing one every field trigger already carries. The
# holder guard is not decoration: AGOT has real unheld land - beyond the Wall
# and in the ruins - and an unheld county returns nothing at all here, so the
# read has to be nested under its own `exists` the same way.
#
# `this = root` OR `target_is_liege_or_above = root`, WHICH IS THE REALM TEST
# AND NOT THE TITLE TEST. A commander whose own vassal holds the county is
# fighting on his own land in every sense a battle cares about, and the version
# that only matched a personally-held county would be true almost nowhere -
# most great lords hold six counties and campaign across four hundred.
#
# `root` IS THE COMMANDER, as in bc_foe_is_countryman_trigger and
# bc_the_war_is_his_trigger. From a knight template this would answer about the
# knight and mean something much narrower, so it is commander-pool only.
bc_the_ground_is_his_own_trigger = {
	exists = scope:combat_side.combat.location
	trigger_if = {
		limit = { exists = scope:combat_side.combat.location }
		exists = scope:combat_side.combat.location.county.holder
		trigger_if = {
			limit = { exists = scope:combat_side.combat.location.county.holder }
			scope:combat_side.combat.location.county.holder = {
				OR = {
					this = root
					target_is_liege_or_above = root
				}
			}
		}
	}
}


# --- who walked onto whom ----------------------------------------------------
#
# THE MOD HAS READ THE TWO SIDES' STRENGTHS SINCE PLAN 1 AND HAS NEVER ASKED
# WHICH OF THEM CHOSE TO FIGHT. `is_combat_side_attacker` sits on the combat
# side scope the mod already touches in every one of these triggers, and in
# sixty-six batches nothing has looked at it.
#
# THIS IS NOT THE WAR'S ATTACKER. The war scope has `attacker_war_score` and a
# war attacker of its own, and they are a different fact entirely: a man
# defending against an invasion who marches his host onto a besieging army is
# the ATTACKER of that combat and the DEFENDER of the war. The combat-level
# question is only ever "who walked onto whom this morning", which is the one
# these two templates are about.
#
# NO GUARD, AND THE PRECEDENT IS THE OLDEST TRIGGER IN THE FILE.
# bc_side_is_losing_trigger reads `scope:combat_side.side_strength` bare, and
# has since plan 1 - the combat side scope is supplied by the event itself and
# is not an optional link like the war or the commander.
bc_his_side_attacked_trigger = {
	scope:combat_side = { is_combat_side_attacker = yes }
}

bc_his_side_was_attacked_trigger = {
	scope:combat_side = { is_combat_side_attacker = no }
}


# --- water on the field ------------------------------------------------------
#
# THE SIX GROUND BANDS PARTITION THE TERRAIN AND SAY NOTHING ABOUT WATER. That
# is not an oversight in bc_ground_triggers.txt - the partition is over AGOT's
# 23 terrain helpers and is checked exhaustively by the validator, and a river
# is not a terrain type in CK3, it is a flag on the province. `floodplains` and
# `wetlands` are the closest the bands come and both of them mean soft ground
# rather than a bank with a crossing on it.
#
# THIS TRIGGER DOES NOT GO IN bc_ground_triggers.txt AND MUST NOT. That file
# holds the six bands and nothing else, and the validator counts every
# `agot_is_*_terrain` mention in it against the 23 helpers; anything added
# there breaks the partition check. The field-level wrapper belongs here with
# the other bc_field_* triggers, which is also where it is used from.
#
# `is_riverside_province` IS AN ENGINE TRIGGER ON THE PROVINCE - "is the
# province riverside" - and is a different question from `is_river_province`,
# which asks whether the province IS a river and is true only of the map's
# navigable river tiles, where no battle is ever fought.
#
# THE GUARD IS THE STANDING ONE, copied from bc_field_is_forest_trigger.
bc_field_is_riverside_trigger = {
	exists = scope:combat_side.combat.location
	trigger_if = {
		limit = { exists = scope:combat_side.combat.location }
		scope:combat_side.combat.location = { is_riverside_province = yes }
	}
}


# --- the edge that cannot be turned ------------------------------------------

# a hundred and forty templates, and a grep of the localisation for "sea",
# "shore", "coast", "tide" and "surf" returns nothing at all - every match is
# "seam", "season" or "seat". Westeros is a coastline with some land behind it
# and the chronicle had never once looked at the water.
#
# `is_coastal` IS AN ENGINE TRIGGER ON THE PROVINCE - "is the province a
# coastal province (next to sea, not including lakes)" - and is a different
# question from the landed_title triggers next to it in the dump.
# `is_coastal_county` and `has_coastal_province` both scope on a title and
# would answer about the county, which is not what a battle is fought in.
#
# THIS IS NOT bc_field_is_riverside_trigger. A riverside province has water
# ACROSS it, which is what batch sixty-three was about: crossings, frontage,
# footing. A coastal province has water at ONE EDGE of it, which is a
# completely different tactical fact - it is a flank that does not exist. The
# two can be true of the same province and the two moments below are about the
# edge, never about a crossing.
#
# AND IT IS NOT A TERRAIN BAND. Like the river, the coast is a flag on the
# province rather than one of AGOT's 23 terrain helpers, so it does not go in
# bc_ground_triggers.txt - that file holds the six bands and nothing else and
# the validator counts every `agot_is_*_terrain` mention in it against the 23.
#
# THE GUARD IS THE STANDING ONE, copied from bc_field_is_riverside_trigger.
bc_field_is_coastal_trigger = {
	exists = scope:combat_side.combat.location
	trigger_if = {
		limit = { exists = scope:combat_side.combat.location }
		scope:combat_side.combat.location = { is_coastal = yes }
	}
}


# --- the men who are here for money ------------------------------------------
#
# AGOT IS FULL OF FREE COMPANIES AND THE MOD HAD NEVER ASKED ABOUT ONE. The
# Golden Company, the Second Sons, the Windblown, the Stormcrows - a whole
# category of host that is not a levy, not a lord's sworn men and not a holy
# order, and a hundred and forty templates had treated every commander on the
# field as a man leading people who owe him something.
#
# THE IDIOM IS AGOT'S OWN. 00_government_types.txt:375 defines
# `mercenary_government` and :397 gives it the flag `government_is_mercenary`,
# and AGOT's own scripts test it that way - see 00_agot_lmf_marriage_triggers
# .txt:645 and 00_clothing_triggers.txt:134. `government_has_flag` is a plain
# character trigger in the 1.19.0.6 dump ("checks if the government of the
# character has a specific flag"), so there is nothing optional to guard here.
#
# NOT `is_mercenary_company`, WHICH AGOT ALSO USES. That one scopes on a
# landed_title and would need a `primary_title` walk and a guard to reach the
# same answer, and it answers about the TITLE rather than the man. AGOT's
# marriage triggers use both idioms interchangeably; the flag is the shorter
# one and the one that works on a man who is commanding rather than holding.
#
# THIS IS NOT `gold < 0`, WHICH THE MOD ALREADY READS. bc_knight_nobody_pays_
# _for_him is a man who is personally in debt while he loses. A sellsword
# captain may be the richest man on the field; the question here is what his
# men are, not what his purse is.
bc_he_leads_a_free_company_trigger = {
	government_has_flag = government_is_mercenary
}


# --- what the man across the field is worth ----------------------------------
#
# THE MOD READS THE ENEMY COMMANDER IN SEVEN PLACES ALREADY AND NONE OF THEM
# ASKS THIS. bc_foe_is_kin_trigger, bc_foe_is_countryman_trigger,
# bc_foe_is_heathen_trigger and bc_foe_is_house_trigger read his IDENTITY;
# bc_foe_is_formidable_trigger and bc_foe_is_outmatched_trigger read his SKILL;
# bc_foe_leader_commands_in_person_trigger reads whether he turned up at all.
# All seven are above in this file and all seven make the same `enemy_side`
# walk this one makes. What none of them reads is his STATION, which is a
# different question from all three of those and, in this setting, the loudest.
#
# `enemy_side` AND NOT var:bc_battle_foe, WHICH IS THE RULE THE SEVEN ABOVE
# ALREADY FOLLOW AND THE REASON THEY DO. The variable is written by
# bc_battle_open_effect, which runs inside a moment's `effect` block - so during
# the FIRST `is_valid` of a battle it does not exist yet, and during later ones
# it exists only because some earlier moment happened to fire. A gate that read
# it would be silently conditional on the chronicle having already said
# something, which is not a fact about the battle. `enemy_side` is the engine's
# own answer and needs no bookkeeping: "get enemy (opposite) combat side of
# scoped side", combat_side to combat_side.
#
# THE BOND LADDER DOES NOT ASK IT EITHER. bc_write_bond_effect sorts the foe
# five ways - nemesis, kin, countryman, heathen, stranger - and none of the
# five is "what is he worth": a king and a landless sworn sword both come out
# `stranger` if they are of another culture and the same faith. That is why
# this gate is not redundant with the bond clause the same toast will print
# underneath it, and not redundant with the four identity triggers above.
#
# `highest_held_title_tier` AND `tier_kingdom` ARE ALREADY IN THE MOD, on the
# commander himself, at bc_commander_phase_events.txt:1850. That template's
# header claims "no `is_valid` anywhere in the mod reads a title tier", which
# was true when it was written and stops being true here; the claim is load-
# bearing for ITS distinctness argument only, and this trigger asks about a
# different man, so nothing there needs revising.
#
# THE GUARD IS THE STANDING ONE. An enemy side can be leaderless - the bond
# ladder has an entire `nofoe` arm for it - so the commander is an optional link
# and the READ is nested in a trigger_if that repeats the `exists`.
bc_facing_a_crown_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	trigger_if = {
		limit = { exists = scope:combat_side.enemy_side.side_commander }
		scope:combat_side.enemy_side.side_commander = {
			highest_held_title_tier >= tier_kingdom
		}
	}
}


# --- the man who is not entirely here ----------------------------------------
#
# THE MOD READS A GREAT DEAL OF THE BODY AND NONE OF THE MIND. It knows about
# wounds, maiming, one eye, greyscale, a bad physique, a good one, illness, age
# and stress - and stress is the closest it has ever come to this, which is not
# close at all. Stress is a thing an afternoon does to a man. This is a thing
# that was true of him at breakfast.
#
# FOUR TRAITS AND ALL FOUR ARE NEEDED. AGOT keeps vanilla's shape here:
# `lunatic_1` and `lunatic_genetic` (00_traits.txt:5981 and :6012) and
# `possessed_1` and `possessed_genetic` (:6048 and :6080). The `_1` and the
# `_genetic` of each pair are SEPARATE TRAIT KEYS sharing a
# `group_equivalence`, not two levels of one key, so an OR of four rows is the
# whole of the available answer and dropping either genetic row silently loses
# every man who was born to it - which in this setting is most of the
# interesting ones.
#
# THIS IS NOT bc_he_is_unwell_trigger. That one is `has_trait_with_flag =
# illness` and none of these four carries that flag; they carry
# `category = health`, which is a UI grouping and not a flag. The two gates are
# disjoint on the trait list, so a template on either can never be a subset of
# a template on the other.
#
# NO SCOPE WALK AND NO GUARD. `has_trait` answers no for a character who does
# not have the trait, so this is one of the plain character triggers.
bc_his_mind_is_not_his_own_trigger = {
	OR = {
		has_trait = lunatic_1
		has_trait = lunatic_genetic
		has_trait = possessed_1
		has_trait = possessed_genetic
	}
}


# --- what the ground underfoot is worth --------------------------------------
#
# THE MOD HAS ASKED SIX QUESTIONS ABOUT THE FIELD AND NOT ONE OF THEM ABOUT ITS
# WORTH. bc_ground_triggers.txt sorts the field into six terrain bands - winter,
# desert, forest, jungle, rough, open - and bc_scripted_triggers.txt adds a
# river and a coast. Every one of those is a question about the SHAPE of the
# place: what it does to a line of men, where a horse can go, what hides an
# ambush. None of them asks what the place is, who lives on it, or what it was
# worth before two armies walked onto it. A rich county and a poor one look
# identical to all eight.
#
# THE ANSWER IS ALREADY IN THE SAVE AND IS FREE. `development_level` is a
# landed_title trigger - "does the county title have the required county
# development level" - and every county in the game carries one. AGOT sets them
# in history: of the twelve hundred and seventy-six counties it writes a level
# for, a hundred and forty sit at five or below and two hundred at twenty or
# above, with the great mass between. Those two bands are the two ends of a
# distribution that already exists, not thresholds invented here, and each is
# common enough to be met and rare enough to be worth a toast.
#
# RAW NUMBERS AND NOT THE NAMED VALUES, DELIBERATELY. Vanilla defines
# terrible/bad/medium/good/great_development_level at 10/20/40/60/80
# (00_basic_values.txt:1406-1410) and AGOT's own scripts use them. Those numbers
# are tuned for a vanilla game that runs to 1453 and develops for eight hundred
# years. AGOT starts in 300 AC with almost nothing above thirty, so
# `medium_development_level` would gate on a condition that essentially never
# holds and `great_development_level` on one that never holds at all. Twenty and
# five are read off AGOT's actual history files. If AGOT ever reflows its
# development history these two numbers are the thing to re-check.
#
# TWO BANDS AND A GAP, WHICH IS THE POINT. The two triggers are disjoint by
# construction - nothing is both at or below five and at or above twenty - so a
# template on one can never be a subset of a template on the other, and the
# nine hundred and thirty-six counties in between get neither, which is what
# keeps this from firing on every battle in Westeros.
#
# TWO OPTIONAL LINKS AND SO TWO GUARDS, NESTED. `combat.location` is the
# standing one this file already guards eight times over. `county` is the second
# and is genuinely optional: AGOT's own hunt.txt writes `exists = county` before
# `county.development_level` at activity_types/hunt.txt:1928, which is the
# engine telling us a province need not resolve to a county title. Each `exists`
# sits as a sibling and each READ is nested in a trigger_if that repeats it.
bc_field_is_rich_country_trigger = {
	exists = scope:combat_side.combat.location
	trigger_if = {
		limit = { exists = scope:combat_side.combat.location }
		scope:combat_side.combat.location = {
			exists = county
			trigger_if = {
				limit = { exists = county }
				county = { development_level >= 20 }
			}
		}
	}
}

bc_field_is_poor_country_trigger = {
	exists = scope:combat_side.combat.location
	trigger_if = {
		limit = { exists = scope:combat_side.combat.location }
		scope:combat_side.combat.location = {
			exists = county
			trigger_if = {
				limit = { exists = county }
				county = { development_level <= 5 }
			}
		}
	}
}


# --- the chronicle looking across the field at its own handwriting ------------
#
# NINE QUESTIONS ABOUT THE MAN ACROSS THE FIELD AND ALL NINE ARE ABOUT WHO HE
# IS OR WHAT HE IS. bc_foe_is_kin_trigger, bc_foe_is_countryman_trigger,
# bc_foe_is_heathen_trigger and bc_foe_is_house_trigger ask his identity.
# bc_foe_is_formidable_trigger and bc_foe_is_outmatched_trigger ask his skill,
# which is a number he had before breakfast and will still have tomorrow.
# bc_foe_leader_commands_in_person_trigger asks whether he turned up.
# bc_facing_a_crown_trigger asks his station. bc_facing_nemesis_trigger asks
# the ledger what the two of them have done to each other in the past. Not one
# of the nine asks HOW HIS AFTERNOON IS GOING.
#
# THE MOD ALREADY KNOWS THE ANSWER AND HAS NEVER ONCE LOOKED IT UP. The rungs
# are character modifiers - bc_faltering_minor, _major, _decisive - and the
# commander templates run for BOTH sides of a combat, so an enemy commander who
# has had a bad hour is carrying the mod's own marks for it at the moment this
# gate is evaluated. This is the chronicle reading its own handwriting from the
# other end of the field, which nothing in the mod has done before.
#
# THIS IS NOT bc_foe_is_outmatched_trigger AND THE DIFFERENCE IS THE WHOLE
# POINT. That one compares martial scores: a permanent fact about a man, true
# in the same way at every battle he will ever fight, and true of him this
# morning before either host formed up. This one is true only today and only
# since some particular thing went wrong. A brilliant commander having a
# catastrophic afternoon answers no to that trigger and yes to this one, and
# that man is the interesting one.
#
# ONLY THE FALTERING SIDE, DELIBERATELY. A matching momentum trigger would be
# trivial to write and there is nothing yet for it to gate, and a trigger with
# no caller is a thing a later reader has to work out the intent of. If a
# momentum-side moment is ever written, its trigger goes here next to this one.
#
# THE GUARD IS THE STANDING ONE. An enemy side can be leaderless - the bond
# ladder has an entire `nofoe` arm for it - so the commander is an optional
# link, the `exists` sits as a sibling and the READ is nested in a trigger_if
# that repeats it. `enemy_side` and not `var:bc_battle_foe`, for the reason set
# out above bc_facing_a_crown_trigger.
bc_foe_commander_is_faltering_trigger = {
	exists = scope:combat_side.enemy_side.side_commander
	trigger_if = {
		limit = { exists = scope:combat_side.enemy_side.side_commander }
		scope:combat_side.enemy_side.side_commander = {
			bc_is_faltering_trigger = yes
		}
	}
}


# --- the men opposite were not trained on the same field ----------------------
#
# THE MOD HAS ONE CULTURE QUESTION AND IT IS A YES-OR-NO ABOUT ONE MAN.
# bc_foe_is_countryman_trigger asks whether the two commanders share a culture
# key, which is a question about kinship of speech and nothing else, and
# bc_he_is_free_folk_trigger asks the one specific `agot_is_wildling_culture`.
# Beyond those two the mod has never once cared where anybody learned to fight,
# which in a setting with an entire second continent in it is a large omission.
#
# `agot_is_westerosi_culture` IS AGOT'S OWN ANSWER AND IS CULTURE-SCOPED.
# 00_agot_cultural_triggers.txt:163 defines it as high_valyrian,
# westerosi_valyrian, salt_dornish or first_man, or any parent culture or above
# being one of those - which is the whole descent tree of the continent and not
# a hand-written list of culture keys that would rot the first time AGOT added
# one. It is called from a CHARACTER scope as `culture = { ... = yes }`; AGOT
# does exactly that itself at 00_agot_first_ranger_laamp_triggers.txt:159.
#
# BOTH HALVES ARE NEEDED. Root must be of the continent and the man opposite
# must not be. Dropping the root half would fire this on two Braavosi captains
# fighting each other in the Disputed Lands, where nothing about the other
# man's drill is unfamiliar to anybody.
#
# THIS IS DISJOINT FROM bc_foe_is_countryman_trigger BY CONSTRUCTION, since two
# men on opposite sides of that descent tree cannot share a culture key, so no
# template on this can ever be a subset of bc_the_same_tongue or
# bc_knight_they_sound_like_home. It OVERLAPS bc_foe_is_heathen_trigger without
# being contained by it in either direction - the Faith and R'hllor cross the
# narrow sea in both directions - and overlap is not subset.
#
# IT ASKS ABOUT THE COMMANDER OPPOSITE AND USES HIM AS A PROXY FOR HIS HOST,
# which is the same simplification the other nine foe triggers make and is
# stated here so nobody later mistakes it for an oversight. The mod has no
# handle on the composition of an enemy army; it has a handle on the man
# leading it.
#
# THE GUARD IS THE STANDING ONE. An enemy side can be leaderless, so the
# commander is an optional link, the `exists` sits as a sibling and the READ is
# nested in a trigger_if that repeats it. `enemy_side` and not
# `var:bc_battle_foe`, for the reason set out above bc_facing_a_crown_trigger.
bc_foe_is_from_across_the_sea_trigger = {
	culture = { agot_is_westerosi_culture = yes }
	exists = scope:combat_side.enemy_side.side_commander
	trigger_if = {
		limit = { exists = scope:combat_side.enemy_side.side_commander }
		scope:combat_side.enemy_side.side_commander = {
			NOT = { culture = { agot_is_westerosi_culture = yes } }
		}
	}
}


# --- the season, which is not the terrain ------------------------------------
#
# THE MOD HAS A WINTER AND IT IS THE WRONG ONE. bc_field_is_winter_trigger
# wraps bc_ground_is_winter_trigger, which is `agot_is_frozen_flats_terrain` or
# `agot_is_glacier_terrain` - a TERRAIN band, a permanent property of a few
# provinces at the top of the map, as true in the eighth year of a summer as in
# the third year of a winter. In a setting whose entire motto is about a season
# arriving, the mod has never once asked whether it has.
#
# `has_winter_combat_trigger` IS AGOT'S OWN ANSWER AND IS THE RIGHT ONE OF THE
# THREE IT OFFERS. 00_weather_triggers.txt:63 defines it as exactly the three
# province modifiers winter_mild_modifier, winter_normal_modifier and
# winter_harsh_modifier - the engine's seasonal state and nothing else. Its
# sibling `has_winter_trigger` at :43 adds an AGOT clause granting PERMANENT
# winter to everything in world_westeros_the_north outside the Neck and to all
# of world_westeros_beyond_the_wall, which would make this gate always-true for
# every battle a northern lord ever fights and would turn a moment about a
# season arriving into a moment about a map region. AGOT itself separates the
# two and names the tighter one for combat; this follows it.
#
# THIS IS NOT bc_field_is_winter_trigger AND NEITHER CONTAINS THE OTHER. A
# glacier in the fourth year of a long summer answers yes to that one and no to
# this. A Riverlands wheat field in a hard winter answers no to that one and
# yes to this. They overlap and overlap is not subset, so a template here can
# never be dead code under bc_winter_bites or bc_knight_stands_in_the_snow -
# and in any case those two own the argument that cold HURTS, which is not the
# argument either template on this trigger makes.
#
# THE SEVERITY BANDS ARE DELIBERATELY NOT SPLIT. winter_harsh_modifier alone
# would be a strict subset of this, so a template gated on harsh would be dead
# code under any template gated on this unless the side rows were disjoint -
# which is a trap worth not laying. One trigger, two opposite side rows.
#
# THE GUARD IS THE STANDING ONE for combat.location, which this file already
# guards a dozen times over.
bc_season_is_winter_trigger = {
	exists = scope:combat_side.combat.location
	trigger_if = {
		limit = { exists = scope:combat_side.combat.location }
		scope:combat_side.combat.location = { has_winter_combat_trigger = yes }
	}
}


# --- this is not the only war he is in ---------------------------------------
#
# THE MOD ASKS FIVE QUESTIONS ABOUT THE WAR AND ALL FIVE ARE ABOUT THIS ONE.
# bc_war_is_long_trigger asks how long it has run, bc_war_is_a_rebellion_trigger
# what kind it is, bc_the_war_is_his_trigger and bc_this_is_his_own_war_trigger
# whose it is, bc_fighting_his_lords_war_trigger whose it is not. Every one of
# them takes for granted that there is a war, singular, and that the man on
# this field is inside it and nothing else. In a setting whose defining
# conflict is called the War of the Five Kings, that is a peculiar assumption.
#
# `any_character_war = { count >= 2 }` IS THE WHOLE TRIGGER. The engine's own
# list - "Wars of the scoped character" - and a count. No walk, no guard, no
# bookkeeping.
#
# IT IS ROOT-RELATIVE ON PURPOSE, LIKE bc_side_has_kin_knight_trigger. Called
# bare it asks about the commander. Called inside a `scope:combat_side.
# side_commander = { }` walk it asks the same question about a knight's lord,
# which is the only way a knight ever gets to be in this moment.
#
# A LANDLESS MAN AND A HEDGE KNIGHT BOTH ANSWER NO, AND THAT IS CORRECT RATHER
# THAN A LIMITATION. `any_character_war` lists wars the character is a
# PARTICIPANT in - attacker or defender - and a vassal sending levies to his
# liege's war is not one. The subject of this axis is a man who OWNS more than
# one war and has to divide himself between them, which is a thing only that
# man can do.
#
# THIS IS NOT bc_side_is_outnumbered_trigger. That is a ratio on this field
# today and says nothing about why. This is a fact about the man's whole
# situation that would be equally true if he had brought twice as many.
bc_he_has_another_war_trigger = {
	any_character_war = { count >= 2 }
}


# --- what is behind each of them if he does not get up -----------------------
#
# THE MOD KILLS AND MAIMS PEOPLE AND HAS NEVER ASKED WHAT THAT COSTS ANYBODY.
# Plan 6's wound gate can put a man on his back and the aftermath events can
# bury him, and in ninety-five commander gates and sixty-three knight gates
# nothing has ever once looked at who would be standing in his place a month
# later. Succession is the engine's largest single subject and the chronicle
# has been fighting battles in front of it with its back turned.
#
# TWO TRIGGERS BECAUSE THE TWO MEN OCCUPY OPPOSITE ENDS OF THE SAME FACT. One
# asks what happens to what a man already holds. The other asks what a man is
# waiting to be given. They are not two bands of one scale and neither is a
# subset of the other - a childless landed lord whose heir is a nephew answers
# yes to the first and no to the second, and an eldest son with no land of his
# own answers no to the first and yes to the second.
#
# `player_heir` IS AN OPTIONAL LINK AND GETS THE STANDING GUARD. An unlanded
# commander has none, which makes the gate false for him, which is right: the
# moment is about a man with something to hand on. `is_adult` is the engine's
# own line and not an age this mod picks, so it moves if a rule set moves it.
#
# THIS IS NOT bc_the_green_boy, WHICH READS `age <= 21` ON THE COMMANDER
# HIMSELF. That template is about a young man leading. This one is about an old
# enough man whose replacement is a child, and the two would fire on the same
# battle from opposite directions without either being the other.
bc_his_heir_is_a_child_trigger = {
	exists = player_heir
	trigger_if = {
		limit = { exists = player_heir }
		player_heir = { is_adult = no }
	}
}

# `any_heir_title` IS THE ENGINE'S OWN LIST - "all titles the scoped character
# is heir to" - and a count of one is the whole question. No walk and no guard:
# an empty list answers no.
#
# THIS IS NOT bc_knight_no_house_behind_him, bc_knight_twice_as_good OR
# bc_knight_the_squires_best_day. Those read `is_lowborn`, the bastard trait
# and the squire trait: three statements about where a man came FROM. This one
# is about where he is going, and a legitimate second son with an elder brother
# alive answers no to all four.
bc_he_stands_to_inherit_trigger = {
	any_heir_title = { count >= 1 }
}


# --- the body he was dealt ---------------------------------------------------
#
# THE MOD READS WHAT A MAN CAN DO AND HAS NEVER READ WHAT HE WAS BORN WITH.
# `prowess` and `martial` are on nearly every chance block in the file, the
# illness flag is read by bc_he_is_unwell_trigger, and both of those are about
# a man's condition today. Nothing in ninety-six commander gates or sixty-four
# knight gates has ever asked what he has been carrying since he was born and
# what everyone around him decided about it before he could hold anything.
#
# `num_of_bad_genetic_traits` IS THE ENGINE'S OWN COUNT and the right test
# because it is deliberately indiscriminate. It covers a clubfoot and a stutter
# equally, a hunched back and a dull wit, and the moment is the same in every
# case: not the defect, but the thing that was decided about him on account of
# it. A gate written as a list of specific traits would be a gate about bodies.
# This one is a gate about being written off.
#
# THIS IS NOT bc_he_is_unwell_trigger, WHICH READS `has_trait_with_flag =
# illness`. That is a fever or a wasting sickness - a thing that arrived, that
# a man was without last year and may be without next year. This is the thing
# that was there first and will outlast the war. The two sets do not overlap,
# and a man can be in both at once without either moment being the other.
#
# ONE TRIGGER, TWO OPPOSITE SIDE ROWS, for the same reason batch seventy-six
# used one winter trigger rather than two severity bands: a second gate reading
# `num_of_bad_genetic_traits >= 2` would be a strict subset of this one and
# would be dead code wherever the side rows agreed.
bc_his_body_was_dealt_to_him_trigger = {
	num_of_bad_genetic_traits >= 1
}


# --- what his own faith makes of him -----------------------------------------
#
# THE MOD READS FAITH FORTY-THREE TIMES AND EVERY ONE OF THEM IS ABOUT THE MAN
# OPPOSITE. bc_foe_is_heathen_trigger asks whether the far side prays to
# something else; the eve and aftermath events ask what a victory or a burial
# is worth to a faith. Not one of them has ever turned the question round and
# asked what a man's OWN faith has already concluded about the man himself.
#
# `num_sinful_traits` AND `num_virtuous_traits` ARE THE ENGINE'S OWN COUNTS AND
# THEY ARE EVALUATED AGAINST HIS OWN FAITH BY DEFAULT. Both accept an optional
# `target = scope:faith` form for asking what somebody else's faith would make
# of him; neither wrapper uses it, because the subject is what he was raised
# inside and not what a stranger would say.
#
# TWO TRIGGERS AND NEITHER IS THE OTHER'S OPPOSITE. This is the important part.
# A man is not scored on one axis from sinful to virtuous - the counts are two
# independent tallies over the same trait list, and a man with three sins and
# three virtues answers yes to both, while a man with a mild temperament and no
# strong traits at all answers no to both. Writing one trigger and negating it
# for the second moment would have been wrong and would have quietly made the
# two templates mutually exclusive when they are not.
#
# THE THRESHOLD IS TWO, NOT ONE, ON BOTH SIDES. One sinful trait is most adult
# characters in the game under most faiths, and a gate that fires for nearly
# everybody is not a gate. Two is a man his septon has a settled opinion about.
#
# BOTH ARE ROOT-RELATIVE and both are used bare, on the commander in one
# template and on the knight in the other - they read the scoped character and
# nothing else, so there is no walk and no guard to get wrong.
bc_his_own_faith_counts_against_him_trigger = {
	num_sinful_traits >= 2
}

bc_his_own_faith_counts_for_him_trigger = {
	num_virtuous_traits >= 2
}


# --- what he was raised to be ------------------------------------------------
#
# `has_education_*` APPEARS ZERO TIMES IN THIS MOD. Ninety-eight commander
# gates and sixty-six knight gates read what a man can do - prowess, martial,
# learning, the trait list, the illness flag, the count of bad genetic traits -
# and not one of them has ever read what somebody spent his childhood trying to
# make him into. Education is the single largest statement the engine makes
# about a character's intended shape and the chronicle has never opened it.
#
# BOTH WRAP AGOT'S OWN TRIGGERS RATHER THAN LISTING TRAITS, for exactly the
# reason bc_sworn_to_the_wall_trigger wraps agot_is_member_of_nights_watch:
# AGOT overrides 00_scripted_triggers.txt and adds a FIFTH tier to every
# education line (00_scripted_triggers.txt:65 and :85 are five-arm ORs, not the
# vanilla four). A hand-written `OR` over education_learning_1 through _4 here
# would silently miss every maester-taught lord in the game with the top rank.
#
# TWO TRIGGERS AND THEY ARE MUTUALLY EXCLUSIVE BY CONSTRUCTION, which is the
# opposite situation from batch eighty's two counts. A character has exactly
# one education trait, so no man ever answers yes to both of these, and the two
# templates can never both be live for the same person - though they roll in
# different pools anyway, so it would not have mattered if they could.
#
# NEITHER IS A NEGATIVE GATE. "Not raised for war" would have been the obvious
# way to write the first one and would have been wrong: it fires for most of
# the peerage, and a gate that fires for nearly everybody is not a gate. Naming
# the education he did get is both narrower and more specific about the man.
bc_he_was_raised_for_books_trigger = {
	has_education_learning_trigger = yes
}

bc_he_was_raised_for_the_dark_trigger = {
	has_education_intrigue_trigger = yes
}
