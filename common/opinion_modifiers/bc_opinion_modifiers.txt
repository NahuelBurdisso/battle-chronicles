# Battle Chronicles - opinion modifiers.
#
# No vanilla or AGOT phase event applies opinion at all. This is the whole point
# of the knight events: a battle should leave grudges and bonds behind it, not
# just a casualty count. Proven working from a knight phase event (probe 11).
#
# All decaying - a battle should fade, slowly.

bc_valiant_in_battle_opinion = {
	monthly_change = -0.1
	decaying = yes
	stacking = yes
}

bc_craven_in_battle_opinion = {
	monthly_change = 0.1
	decaying = yes
	stacking = yes
}

# Slowest decay of the three. This is the one meant to outlive the war.
bc_shieldbrother_opinion = {
	monthly_change = -0.05
	decaying = yes
	stacking = yes
}
