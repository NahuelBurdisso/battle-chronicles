# Battle Chronicles - opinion modifiers.
#
# No vanilla or AGOT phase event applies opinion at all. This is the whole point
# of the knight events: a battle should leave grudges and bonds behind it, not
# just a casualty count. Proven working from a knight phase event (probe 11).
#
# All decaying - a battle should fade, slowly.
#
# `monthly_change` is a decay RATE, not a signed drift, so it must be POSITIVE
# even for a negative opinion. A negative value is rejected with
#   Decaying modifier 'x' should have a duration or monthly_change!
# and the modifier is dropped. The sign of the opinion comes from the `opinion =`
# passed at the add_opinion call site. Verified in game 2026-08-12; every vanilla
# decaying modifier in 00_activity_system_opinons.txt uses `monthly_change = 0.1`.

bc_valiant_in_battle_opinion = {
	monthly_change = 0.1
	decaying = yes
	stacking = yes
}

bc_craven_in_battle_opinion = {
	monthly_change = 0.1
	decaying = yes
	stacking = yes
}

# Second slowest of the four, and one of the two meant to outlive the war.
bc_shieldbrother_opinion = {
	monthly_change = 0.05
	decaying = yes
	stacking = yes
}

# SLOWEST OF THE FOUR, and the only one that is not a judgement. The other three
# are opinions a man forms about how somebody fought; this is what he owes
# somebody who came and got him. At 0.02 a month a +40 takes over a century and
# a half to reach zero, which is a way of saying it does not decay inside a
# lifetime.
#
# IT IS STILL WRITTEN AS A DECAY AND NOT AS A PERMANENT MODIFIER, for one
# reason: this mod does not model what happens between the two men afterwards,
# and an opinion that literally cannot move would outlast anything that does.
#
# `stacking = yes` LIKE THE OTHER THREE. A knight who does this twice, in two
# different wars, has done it twice.
bc_lifesaver_opinion = {
	monthly_change = 0.02
	decaying = yes
	stacking = yes
}
