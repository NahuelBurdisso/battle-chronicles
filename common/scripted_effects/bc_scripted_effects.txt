# Battle Chronicles - shared effects.
#
# Every event reaches for momentum through the four wrappers below, never through
# add_character_modifier directly. Tuning the whole mod is then one duration, one
# ladder and the two depth rungs in this file.
#
# NOTE: `$PARAM$` substitution inside a quoted string EATS THE SPACE IN FRONT OF
# IT ("applied $MODIFIER$" prints "appliedbc_momentum_major"). Harmless in a log,
# never build a display string on it.

# --- the depth escalator ----------------------------------------------------
#
# THE LADDER HAD COLLAPSED ONTO ITS BOTTOM RUNG. Censused 2026-08-19 across all
# 444 phase-event templates: 411 of the 416 momentum applications ask for
# `minor`, 5 ask for `major`, and NOT ONE asks for `decisive`. The reason is not
# a bug, it is how the 434 templates of the 08-17/08-18 batches were generated -
# magnitude was assigned from swing direction alone, every `down` getting
# bc_faltering_minor_effect and every `up` bc_momentum_minor_effect, so no
# magnitude judgment was ever made at any of them. The advantage = +/-6 and
# +/-10 rungs in bc_momentum_modifiers.txt were dead content.
#
# THE FIX IS NOT TO PROMOTE THE TEMPLATES. Reading the 434 names makes the case:
# they are CONTEXT READS, not field events - "he has three living sisters",
# "the seals are still wet enough", "the name is a weight". A context read is
# worth a minor nudge and nothing more, and asking one of them for a decisive
# swing would be worse design than the flat ladder it replaced. Only the ~14
# hand-authored templates are physical events (banner falls, flank collapses,
# van breaks through, knight unhorses a lord) and those already ask for major.
#
# SO DEPTH ESCALATES INSTEAD. What a moment is worth is not only what it says,
# it is also WHEN IN THE BATTLE it says it. The sixth blow of a long grind lands
# harder than the first regardless of which context read won the roll, and that
# battles build is the whole premise of the mod. The requested rung is now a
# FLOOR and the ledger depth raises it:
#
#   moments 1-3  ->  as requested
#   moments 4-5  ->  one rung up
#   moments 6+   ->  two rungs up
#
# RUNGS 11 AND 16, THE SECOND RE-TUNE, 2026-08-19. History: 4 and 6 were fitted
# to 4.2 moments per battle, stale since the 434-template batch, and gave 41%
# decisive against a 5% target; 8 and 13, fitted to that run, gave 18%. This pair
# is fitted to the 8/13 run - 1432 moments over 192 commander-sides, ~7.5 each,
# P(depth<8)=0.61, P(8-12)=0.21, P(>=13)=0.18. 11 puts the low rung at the 72nd
# percentile. 16 IS A DELIBERATE MISS OF THE 5% TARGET - see the ledger file.
#
# THE MOMENTS COUNT IS READ ON THE SCOPE THIS RUNS ON, WHICH IS ALWAYS RIGHT.
# Knight templates wrap both bc_momentum_*_effect and bc_record_moment_effect in
# the same `scope:combat_side.side_commander ?= { ... }` block, so the modifier
# and the ledger land on one character. No reach-through belongs here, and one
# added would break the commander path.
#
# GUARDED, AND THE ABSENT CASE IS REAL. Pop-up options answered after combat end
# have had bc_battle_moments removed by bc_clear_scratch_effect - the same case
# bc_record_moment_effect documents at length. Nested, not two sibling rows in
# one `limit`: CK3 does not short-circuit.
bc_add_momentum_effect = {
	if = {
		limit = { exists = var:bc_battle_moments }
		if = {
			limit = { var:bc_battle_moments >= 13 }
			debug_log = "BC: momentum depth two"
			bc_momentum_step2_effect = { MODIFIER = $MODIFIER$ }
		}
		else_if = {
			limit = { var:bc_battle_moments >= 9 }
			debug_log = "BC: momentum depth one"
			bc_momentum_step1_effect = { MODIFIER = $MODIFIER$ }
		}
		else = {
			debug_log = "BC: momentum depth none"
			bc_apply_momentum_rung_effect = { MODIFIER = $MODIFIER$ }
		}
	}
	else = {
		debug_log = "BC: momentum depth none"
		bc_apply_momentum_rung_effect = { MODIFIER = $MODIFIER$ }
	}
}

# One rung up. The four escalating arms strip the rungs BELOW the one they are
# about to grant, because add_character_modifier of a DIFFERENT modifier stacks
# where the same one merely refreshes - without the strip a long battle would
# end holding minor + major + decisive at once, which is advantage 19 off a
# ladder whose top rung is 10.
#
# NO UPPER GUARD, DELIBERATELY. The resolved rung is monotonic inside a battle
# because bc_battle_moments only ever increases, so a later moment cannot
# resolve lower than an earlier one asking for the same tier. The one exception
# is the single template in the mod that requests `decisive` outright being
# followed by a minor request in the same battle, which is not worth a guard.
bc_momentum_step1_effect = {
	if = {
		limit = { flag:$MODIFIER$ = flag:bc_momentum_minor }
		remove_character_modifier = bc_momentum_minor
		bc_apply_momentum_rung_effect = { MODIFIER = bc_momentum_major }
	}
	else_if = {
		limit = { flag:$MODIFIER$ = flag:bc_momentum_major }
		remove_character_modifier = bc_momentum_minor
		remove_character_modifier = bc_momentum_major
		bc_apply_momentum_rung_effect = { MODIFIER = bc_momentum_decisive }
	}
	else_if = {
		limit = { flag:$MODIFIER$ = flag:bc_faltering_minor }
		remove_character_modifier = bc_faltering_minor
		bc_apply_momentum_rung_effect = { MODIFIER = bc_faltering_major }
	}
	else_if = {
		limit = { flag:$MODIFIER$ = flag:bc_faltering_major }
		remove_character_modifier = bc_faltering_minor
		remove_character_modifier = bc_faltering_major
		bc_apply_momentum_rung_effect = { MODIFIER = bc_faltering_decisive }
	}
	else = {
		bc_apply_momentum_rung_effect = { MODIFIER = $MODIFIER$ }
	}
}

# Two rungs up. Both non-top rungs land on the same place, so this is two arms
# rather than four.
bc_momentum_step2_effect = {
	if = {
		limit = {
			OR = {
				flag:$MODIFIER$ = flag:bc_momentum_minor
				flag:$MODIFIER$ = flag:bc_momentum_major
			}
		}
		remove_character_modifier = bc_momentum_minor
		remove_character_modifier = bc_momentum_major
		bc_apply_momentum_rung_effect = { MODIFIER = bc_momentum_decisive }
	}
	else_if = {
		limit = {
			OR = {
				flag:$MODIFIER$ = flag:bc_faltering_minor
				flag:$MODIFIER$ = flag:bc_faltering_major
			}
		}
		remove_character_modifier = bc_faltering_minor
		remove_character_modifier = bc_faltering_major
		bc_apply_momentum_rung_effect = { MODIFIER = bc_faltering_decisive }
	}
	else = {
		bc_apply_momentum_rung_effect = { MODIFIER = $MODIFIER$ }
	}
}

# The leaf. Everything funnels here so duration lives in exactly one place.
bc_apply_momentum_rung_effect = {
	add_character_modifier = {
		modifier = $MODIFIER$
		days = 20
	}
}

# --- the swing ladder -------------------------------------------------------
# `bc_swing` shifts which rung an event reaches for instead of scaling numbers.
# subtle steps down, dramatic steps up, and the ladder is defined once in
# bc_momentum_modifiers.txt. An event asking for "minor" under `dramatic` gets
# major; an event asking for "major" under `subtle` gets minor.

bc_momentum_minor_effect = {
	if = {
		limit = { has_game_rule = bc_swing_dramatic }
		bc_add_momentum_effect = { MODIFIER = bc_momentum_major }
	}
	else = {
		bc_add_momentum_effect = { MODIFIER = bc_momentum_minor }
	}
}

bc_momentum_major_effect = {
	if = {
		limit = { has_game_rule = bc_swing_subtle }
		bc_add_momentum_effect = { MODIFIER = bc_momentum_minor }
	}
	else_if = {
		limit = { has_game_rule = bc_swing_dramatic }
		bc_add_momentum_effect = { MODIFIER = bc_momentum_decisive }
	}
	else = {
		bc_add_momentum_effect = { MODIFIER = bc_momentum_major }
	}
}

bc_faltering_minor_effect = {
	if = {
		limit = { has_game_rule = bc_swing_dramatic }
		bc_add_momentum_effect = { MODIFIER = bc_faltering_major }
	}
	else = {
		bc_add_momentum_effect = { MODIFIER = bc_faltering_minor }
	}
}

bc_faltering_major_effect = {
	if = {
		limit = { has_game_rule = bc_swing_subtle }
		bc_add_momentum_effect = { MODIFIER = bc_faltering_minor }
	}
	else_if = {
		limit = { has_game_rule = bc_swing_dramatic }
		bc_add_momentum_effect = { MODIFIER = bc_faltering_decisive }
	}
	else = {
		bc_add_momentum_effect = { MODIFIER = bc_faltering_major }
	}
}

# --- rallying ---------------------------------------------------------------
# Clearing a setback is worth more than adding a bonus, so the rally events
# strip whatever tier is on and then grant a small push.

bc_clear_faltering_effect = {
	if = {
		limit = { has_character_modifier = bc_faltering_minor }
		remove_character_modifier = bc_faltering_minor
	}
	if = {
		limit = { has_character_modifier = bc_faltering_major }
		remove_character_modifier = bc_faltering_major
	}
	if = {
		limit = { has_character_modifier = bc_faltering_decisive }
		remove_character_modifier = bc_faltering_decisive
	}
}

bc_rally_effect = {
	bc_clear_faltering_effect = yes
	bc_momentum_minor_effect = yes
}

# --- cleanup ----------------------------------------------------------------
# Called from the combat-end graders. Without this, momentum earned in one
# battle bleeds into the next 20 days of the war.
#
# FLAT, NOT GUARDED. remove_character_modifier on an absent modifier is a no-op
# — vanilla's own vizierate_remove_extravagance_modifiers_effect
# (00_diarchy_scripted_effects.txt:2318-2335) is an unguarded list of 16 of
# which at most 4 are ever held. Guards here bought nothing and made the list a
# place where a new modifier gets forgotten.
#
# EVERY modifier the mod can apply during a battle must appear here. That is the
# one rule this effect exists to make easy to obey.
bc_cleanup_effect = {
	remove_character_modifier = bc_faltering_minor
	remove_character_modifier = bc_faltering_major
	remove_character_modifier = bc_faltering_decisive
	remove_character_modifier = bc_momentum_minor
	remove_character_modifier = bc_momentum_major
	remove_character_modifier = bc_momentum_decisive
	# Family 2, from bc_battle_modifiers.txt, whose header states the same rule
	# from the other end. NOT in bc_clear_faltering_effect above - that one runs
	# MID-BATTLE from bc_rally_effect, and adding these to it would make every
	# rally strip the character of the fight along with the setback, silently and
	# with no marker.
	remove_character_modifier = bc_bloodied
	remove_character_modifier = bc_shieldwall
	remove_character_modifier = bc_spent_horse
	remove_character_modifier = bc_pursuit
	remove_character_modifier = bc_ground_held
}

# --- pacing -----------------------------------------------------------------
# Every commander chronicle sets this, and every commander chronicle refuses to
# fire while it is set. One shared flag is what actually enforces the design
# budget of "a notable moment roughly every 25-30 days of battle" — phase events
# roll every 5 days (COMBAT_EVENT_DAYS), so without it seven events at base 60
# would fire in a constant stream.

bc_mark_chronicle_effect = {
	add_character_flag = {
		flag = bc_chronicle_recent
		days = 15
	}
}

# The commander-only variant. bc_mark_chronicle_effect above stays exactly as it
# is, at a flat 15 days, because the knight events call it - including
# bc_shieldbrother_forged, which calls it TWICE on two different characters. Do
# not collapse those two calls and do not point them here: a knight has no
# bc_battle_size, so bc_chronicle_flag_days_value would fall to its 15-day else
# arm on every single knight in every combat in the world. It would happen to
# produce the right number and would log a read of nothing, but it would be a
# lie about why.
#
# THE FLOOR DIAG IS FIRST, AND THAT IS LOAD-BEARING. It has to read both pieces
# of state BEFORE add_character_flag rewrites one of them. Every commander event
# calls this effect near the top of its `effect` block - after
# bc_battle_open_effect, before any bc_record_moment_effect, before any
# trigger_event - so here bc_battle_moments still holds the state from BEFORE
# this moment. Put this block below add_character_flag and it is always false.
#
# WHAT THE LIMIT MEANS: is_valid has already passed
# bc_chronicle_ready_commander_trigger. If the cooldown flag is set, that
# trigger's first arm was FALSE, so the only thing that let this event through
# was the floor arm. The two rows below re-test exactly that pair, so the marker
# fires when and only when the floor did work the cooldown would have blocked -
# and reading it needs no attribution of log lines to a particular battle.
bc_mark_chronicle_commander_effect = {
	if = {
		limit = {
			has_character_flag = bc_chronicle_recent
			NOT = { exists = var:bc_battle_moments }
		}
		debug_log = "BC: floor admitted"
	}

	add_character_flag = {
		flag = bc_chronicle_recent
		days = bc_chronicle_flag_days_value
	}
	debug_log = "BC: chronicle mark commander"
}

# --- pop-up throttle --------------------------------------------------------
# One prompt per 60 days per commander. A long siege would otherwise produce a
# stream of them. This is a FLAG on purpose: modifiers are wiped at combat end,
# so a modifier-based cooldown would reset with every battle.

bc_start_popup_cooldown_effect = {
	add_character_flag = {
		flag = bc_popup_cooldown
		days = 60
	}
}

# --- toast budget -----------------------------------------------------------
# 10 DAYS, ON THE RECIPIENT. See bc_can_toast_trigger in bc_scripted_triggers.txt
# for why the recipient needs a budget of his own at all - in short, every other
# cooldown in the mod is taken on the SUBJECT, and a knight moment's subject is
# not the man who gets the message.
#
# WHY 10 AND NOT 15. Phase events roll every COMBAT_EVENT_DAYS = 5, so this is a
# budget of one toast per two rolls, or about three across a 30-day battle. 15
# would have matched bc_chronicle_recent and rounded to two per battle, which is
# fewer moments than a battle with 444 templates behind it deserves; 5 would be
# no throttle at all, since it expires before the next roll. The number is the
# one dial to turn if the volume is still wrong - turn THIS, not the `chance`
# weights, because the weights decide WHICH moment speaks and this decides HOW
# OFTEN the player hears one.
#
# A FLAG, not a modifier, for the reason bc_start_popup_cooldown_effect records:
# modifiers are wiped at combat end, so a modifier-based lockout silently resets
# every battle - which for this budget would mean it never survives to throttle
# the second battle of a war.
#
# SET BEFORE THE TOAST, INSIDE THE SAME `if`. Not inside the toast body: the
# mod's rule is that a toast is purely presentational, and this is state.
bc_start_toast_cooldown_effect = {
	# READ BEFORE WRITE, so the OLD flag state is still visible. Every one of the
	# 444 call sites is the first statement inside a bc_can_toast_trigger gate, so
	# "base held at entry" is equivalent to "passed through the high arm while base
	# was held" - i.e. this branch IS the preemption, exactly. It is also the only
	# obtainable counter, because a trigger cannot log.
	#
	# THE HIGH FLAG IS SET ONLY BY A PREEMPTION, NEVER BY AN ORDINARY RANK-2+
	# TOAST, AND THE FEATURE DIES IF THAT IS GOT WRONG. Setting it whenever the
	# rank is 2+ is self-defeating in exactly the case this exists to serve: past
	# rung two every moment is rank 3, so every toast would refresh the high flag
	# while the 10-day base expires under it, the high arm could never pass again,
	# and preemption would become structurally impossible for the rest of the
	# battle. It would also make the verification undecidable - "preempt 0" would
	# be produced both by a feature that is unnecessary and by a correctly wired
	# one that over-refreshed itself. The high budget pays for INTERRUPTIONS, so
	# only an interruption spends it.
	if = {
		limit = { has_character_flag = bc_toast_cooldown }
		debug_log = "BC: toast preempt"
		add_character_flag = {
			flag = bc_toast_cooldown_high
			days = 20
		}
	}

	# UNCONDITIONAL, AND THE SUB-ADDITIVITY IS DELIBERATE. A preemption also resets
	# the ordinary clock, so an interruption costs the player a later routine
	# toast. Moving this into an `else` would "restore independence" and raise the
	# volume ceiling to the full additive bound - a real volume change dressed up
	# as a tidy-up.
	add_character_flag = {
		flag = bc_toast_cooldown
		days = 10
	}

	# THE DENOMINATOR, not a budget. How often a rank-2+ moment spoke at all, by
	# either arm. Without it, "preempt 0" cannot be told apart from "no rank-2+
	# moment ever reached the gate".
	#
	# NESTED, NOT TWO ROWS IN ONE `limit`, for the reason bc_can_toast_trigger
	# records: CK3 does not short-circuit.
	if = {
		limit = { exists = var:bc_last_moment_rank }
		if = {
			limit = { var:bc_last_moment_rank >= 2 }
			debug_log = "BC: toast rank2"
		}
	}
}

# --- the eve's own throttle -------------------------------------------------
# 90 days, and a SEPARATE FLAG from bc_popup_cooldown on purpose.
#
# Sharing the 60-day pop-up cooldown would starve the mid-battle chronicles: the
# eve fires FIRST, every time, before a single phase event has rolled, so it would
# consume the budget before the battle began and the player would never see a
# moment from inside a fight again.
#
# Longer than 60 because the eve is rarer by construction - it gates on
# var:bc_battle_size >= 3, which is num_total_troops >= 10000 on the COMBAT
# scope (the size ladder in bc_battle_open_effect, bc_ledger_effects.txt), so ten
# thousand men across both hosts, not per side - and because two eves in a season
# reads as a tic rather than as an occasion.
#
# A FLAG, not a modifier, for the reason bc_start_popup_cooldown_effect records:
# modifiers are wiped at combat end, so a modifier-based lockout silently resets
# every battle.
bc_start_eve_cooldown_effect = {
	add_character_flag = {
		flag = bc_eve_cooldown
		days = 90
	}
}
