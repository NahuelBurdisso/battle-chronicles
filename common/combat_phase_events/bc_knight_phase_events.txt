# Battle Chronicles - knight phase events.
#
# `type = knight` has NO is_ai gate. Unlike `type = commander` it fires for every
# knight on every side of every combat in the world - the probe run saw 41 fires
# from a single battle. Two consequences:
#   - gate hard in is_valid, cheapest test first, and never do expensive work
#     before the gate
#   - bc_chronicle_ready_trigger / bc_mark_chronicle_effect are per-CHARACTER, so
#     a knight's 15-day cooldown is independent of his commander's
#
# Knights push momentum onto the SIDE COMMANDER, not onto themselves. Only the
# commander's advantage modifiers were proven to reach the combat roll; a knight's
# own advantage is unverified, and a knight is the story rather than the mechanic.
# `?=` on side_commander because a side can be leaderless.
#
# TWO RULES LEARNED FROM THE FIRST LIVE RUN, both of which this file now follows:
#
# 1. STATE FIRST, PRESENTATION SECOND. Effects go in the `effect` block; the
#    toast stays purely presentational. Vanilla does the same -
#    00_knight_phase_events.txt applies add_prestige and friends in the effect
#    block and keeps the message presentational.
#
#    The reason ORIGINALLY given for this rule was wrong, and the correction is
#    worth keeping so nobody reinstates it. `send_interface_toast` is documented
#    as "sends a message to the player playing the character in the scope and
#    then executes any effects inside", which reads as though a nested effect
#    cannot run for an AI subject. Measured at this exact site on 2026-08-14 by
#    splitting the count on `is_ai` outside the toast and comparing to a marker
#    inside it: human 0, AI 128, inside 128. The body runs for AI exactly as it
#    does for a human. Keep the rule, drop the justification.
#
# 2. THE TOAST GOES TO WHOEVER IS LEADING THE SIDE. A toast raised on `root`
#    reaches "the player playing root", and nobody is playing a courtier knight,
#    so the player never saw these at all. Vanilla solves it the same way, by
#    re-scoping the message (`liege = { send_interface_message = ... }`). Here the
#    right recipient is the side commander: he is who the knight fought under, and
#    he is who the momentum lands on.
#
# WEIGHTS. The competition for a knight roll is vanilla `knight_none` at
# base = 2000 - twice the commander pool's `commander_none`. Vanilla's own knight
# events sit at base 30-100. AGOT adds only dragonrider events, one of which is
# base = 100000 but is gated behind `is_current_dragonrider_warfare`, so it does
# not crowd anything out for an ordinary knight. base = 150 here is ~7% of a
# valid roll. The previous base = 40 combined with narrow gates produced zero
# fires across an entire session.

# ---------------------------------------------------------------------------
# A knight holds the breach. The side is losing and one man buys it time.
# ---------------------------------------------------------------------------
bc_knight_holds_the_breach = {
	type = knight

	# Cheapest first: a flag test, then a stat, then the cross-scope strength
	# comparison last. Tooltip rendering evaluates every branch regardless of
	# early-out, but ordering still pays off on the ordinary path.
	is_valid = {
		is_alive = yes
		bc_chronicle_ready_trigger = yes
		prowess >= 10
		bc_side_is_losing_trigger = yes
	}

	chance = {
		base = 150
		modifier = {
			factor = 0.5
			has_game_rule = bc_frequency_low
		}
		modifier = {
			factor = 2
			has_game_rule = bc_frequency_high
		}
		modifier = {
			add = 100
			prowess >= 16
		}
	}

	effect = {
		save_scope_as = bc_knight
		bc_mark_chronicle_effect = yes
		debug_log = "BC: fired bc_knight_holds_the_breach"

		add_prestige = 100

		# reverse_add_opinion, not add_opinion. `add_opinion = { target = liege }`
		# changes ROOT's opinion of the liege - it would have made the knight
		# think better of his lord for his own courage. The admiration runs the
		# other way, which is what "Stood Fast in Battle" says, and it is the
		# same direction bc_knight_shames_himself uses for the grudge.
		if = {
			limit = { exists = liege }
			reverse_add_opinion = {
				target = liege
				modifier = bc_valiant_in_battle_opinion
				opinion = 15
			}
		}

		scope:combat_side.side_commander ?= {
			bc_battle_open_effect = yes
			bc_momentum_minor_effect = yes
			bc_record_moment_effect = { TIER = major KIND = entrenched SWING = up }

			send_interface_toast = {
				type = event_toast_effect_good
				title = bc_knight_holds_the_breach_toast
				desc = bc_knight_holds_the_breach_toast_desc
				left_icon = scope:bc_knight
			}
		}
	}
}

# ---------------------------------------------------------------------------
# A knight shames himself. Non-lethal: he lives, and everyone saw it.
# ---------------------------------------------------------------------------
bc_knight_shames_himself = {
	type = knight

	# `prowess <= 6` never fired once, and it never will: knights are selected
	# for prowess, so a knight that feeble is close to nonexistent. Craven is the
	# better story anyway - the man who breaks is the one who was always going to.
	is_valid = {
		is_alive = yes
		bc_chronicle_ready_trigger = yes
		OR = {
			has_trait = craven
			prowess <= 8
		}
		bc_side_is_losing_trigger = yes
	}

	chance = {
		base = 150
		modifier = {
			factor = 0.5
			has_game_rule = bc_frequency_low
		}
		modifier = {
			factor = 2
			has_game_rule = bc_frequency_high
		}
		modifier = {
			add = 100
			has_trait = craven
		}
	}

	effect = {
		save_scope_as = bc_knight
		bc_mark_chronicle_effect = yes
		debug_log = "BC: fired bc_knight_shames_himself"

		add_stress = 15

		# The grudge runs from the liege toward the knight, so it is the liege's
		# opinion of him that drops.
		if = {
			limit = { exists = liege }
			reverse_add_opinion = {
				target = liege
				modifier = bc_craven_in_battle_opinion
				opinion = -20
			}
		}

		scope:combat_side.side_commander ?= {
			bc_battle_open_effect = yes
			bc_faltering_minor_effect = yes
			bc_record_moment_effect = { TIER = minor KIND = wild SWING = down }

			send_interface_toast = {
				type = event_toast_effect_bad
				title = bc_knight_shames_himself_toast
				desc = bc_knight_shames_himself_toast_desc
				left_icon = scope:bc_knight
			}
		}
	}
}

# ---------------------------------------------------------------------------
# Shieldbrothers. The seed of the wider idea - two knights who came through the
# same press together. Free to hook here because we are already iterating
# knights, and every_side_knight / random_side_knight are proven (probe 14).
#
# This is the only chronicle that leaves something permanent. The opinion decays
# at 0.05/month; the memory lasts forty years, and both men carry it.
# ---------------------------------------------------------------------------
bc_shieldbrother_forged = {
	type = knight

	is_valid = {
		is_alive = yes
		bc_chronicle_ready_trigger = yes
		scope:combat_side = {
			any_side_knight = {
				count >= 2
			}
		}
	}

	chance = {
		base = 60
		modifier = {
			factor = 0.5
			has_game_rule = bc_frequency_low
		}
		modifier = {
			factor = 2
			has_game_rule = bc_frequency_high
		}
	}

	# NOTHING happens before the partner is found. `is_valid` can only ask
	# whether the side has two knights, not whether a second one is off
	# cooldown, so this event can win a roll and still have nobody to pair
	# with. Marking the chronicle up front cost the knight a 15-day lockout for
	# an event that did not happen, and logged a fire that produced no story.
	# A wasted roll is free; a wasted lockout is not.
	effect = {
		save_scope_as = bc_knight

		# The one exception to the rule stated just above, and it does not
		# violate it in substance: opening the battle is not a lockout. It
		# writes two variables that every other phase event overwrites anyway,
		# so running it on a roll that finds no partner costs nothing. Do not
		# "fix" this by moving it inside the if - the seam calls below need
		# bc_battle_size, and no other event is guaranteed to have run first.
		scope:combat_side.side_commander ?= {
			bc_battle_open_effect = yes
		}

		scope:combat_side = {
			random_side_knight = {
				limit = {
					NOT = { this = scope:bc_knight }
					is_alive = yes
					bc_chronicle_ready_trigger = yes
				}
				save_scope_as = bc_shieldbrother
			}
		}

		if = {
			limit = { exists = scope:bc_shieldbrother }

			bc_mark_chronicle_effect = yes
			debug_log = "BC: fired bc_shieldbrother_forged"

			scope:bc_shieldbrother = {
				bc_mark_chronicle_effect = yes
			}

			add_opinion = {
				target = scope:bc_shieldbrother
				modifier = bc_shieldbrother_opinion
				opinion = 25
			}
			reverse_add_opinion = {
				target = scope:bc_shieldbrother
				modifier = bc_shieldbrother_opinion
				opinion = 25
			}

			# One memory each, naming the other man. The `shieldbrother` tag has
			# to match the participants list in bc_memories.txt, and it is what
			# the loc reads back as [shieldbrother.GetFullName].
			create_character_memory = {
				type = bc_shieldbrother_memory
				participants = {
					shieldbrother = scope:bc_shieldbrother
				}
			}
			scope:bc_shieldbrother = {
				create_character_memory = {
					type = bc_shieldbrother_memory
					participants = {
						shieldbrother = scope:bc_knight
					}
				}
			}

			scope:combat_side.side_commander ?= {
				bc_record_moment_effect = { TIER = minor KIND = ordered SWING = up }

				send_interface_toast = {
					type = event_toast_effect_good
					title = bc_shieldbrother_forged_toast
					desc = bc_shieldbrother_forged_toast_desc
					left_icon = scope:bc_knight
					right_icon = scope:bc_shieldbrother
				}
			}
		}
	}
}

# BATCH 2026-08-17 - three more knight cases.
#
# WHY. Ten commander templates and three knight ones produced 233 firings in
# 44 battles, so the same sentences repeat inside a single war. The rate
# limiter is the per-character cooldown, not the weights, so a template that
# fires in a slot an old one already owned buys variety at no rate cost.
# docs/specs/2026-08-17-battle-chronicles-cases-design.md carries the argument
# in full.
#
# WHAT IS NEW HERE. Every knight case shipped before this batch gates on
# `bc_side_is_losing_trigger`. A knight on a WINNING side matched nothing but
# bc_shieldbrother_forged, so two of the three below open that half of the
# space, and the third opens ground - an axis no knight event read at all.
#
# WEIGHTS: base = 80, roughly half the shipped 150. The competition is vanilla
# `knight_none` at base = 2000; the three shipped total 360, and three more at
# 150 would have doubled the mod's share of a knight roll. 80 was chosen over
# the commander batch's 30 because a knight roll is a much harder pool to be
# seen in - the file header records that base = 40 with narrow gates produced
# ZERO fires across a whole session.
#
# MOMENTUM STACKING. Each grant below sits behind an explicit
# `NOT = { bc_has_momentum_trigger = yes }` (or the faltering equivalent)
# evaluated INSIDE the side_commander scope. It cannot go in `is_valid`: that
# block is rooted on the KNIGHT, and bc_has_momentum_trigger reads
# has_character_modifier on its own root, so there it would test the wrong
# man's modifiers. The two shipped combat knight events do not carry this
# guard and can therefore put a second live rung on a commander who already
# has one; that is a real defect, but fixing it changes calibrated behaviour
# and is left for a run to confirm rather than folded into a content batch.
# ===========================================================================

# BATCH 2026-08-17, THE THIRD - three knight cases about the battle around him.
#
# The nine above read the knight (prowess, age, kin, craven) and the side he is
# on (winning, losing, outnumbering). None of them reads the BATTLE ITSELF -
# how bloody it has got, how long the war behind it has run, how many men are
# on the field. All three of those triggers were written for commander cases
# and have never been asked from a knight's root. They are all rooted on a
# character but read only scope:combat_side, which a knight event has, so they
# work here unchanged - the same argument bc_knight_meets_his_cousin makes

# ===========================================================================
# BATCH 2026-08-17, THE EIGHTH ON THIS SIDE: ONE TEMPLATE, AND IT LEAVES
# SOMETHING BEHIND.
#
# TWENTY-FOUR KNIGHT AND COMMANDER TEMPLATES HAVE SHIPPED SINCE PLAN 1 AND NOT
# ONE OF THEM LEAVES ANYTHING PERMANENT. bc_shieldbrother_forged is still the
# only phase event in the mod that writes a memory and an opinion, and the only
# other permanent thing anywhere is bc_nemesis_memory, which is written from the
# aftermath rather than from a moment. Content batches are cheap; this is the
# expensive kind and it is worth one template rather than three.
#
# IT RUNS BETWEEN A KNIGHT AND HIS COMMANDER, which is the relation the mod has
# never touched. bc_shieldbrother_forged pairs two knights and the rivalry
# ladder pairs two commanders. The vertical one - the man in the ranks and the
# man whose day it is - had a toast at most.
#
# IT IS NOT A SUBSET OF bc_knight_holds_the_breach, which is the nearest gate
# (prowess >= 10, losing). This wants prowess >= 12, losing, AND a living
# commander who is not the knight himself, so it is strictly narrower - and that
# is the right direction: a narrower gate cannot steal a roll from the wider one
# it sits under any more often than it fires, and what it does with a roll is
# permanent.
#
# base = 60, NOT 80. It is the rarest thing on this side and the only one that
# writes a memory, and bc_shieldbrother_forged - the other template that leaves
# something behind - sits at exactly 60 too.
# ===========================================================================

# ---------------------------------------------------------------------------
# The knight who goes in after his lord.
#
# THE COMMANDER MUST EXIST AND MUST NOT BE THE KNIGHT. Both halves are load
# bearing. `?=` handles the leaderless side everywhere else in this file, but
# here the commander is not just the recipient of a toast - he is the man the
# memory is about, so a missing one makes the whole moment meaningless rather
# than merely unaddressed. And nothing in the API settles whether
# every_side_knight can return the side commander, so the identity test is the
# same guard bc_chronicles.0091 puts on its memory handout.
#
# THE GUARD IS WRITTEN IN is_valid AND NOT IN THE EFFECT, unlike the momentum
# guards this file's earlier batches carry. Those have to be in the effect
# because they read the COMMANDER's modifiers and is_valid is rooted on the
# knight; this one reads a link off scope:combat_side, which is available in
# both places, and a gate that decides whether the moment happens at all
# belongs in is_valid.
#
# NO RUNG, DELIBERATELY, AND THE MOMENT IS STILL `major`. The tier is what the
# ledger records about the day; the rung is what the army feels for the next
# twenty days. A commander being dragged out of a press does not make his line
# fight better - it is a thing that happened to him - and `adapted`/`up` maps
# to no modifier at all, which is the honest answer.
#
# THE OPINION RUNS FROM THE COMMANDER TOWARD THE KNIGHT, so it is written with
# add_opinion from inside the commander's scope rather than reverse_add_opinion
# from the knight's. bc_knight_shames_himself writes the mirror image of this
# and does it the other way round, from the knight's scope, because there the
# knight is root; both are correct and neither is a template for the other.
# ---------------------------------------------------------------------------
bc_knight_pulls_him_out = {
	type = knight

	is_valid = {
		is_alive = yes
		bc_chronicle_ready_trigger = yes
		prowess >= 12
		bc_side_is_losing_trigger = yes
		exists = scope:combat_side.side_commander
		trigger_if = {
			limit = { exists = scope:combat_side.side_commander }
			NOT = { this = scope:combat_side.side_commander }
			scope:combat_side.side_commander = { is_alive = yes }
		}
	}

	chance = {
		base = 60
		modifier = {
			factor = 0.5
			has_game_rule = bc_frequency_low
		}
		modifier = {
			factor = 2
			has_game_rule = bc_frequency_high
		}
		# A man who was already going to do something stupid.
		modifier = {
			add = 40
			has_trait = brave
		}
		modifier = {
			factor = 0.5
			has_trait = craven
		}
	}

	effect = {
		save_scope_as = bc_knight
		bc_mark_chronicle_effect = yes
		debug_log = "BC: fired bc_knight_pulls_him_out"

		# THE FIFTH POP-UP BRANCH, 2026-08-17, AND THE FIRST ON THIS SIDE OF
		# THE MOD. Four pop-ups have shipped and all four ask a COMMANDER
		# something; a player who is somebody's sworn sword has never once been
		# asked anything by this mod, which in AGOT is not an edge case.
		#
		# bc_battle_open_effect IS HOISTED OUT OF THE BRANCH AND RUN BEFORE THE
		# RAISE. It is the only way the commander crosses the trigger_event
		# boundary: scope:combat_side does not, but a scope SAVED before the
		# raise does, and this effect's first line saves the commander as
		# scope:bc_chronicle_subject. That is also the scope the pop-up's three
		# Custom clauses hang off, so hoisting buys the ground clause too.
		# bc_eve_offer_effect writes scope:bc_eve_foe one line before its own
		# raise for exactly this reason; this is the same trick.
		#
		# IT IS STILL CALLED EXACTLY ONCE PER EXECUTION, which the effect's own
		# header requires - both arms of the branch are downstream of it and
		# neither calls it again.
		#
		# THE `else` ARM IS THE UNCHANGED ORIGINAL, down to the prestige. Every
		# AI knight and every player inside the 60-day pop-up cooldown gets the
		# memory, the opinion and the toast this template has always given.
		scope:combat_side.side_commander ?= {
			bc_battle_open_effect = yes
		}

		if = {
			limit = { bc_can_interrupt_player_trigger = yes }
			bc_start_popup_cooldown_effect = yes
			trigger_event = bc_chronicles.0005
		}
		else = {
			add_prestige = 100

			scope:combat_side.side_commander ?= {
				add_opinion = {
					target = scope:bc_knight
					modifier = bc_lifesaver_opinion
					opinion = 40
				}

				# The tag `lifesaver` must match the participants list in
				# bc_memories.txt; it is what the loc reads back with
				# [lifesaver.GetFullName].
				create_character_memory = {
					type = bc_lifesaver_memory
					participants = {
						lifesaver = scope:bc_knight
					}
				}
				debug_log = "BC: lifesaver memory"

				bc_record_moment_effect = { TIER = major KIND = adapted SWING = up }

				send_interface_toast = {
					type = event_toast_effect_good
					title = bc_knight_pulls_him_out_toast
					desc = bc_knight_pulls_him_out_toast_desc
					left_icon = scope:bc_knight
				}
			}
		}
	}
}
