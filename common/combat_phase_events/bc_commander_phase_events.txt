# Battle Chronicles - commander phase events.
#
# New additive file. CK3 reads every file in common/combat_phase_events/ and
# merges the keys, so nothing here replaces vanilla, AGOT or Battlefield Duel+.
#
# Firing model, all verified in game:
#   - a phase event rolls every COMBAT_EVENT_DAYS = 5 while a battle runs
#   - `type = commander` roots on the commander, so `is_ai = no` means the PLAYER
#     is personally leading that army - it is not a "player is watching" test
#   - the competition is vanilla `commander_none` at base = 1000 plus AGOT's own
#     casualty events, so base = 60 across seven events targets roughly 15-20% of
#     commander rolls
#   - bc_chronicle_ready_trigger, not the weights, is what stops these firing
#     back to back; see bc_scripted_effects.txt
#
# The weights below are guesses and are meant to be retuned against real play.
# That is exactly why bc_frequency exists.

# ---------------------------------------------------------------------------
# The banner falls. The signature setback: the army loses its focal point.
# ---------------------------------------------------------------------------
bc_banner_falls = {
	type = commander

	is_valid = {
		exists = root
		bc_chronicle_ready_trigger = yes
		bc_side_is_losing_trigger = yes
		NOT = { bc_is_faltering_trigger = yes }
	}

	chance = {
		base = 60
		modifier = {
			factor = 0.5
			has_game_rule = bc_frequency_low
		}
		modifier = {
			factor = 2
			has_game_rule = bc_frequency_high
		}
	}

	effect = {
		save_scope_as = bc_commander
		bc_mark_chronicle_effect = yes
		debug_log = "BC: fired bc_banner_falls"

		if = {
			limit = { bc_can_interrupt_player_trigger = yes }
			bc_start_popup_cooldown_effect = yes
			trigger_event = bc_chronicles.0001
		}
		else = {
			send_interface_toast = {
				type = event_toast_effect_bad
				title = bc_banner_falls_toast
				desc = bc_banner_falls_toast_desc
				left_icon = scope:bc_commander

				bc_faltering_major_effect = yes
				add_stress = 10
			}
		}
	}
}

# ---------------------------------------------------------------------------
# A horse killed under him. Non-lethal by design decision - no death, no maiming
# anywhere in this mod. The commander is shaken, and is admired for standing up.
# ---------------------------------------------------------------------------
bc_horse_killed_under_him = {
	type = commander

	is_valid = {
		exists = root
		bc_chronicle_ready_trigger = yes
		prowess >= 8
		NOT = { bc_is_faltering_trigger = yes }
	}

	chance = {
		base = 60
		modifier = {
			factor = 0.5
			has_game_rule = bc_frequency_low
		}
		modifier = {
			factor = 2
			has_game_rule = bc_frequency_high
		}
		# Fighting commanders are the ones who get unhorsed.
		modifier = {
			add = 30

			add_prestige = 50
		}
	}
}
