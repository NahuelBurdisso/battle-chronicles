# Battle Chronicles - commander phase events.
#
# New additive file. CK3 reads every file in common/combat_phase_events/ and
# merges the keys, so nothing here replaces vanilla, AGOT or Battlefield Duel+.
#
# Firing model, all verified in game:
#   - a phase event rolls every COMBAT_EVENT_DAYS = 5 while a battle runs
#   - `type = commander` roots on the commander, so `is_ai = no` means the PLAYER
#     is personally leading that army - it is not a "player is watching" test
#   - the chance is a weight against every OTHER VALID event, per the engine's own
#     `_combat_phase_events.info`. The pool is vanilla `commander_none` at
#     base = 1000 plus AGOT's casualty events
#   - bc_chronicle_ready_commander_trigger, not the weights, is what stops these firing
#     back to back; see bc_scripted_effects.txt
#
# STATE FIRST, PRESENTATION SECOND. Every momentum change used to live inside the
# `send_interface_toast` block. That is not safe: a toast is documented as
# "sends a message to the player playing the character in the scope and then
# executes any effects inside", and most commanders in the world are AI, with no
# player playing them. Left as it was, the entire momentum system risked applying
# to the player's battles and to nobody else's - and bc_lines_bogged_down proved
# the asymmetry, because it penalised the enemy commander from outside a toast
# while penalising its own commander from inside one. Vanilla keeps state changes
# in the effect block and messages purely presentational; so do we now.
#
# WEIGHTS, retuned after the first live session (59 commander fires):
#   banner_falls 10, rally 5, horse_killed 15, flank 7, van 16, ground 5, bogged 1
# horse_killed has the weakest situational gate of the seven and was firing like
# a default, so its base drops to 40. bogged_down's terrain gate is narrow enough
# that it needs a higher base to appear at all. bc_frequency exists so the rest
# stays retunable without a script edit.

# ---------------------------------------------------------------------------
# The banner falls. The signature setback: the army loses its focal point.
# ---------------------------------------------------------------------------
bc_banner_falls = {
	type = commander

	is_valid = {
		bc_chronicle_ready_commander_trigger = yes
		NOT = { bc_is_faltering_trigger = yes }
		bc_side_is_losing_trigger = yes
	}

	chance = {
		base = 60
		modifier = {
			factor = 0.5
			has_game_rule = bc_frequency_low
		}
		modifier = {
			factor = 2
			has_game_rule = bc_frequency_high
		}
		# Plan 4: the read-back. A grudge match produces more moments than a
		# battle against a stranger. Same factor on every commander event - a
		# per-event tuning pass needs its own runtime measurement.
		modifier = {
			factor = 1.5
			bc_facing_nemesis_trigger = yes
		}
	}

	effect = {
		save_scope_as = bc_commander
		bc_battle_open_effect = yes
		bc_mark_chronicle_commander_effect = yes
		debug_log = "BC: fired bc_banner_falls"

		if = {
			limit = { bc_can_interrupt_player_trigger = yes }
			bc_start_popup_cooldown_effect = yes
			trigger_event = bc_chronicles.0001
		}
		else = {
			bc_faltering_major_effect = yes
			add_stress = 10
			bc_record_moment_effect = { TIER = major KIND = wild SWING = down }

			send_interface_toast = {
				type = event_toast_effect_bad
				title = bc_banner_falls_toast
				desc = bc_banner_falls_toast_desc
				left_icon = scope:bc_commander
			}
		}
	}
}

	# end of the ladder the whole mod is tuned against.

		NOT = { bc_is_faltering_trigger = yes }

	}

	chance = {
		base = 60
		modifier = {
			factor = 0.5
			has_game_rule = bc_frequency_low
		}
		modifier = {
			factor = 2
			has_game_rule = bc_frequency_high
		}
		# Plan 4: the read-back. A grudge match produces more moments than a
		# battle against a stranger. Same factor on every commander event - a
		# per-event tuning pass needs its own runtime measurement.
		modifier = {
			factor = 1.5
			bc_facing_nemesis_trigger = yes
		}
	}

	effect = {
		save_scope_as = bc_commander
		bc_battle_open_effect = yes
		bc_mark_chronicle_commander_effect = yes
		debug_log = "BC: fired bc_flank_collapses"

		# THE FOURTH POP-UP BRANCH, 2026-08-17, and the first template to grow
		# one since plan 1. Shape copied verbatim from bc_banner_falls above:
		# no rung is applied before the raise, because the option is the
		# outcome, and the cooldown is started at the call site rather than
		# inside the event.
		#
		# THE `else` ARM IS THE UNCHANGED ORIGINAL. Every AI commander and
		# every player inside the 60-day pop-up cooldown still gets exactly the
		# decisive setback and the toast this template has always given, which
		# is what keeps the change additive rather than a retune.
		if = {
			limit = { bc_can_interrupt_player_trigger = yes }
			bc_start_popup_cooldown_effect = yes
			trigger_event = bc_chronicles.0004
		}
		else = {
			bc_faltering_major_effect = yes
			bc_record_moment_effect = { TIER = decisive KIND = wild SWING = down }

			send_interface_toast = {
				type = event_toast_effect_bad
				title = bc_flank_collapses_toast
				desc = bc_flank_collapses_toast_desc
				left_icon = scope:bc_commander
			}
		}
	}
}

# ===========================================================================
# THE SECOND TEN, 2026-08-17. Design:
# docs/specs/2026-08-17-battle-chronicles-cases-design.md
#
# WHY MORE TEMPLATES DOES NOT MEAN MORE INTERRUPTIONS. The rate limiter is
# bc_chronicle_ready_commander_trigger, not the weights - this file's own header
# says so and the code agrees. A commander who has just taken a moment makes
# EVERY event in this file invalid for the next bc_chronicle_flag_days_value
# days, so a blocked roll is not a Battle Chronicles roll at all; it goes to
# vanilla's commander_none. Adding templates therefore mostly changes WHICH
# moment a battle produces, not how many.
#
# IT IS NOT PURELY FREE, AND THE HONEST HALF IS THIS. Six of the original seven
# carry a situational gate, so an even battle on open ground with an unremarkable
# commander currently matches NOTHING and passes in silence. Those battles will
# now speak. That is an accepted, deliberate rate increase in the one place the
# mod was mute, and it is still capped by the same cooldown.
#
# BASE 30 IS THE DEFAULT HERE, against 40-100 for the original seven. `chance` is
# a weight against every other VALID event, including commander_none at
# base = 1000, so ten more templates at the old bases would roughly triple the
# mod's share of the pool. Thirty keeps the calibrated seven dominant among BC
# events while still letting the new ones appear. Two exceptions, both argued at
# their event.
#
# NO POP-UPS IN THIS BATCH. A pop-up spends bc_popup_cooldown, and the three
# shipped ones already compete with the eve for it. These are toasts only, which
# is also why none of them branches on bc_can_interrupt_player_trigger.
#
# NO NEW MODIFIERS AND NO NEW GAME RULE. Every one of these routes through the
# existing (KIND, SWING) table in bc_apply_verb_effect and the existing
# momentum/faltering pairs.
#
# STACKING, THE RULE THAT DECIDES EVERY is_valid BELOW: a template that grants
# momentum gates `NOT = { bc_has_momentum_trigger = yes }`, one that grants
# faltering gates `NOT = { bc_is_faltering_trigger = yes }`, and one that grants
# neither needs neither. bc_faltering_minor and bc_faltering_major are different
# modifier keys, so two live at once runs off the end of the ladder the mod is
# tuned against.
# ===========================================================================

# THE THIRD BATCH: WHO THE COMMANDER IS (2026-08-17)
#
# Six templates on the ONE AXIS THE OTHER TWENTY DO NOT GATE ON - the man's
# character. Every shipped template asks about the battle: the ground, the odds,
# the dead, the war, his age, whose quarrel it is. `has_trait` appears in this
# file eleven times and every one of them is a `chance` modifier, never a row in
# an is_valid. So two commanders in the same battle have always been able to
# produce only the same set of moments, which is the wrong answer for a mod
# whose whole output is character.
#
# EVERY GATE HERE IS A TRAIT PLUS A SITUATION, never a trait alone. A bare
# `has_trait = wrathful` is valid in EVERY battle that man ever fights, which
# makes the template a property of the commander instead of a moment in a fight
# - and it would crowd out the calibrated seven for exactly the characters most
# likely to be somebody's player. The second row also gives the prose something
# to describe. bc_the_wine_at_noon is the single exception and says why at its
# own comment.
#
# ALL SIX AT base = 30, the second batch's default, for the second batch's
# reason. The running total is the thing to watch and it is written down here so
# the next reader does not have to add it up: seven calibrated templates at
# 40-100 (~400), ten at 30 (300), six more at 30 (180), against vanilla's
# commander_none at base = 1000. If a sweep shows the original seven crowded
# out, THIS BATCH IS THE FIRST TO CUT - it is the newest and the least measured.
#
# ALL SIX TRAIT KEYS ARE AGOT KEYS, checked in AGOT's own common/traits/00_traits.txt
# and not in vanilla's: wrathful :2821, patient :3074, drunkard :5437,
# zealous :4423, craven :3668, strategist :1543. AGOT overrides that file
# wholesale, so a key that exists only in vanilla is not a key at all here.
#
# NO TRAIT IS AWARDED OR REMOVED BY ANY OF THEM. These READ character; the
# reputation chain in bc_reputation_effects.txt is the only thing in the mod
# that writes it, and it does so off the style streak. A template that both
# gated on a trait and granted one would be a feedback loop.
# ===========================================================================

bc_no_fear_in_them = {
	type = commander

	is_valid = {
		bc_chronicle_ready_commander_trigger = yes
		dread <= low_dread
		bc_side_is_losing_trigger = yes
	}

	chance = {
		base = 30
		modifier = {
			factor = 0.5
			has_game_rule = bc_frequency_low
		}
		modifier = {
			factor = 2
			has_game_rule = bc_frequency_high
		}
		# A commander with nothing to hold them and no name to hold them with.
		modifier = {
			add = 20
			has_trait = craven
		}
		modifier = {
			factor = 0.5
			martial >= 14
		}
	}

	effect = {
		save_scope_as = bc_commander
		bc_battle_open_effect = yes
		bc_mark_chronicle_commander_effect = yes
		debug_log = "BC: fired bc_no_fear_in_them"

		add_stress = 10

		if = {
			limit = { NOT = { bc_is_faltering_trigger = yes } }
			bc_faltering_minor_effect = yes
		}

		bc_record_moment_effect = { TIER = minor KIND = wild SWING = down }

		send_interface_toast = {
			type = event_toast_effect_bad
			title = bc_no_fear_in_them_toast
			desc = bc_no_fear_in_them_toast_desc
			left_icon = scope:bc_commander
		}
	}
}

# ---------------------------------------------------------------------------
# The third answer, and the only one that is not about whether the line holds.
# At `>= major_dread_value` he does not have to argue with anyone. He sends
# them where they will not come back from and they go, and the battle gets the
# ground he wanted at the price he was always going to pay.
#
# NO SIDE ROW, DELIBERATELY. The other two are about a side that is losing;
# this one is about how the orders are given, and they are given the same way
# on a good day. bc_nothing_but_the_horse is the shipped precedent for a
# commander gate with no second row.
#
# NO RUNG AT ALL. Both rungs are about the side's standing, and this template
# does not claim the day moved - it claims the day cost. bc_knight_will_not_be_
# carried is the shipped precedent for a template that records a moment and
# grants nothing.
#
# HE IS PAID IN PRESTIGE AND HE PAYS IN STRESS, because both are true at once:
# the field respects it and he watched it.
#
# `bloodied`/`up` IS THE POINT OF THE WHOLE TEMPLATE - ground bought with men,
# and bought on purpose.
# ---------------------------------------------------------------------------
bc_the_hard_hand = {
	type = commander

	is_valid = {
		bc_chronicle_ready_commander_trigger = yes
		dread >= major_dread_value
	}

	chance = {
		base = 30
		modifier = {
			factor = 0.5
			has_game_rule = bc_frequency_low
		}
		modifier = {
			factor = 2
			has_game_rule = bc_frequency_high
		}
		modifier = {
			add = 20
			has_trait = callous
		}
		modifier = {
			factor = 0.5
			has_trait = compassionate
		}
	}

	effect = {
		save_scope_as = bc_commander
		bc_battle_open_effect = yes
		bc_mark_chronicle_commander_effect = yes
		debug_log = "BC: fired bc_the_hard_hand"

		add_prestige = 50
		add_stress = 10

		bc_record_moment_effect = { TIER = minor KIND = bloodied SWING = up }

		send_interface_toast = {
			type = event_toast_effect_good
			title = bc_the_hard_hand_toast
			desc = bc_the_hard_hand_toast_desc
			left_icon = scope:bc_commander
		}
	}
}
