# Battle Chronicles - tuning rules.
#
# The starting weights in the phase events are guesses. These rules exist so the
# player (and the author) can retune without a script edit and without a restart
# of the whole modding loop.
#
# Read in script with `has_game_rule = <option>`, exactly as vanilla's
# commander_none reads `has_game_rule = easy_difficulty`. Verified in game.
#
# WARNING: a modded game rule is baked into every save made with it. Disabling
# this mod afterwards makes that save fail to load with
#   Failed to read key reference: bc_frequency_default ... in file: "save games/x.ck3"
# That is expected, not a bug.
#
# No option sets blocks_achievements.

bc_frequency = {
	categories = { flavor }

	default = bc_frequency_default

	bc_frequency_low = {}
	bc_frequency_default = {}
	bc_frequency_high = {}
}

# The rung cap, not an on/off switch for the counter.
#
# AGOT hangs murder schemes off `nemesis`. A mod that silently manufactures
# nemeses is changing far more than battles, so the top rung is opt-in and the
# default stops at `rival`.
#
# THE CAP CLAMPS THE RUNG, IT DOES NOT SKIP THE AWARD. The rule test sits inside
# the TOP arm's limit in bc_rivalry_set_relation_effect, so under bc_rivalry_rival
# a counter of 5 fails that arm, falls through to `>= 2`, and sets `rival` - not
# nothing. Putting the rule test in an outer gate instead would make a man who
# beat you five times LESS hated than one who beat you twice.
#
# bc_rivalry_off SUPPRESSES THE RELATION ONLY. bc_nemesis, bc_nemesis_losses and
# bc_facing_nemesis_trigger all keep working, so a player who turns it off still
# gets the narrative - the man who has beaten you twice - without the mod touching
# CK3's relation graph or AGOT's murder schemes. Suppressing the counter too would
# make the off switch delete the feature rather than its side effects, and would
# leave bc_rivalry_clear_effect with nothing to remove.
#
# THE COUNTER IS NEVER CAPPED. It keeps climbing past the cap, so lowering this
# design's own thresholds in a later version is not a data migration. Raising the
# RULE mid-campaign is impossible anyway - game rules are baked into a save.
bc_rivalry = {
	categories = { flavor }

	default = bc_rivalry_rival

	bc_rivalry_off = {}
	bc_rivalry_rival = {}
	bc_rivalry_nemesis = {}
}

# TWO OPTIONS, NOT THREE. The other rules above are low/default/high ladders
# because each scales a frequency. This one is a switch: a wound either can
# happen or it cannot. A "frequent wounds" rung would need a second set of gate
# values and there is nothing to tune them against until family 3's firing rate
# has been measured at least once.
#
# DEFAULTS ON, like bc_aftermath. The feature is the point of the plan that adds
# it; a rule defaulting off is a feature nobody sees.
bc_wounds = {
	categories = { flavor }

	default = bc_wounds_default

	bc_wounds_off = {}
	bc_wounds_default = {}
}
