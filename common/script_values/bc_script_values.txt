# Battle Chronicles - thresholds.
#
# A trigger can only compare one side_strength against the other. To ask "are we
# outnumbered by half again" the multiplied figure has to come from a script
# value. Used by bc_scripted_triggers.txt.
#
# VERIFIED IN GAME 2026-08-12: bc_flank_collapses and bc_van_breaks_through both
# fired, so reading side_strength into a script value and multiplying works.
#
# This is exactly the shape vanilla uses - 00_combat_values.txt defines
# `vastly_outnumbered_combat_side_threshold = { value = side_strength multiply = 5 }`.
# Not reusable here: 5x is a rout, 1.5x is a battle going badly, which is the
# whole point of these two events.
#
# side_strength is scaled down by a factor of 1000 by the engine, so these are
# ratios of an already-scaled number. Fine, because both sides are scaled alike.
#
# SCOPE WARNING: THIS FILE NOW HOLDS THREE SCOPE CONVENTIONS. The two values
# below are written against `scope:combat_side` because they are read from phase
# events, where that scope exists. The aftermath values further down are written
# against `root`, because they are read from a combat_side-scoped event where the
# side IS root and `scope:combat_side` does not exist. The third is
# bc_chronicle_flag_days_value at the bottom, which is neither: it is evaluated
# inside bc_mark_chronicle_commander_effect, which runs ON THE COMMANDER, so it
# reads `var:bc_battle_size` bare off `this` with no scope prefix at all.
# Do not copy any one style into another's call site.

bc_outnumbered_threshold_value = {
	value = scope:combat_side.side_strength
	multiply = 1.5
}

bc_outnumbering_threshold_value = {
	value = scope:combat_side.enemy_side.side_strength
	multiply = 1.5
}

# --- aftermath: how lopsided were the hosts ---------------------------------
#
# Read from bc_chronicles.0091 / .0092, where the combat side is ROOT. The
# `root.` prefix is not decoration. Both call sites put a scope path on the
# left - `enemy_side.side_max_army_size >= bc_our_host_gap_value` - and if the
# engine resolves an unprefixed script value on the right in the LEFT operand's
# scope, this would evaluate as `enemy >= enemy * 2`: permanently false, on
# both sides, while still passing the winner-versus-loser symmetry check that
# exists to catch exactly this. That resolution rule is unverified. Anchoring
# means it never has to be known.
#
# VERIFIED IN GAME 2026-08-13: side_max_army_size works on the LEFT of a
# comparison, which had no script precedent anywhere in vanilla, AGOT or the
# playset. A control ladder against plain constants left its else branch, and
# the odds ladder spread across all four buckets over 39 battles rather than
# collapsing to one.
#
# multiply MUST stay strictly greater than 1. At exactly 1 the row degenerates
# to "the hosts were not identical", which is true in nearly every battle and
# donates a free calc_true_if point to both sides. It was 1.5 until the
# calibration run showed that fired on 54% of battles; 2.0 is the top quarter.
#
# side_max_army_size, not side_army_size or troops_ratio: at combat END a
# current-strength ratio measures the outcome, not the odds. Grand Remembrance
# corroborates the reading by computing casualties as
# `side_max_army_size - side_army_size` (zz_gr_memory_tracking.txt:74).

bc_our_host_gap_value = {
	value = root.side_max_army_size
	multiply = 2.0
}

bc_their_host_gap_value = {
	value = root.enemy_side.side_max_army_size
	multiply = 2.0
}

# The chronicle cooldown, in days, scaled by battle size.
#
#   size 3 (10,000+)  ->  10 days   a moment every other phase roll
#   size 2 (2,000+)   ->  20 days
#   size 1 (skirmish) ->  40 days
#
# Phase events roll every 5 days (COMBAT_EVENT_DAYS), so the flag length IS the
# pacing. The old flat 15 gave a two-month siege battle a constant stream and a
# skirmish nothing at all.
#
# SCOPE: `this`, a character - the commander. NOT scope:combat_side, NOT root.
# This is the third convention in this file; see the header.
#
# SHAPE: every branch is a bare `value = X`, there is NO base value outside the
# ladder, and the ladder is an if nested inside an if. All three at once are
# shipped in ONE vanilla value - recent_epidemics_outbreak_mult_value,
# 06_ce1_epidemics_values.txt:1869-1917. Its outer arm even gates on a variable
# existing (`has_variable = epidemic_cooldown_general`) exactly as this one does,
# and its inner leaves are all bare `value = <ident>`.
#
# The one place this is SAFER than the vanilla original: that value has no outer
# `else`, so a character matching neither outer arm falls through to nothing.
# This one has the else below.
#
# THE `exists` GUARD IS THE OUTER BRANCH, NOT A SIBLING ROW, for the reason
# stated at the seam's size gate: CK3 does not short-circuit, so a sibling
# `exists` would not stop the comparison from evaluating and logging a read of a
# missing variable.
#
# The 15 in the else arm is the old flat duration. It is unreachable if the call
# ordering holds - bc_battle_open_effect writes bc_battle_size before
# bc_mark_chronicle_commander_effect is reached in all seven commander events -
# and it is there so that a future call site that breaks that ordering degrades
# to the shipped behaviour instead of to zero days, which would mean no cooldown
# at all.
bc_chronicle_flag_days_value = {
	if = {
		limit = { exists = var:bc_battle_size }
		if = {
			limit = { var:bc_battle_size >= 3 }
			value = 10
		}
		else_if = {
			limit = { var:bc_battle_size >= 2 }
			value = 20
		}
		else = {
			value = 40
		}
	}
	else = {
		value = 15
	}
}

# --- rivalry step size ------------------------------------------------------
#
# ONE STEP SIZE, USED IN ALL FOUR ARITHMETIC SITES: the create branch, the
# escalate branch, the floor guard, and the decrement. Section 4 of the design
# records that an earlier draft hardcoded the create branch to 1, which quietly
# made "a wipe takes a stranger straight to rival" false and made the design's own
# worked example unreachable. Do not inline the 1 anywhere.
#
# scope:wipe IS ENGINE-PROVIDED AND THIS VALUE IS LIVE TODAY. It is not plan 5's
# and no script saves it: AGOT documents it in its own on_action headers, at
# common/on_action/combat_on_actions.txt:13 above on_combat_end_winner and :527
# above on_combat_end_loser, and it survives into events raised from those hooks'
# `events = { }` lists. Plan 5 only wraps the read in a named trigger. Test this
# branch when you test the plan; do not assume it is dormant.
#
# THE READING OF scope:wipe NOW LIVES IN ONE PLACE, bc_was_wipe_trigger. This
# value inlined the same exists-plus-trigger_if conjunction until plan 5 task 4;
# the swap is behaviour-neutral by construction, because the trigger's body IS
# that conjunction. Why it matters anyway: shipped AGOT uses both idioms for this
# scope in one file - `exists = scope:wipe` as the entire test at :100,
# `scope:wipe = yes` at :758 - so it is unsettled whether the scope always exists
# carrying a boolean or exists only when true. Two copies of an unsettled reading
# diverge the moment someone learns the answer. There is now exactly one, and
# bc_style_step_value goes through it too.
#
# SHAPE PRECEDENT, verified by reading the file: bare `value = X` in each branch
# with no base value outside is swing_scales_discounts_multiplier_value,
# 00_diarchy_values.txt:3224-3237.
bc_rivalry_step_value = {
	if = {
		limit = { bc_was_wipe_trigger = yes }
		value = 2
	}
	else = {
		value = 1
	}
}

# --- B's streak step ---------------------------------------------------------
#
# TWO ON A WIPE, ONE OTHERWISE, AND IT IS READ IN EXACTLY ONE PLACE - the MATCH
# branch of bc_style_streak_effect. The two CREATE branches stay a literal 1.
#
# THE ASYMMETRY WITH C IS DELIBERATE AND IS THE WHOLE MECHANISM. C's counter uses
# its step in the create branch too, because it is a tally of damage done to you
# and the first blow's severity is exactly the thing being counted. A STREAK is a
# claim about repetition, and nothing has repeated on the first battle. Because
# the create branch stays 1, "two wipes fought the same way earn the trait" is
# true: the first wipe is a mismatch, creates the streak at 1, the second matches
# and adds 2, reaching the threshold of 3. Use the step in the create branch and
# it becomes "one wipe plus one mismatch", which is a different feature.
#
# THE TERRAIN STREAK IS NOT AFFECTED AND MUST NOT BE. Its threshold is 2, so a
# by-2 step would award a terrain trait from a single battle and reopen exactly
# the budget problem plan 3's award-ordering rule exists to close.
bc_style_step_value = {
	if = {
		limit = { bc_was_wipe_trigger = yes }
		value = 2
	}
	else = {
		value = 1
	}
}
