# Battle Chronicles - character memories.
#
# The design always called for shieldbrothers to remember each other rather than
# just carry an opinion modifier that decays away. Opinion fades; a memory is the
# thing a character can still be asked about in twenty years.
#
# Shape copied verbatim from vanilla `became_blood_brother` in mpo_memories.txt -
# `description = { random_valid = { desc = KEY } }`, one perspective each, a
# `participants = { }` tag list, and a duration. The tag `shieldbrother` is what
# the loc keys read back with `[shieldbrother.GetFullName]`, and it is also the
# name that must be used at the `create_character_memory` call site.
#
# `icon` must name a file that already exists in the vanilla memory icon set -
# `battle.dds` is used by vanilla's own `witnessed_death_battle_good`.

# THREE LINES PER PERSPECTIVE SINCE 2026-08-17. The `random_valid` wrapper was
# always here - it is part of the shape copied from vanilla - and it held one
# desc, so every shieldbrother in a save remembered the day in the same words.
# Filling the pool is the whole change; no structure moved.
#
# THE THREE PERSPECTIVES OF ONE VARIANT ARE ONE SENTENCE IN THREE VOICES and
# are NOT independently pooled by the engine - it picks a line per perspective
# block. So the three pools are kept in the same ORDER, and a reader comparing
# two of them side by side will usually see the same variant, which is what the
# UI expects. Do not reorder one pool without reordering all three.
#
# ALL NINE ARE PAST TENSE AND LOWER CASE with no full stop, per the vanilla
# standard in _character_memories.info: the UI builds a sentence around them.
bc_shieldbrother_memory = {
	categories = { positive war relationships friend }
	icon = "battle.dds"

	description = {
		random_valid = {
			desc = bc_shieldbrother_memory_desc
			desc = bc_shieldbrother_memory_desc_b
			desc = bc_shieldbrother_memory_desc_c
		}
	}

	second_perspective_description = {
		random_valid = {
			desc = bc_shieldbrother_memory_desc_second_perspective
			desc = bc_shieldbrother_memory_desc_b_second_perspective
			desc = bc_shieldbrother_memory_desc_c_second_perspective
		}
	}

	third_perspective_description = {
		random_valid = {
			desc = bc_shieldbrother_memory_desc_third_perspective
			desc = bc_shieldbrother_memory_desc_b_third_perspective
			desc = bc_shieldbrother_memory_desc_c_third_perspective
		}
	}

	participants = { shieldbrother }

	# Deliberately long. The opinion modifier decays in a few years; this is what
	# is left after it has.
	duration = {
		years = 40
	}
}

# --- the battle memory ------------------------------------------------------
#
# One field, remembered by every knight who stood on it. Two types rather than
# one so the polarity token in `categories` is right and the localization does
# not have to branch on the outcome.
#
# NO `participants` block on purpose. This memory is about a place, not a
# person - vanilla's `became_incapable_due_to_battle_concussion` is the
# precedent for a participant-free type carrying `battle_location`.
#
# `duration` is forty years and MUST NOT be `memory_default_duration`. The
# default is `value = 5` plus bonuses that only rulers, duchy-tier holders and
# the player's own circle collect, so a landless AI knight - precisely the man
# this feature exists for - would score 5 and forget the field inside five
# years. Same decision, same reason, as bc_shieldbrother_memory above.

bc_battle_memory_won = {
	categories = { positive martial war battle }
	icon = "battle.dds"

	description = {
		random_valid = {
			desc = bc_battle_memory_won_desc

		}
	}

	second_perspective_description = {
		random_valid = {
			desc = bc_battle_memory_won_desc_second_perspective
			desc = bc_battle_memory_won_desc_b_second_perspective
			desc = bc_battle_memory_won_desc_c_second_perspective
		}
	}

	third_perspective_description = {
		random_valid = {
			desc = bc_battle_memory_won_desc_third_perspective
			desc = bc_battle_memory_won_desc_b_third_perspective

# --- the nemesis memory -----------------------------------------------------
#
# THE TOP RUNG OF THE RIVALRY LADDER LEFT NOTHING PERMANENT BEHIND. A third
# defeat at the same man's hands sets `set_relation_nemesis`, and a relation is
# exactly the thing that can be cleared - bc_rivalry_effects.txt's own header
# describes the symmetric clear, and a commander who beats the man back down
# loses the relation and every trace of the three fields with it. That is the
# same argument bc_shieldbrother_memory was written to answer, pointed the other
# way: opinion and relations fade, a memory can still be asked about in twenty
# years.
#
# ONE MEMORY PER RIVALRY, NOT ONE PER DEFEAT. The call site fires it only on the
# loss that makes the count exactly three, so the fourth and fifth beatings add
# nothing. The memory is the moment a run of bad luck became a name.
#
# THE LOSER REMEMBERS AND THE WINNER DOES NOT, which is the one place this type
# breaks the shieldbrother pattern of writing both sides. Being beaten three
# times by one man is a thing that happens to somebody; beating a man three times
# is a season's work and belongs to no single field, so there is no day for the
# winner's memory to be about. The `participants` tag still names him, so the
# winner appears in the sentence - he simply does not carry his own copy.
#
# `rival.dds` IS A VANILLA ICON, used by vanilla's own `became_rivals` in
# character_memories_1.txt, and the category set `negative relationships rival`
# is copied from the same block. Same rule as `battle.dds` above: the icon must
# already exist in the vanilla set.
bc_nemesis_memory = {
	categories = { negative war relationships rival }
	icon = "rival.dds"

	description = {
		random_valid = {
			desc = bc_nemesis_memory_desc
			desc = bc_nemesis_memory_desc_b
			desc = bc_nemesis_memory_desc_c
		}
	}

	second_perspective_description = {
		random_valid = {
			desc = bc_nemesis_memory_desc_second_perspective
			desc = bc_nemesis_memory_desc_b_second_perspective
			desc = bc_nemesis_memory_desc_c_second_perspective
		}
	}

	third_perspective_description = {
		random_valid = {
			desc = bc_nemesis_memory_desc_third_perspective
			desc = bc_nemesis_memory_desc_b_third_perspective
			desc = bc_nemesis_memory_desc_c_third_perspective
		}
	}

	participants = { nemesis }

	duration = {
		years = 40
	}
}

# --- the man who came and got him -------------------------------------------
#
# THE SECOND PERMANENT THING A BATTLE CAN LEAVE BEHIND, and the first one that
# runs between a knight and the man commanding him. bc_shieldbrother_memory
# pairs two knights and bc_nemesis_memory pairs two commanders; the vertical
# relation - the man in the ranks and the man whose day it was - had nothing.
#
# THE RESCUED MAN CARRIES IT AND THE KNIGHT DOES NOT, which is the same split
# bc_nemesis_memory makes and for a related reason. Being pulled out of a press
# is one moment and it happens to one man. Doing the pulling is a thing a good
# knight does more than once, and giving him his own copy every time would fill
# a career soldier's memory list with the same sentence. The `participants` tag
# names him, so he is in the sentence either way.
#
# THE OPINION IS THE KNIGHT'S SIDE OF IT. bc_lifesaver_opinion runs from the
# rescued man toward the knight at 0.02 a month, which is slower than anything
# else in that file; between them the memory and the opinion say the same thing
# from the two ends.
#
# `friend.dds` IS A VANILLA ICON - it is in the set used by vanilla's own
# character_memory_types - and the category set is the shieldbrother's with
# `martial` in place of nothing: this is a debt, not a friendship, but the UI
# has no token for that.
bc_lifesaver_memory = {
	categories = { positive war martial relationships friend }
	icon = "friend.dds"

	description = {
		random_valid = {
			desc = bc_lifesaver_memory_desc
			desc = bc_lifesaver_memory_desc_b
			desc = bc_lifesaver_memory_desc_c
		}
	}

	second_perspective_description = {
		random_valid = {
			desc = bc_lifesaver_memory_desc_second_perspective
			desc = bc_lifesaver_memory_desc_b_second_perspective
			desc = bc_lifesaver_memory_desc_c_second_perspective
		}
	}

	third_perspective_description = {
		random_valid = {
			desc = bc_lifesaver_memory_desc_third_perspective
			desc = bc_lifesaver_memory_desc_b_third_perspective
			desc = bc_lifesaver_memory_desc_c_third_perspective
		}
	}

	participants = { lifesaver }

	duration = {
		years = 40
	}
}
