namespace = bc_chronicles

# Battle Chronicles - the eve of a battle.
#
# ROOT IS THE COMBAT. This file holds the only `scope = combat` event in the mod.
# Nothing in vanilla or AGOT declares that scope; DIAG Q2 measured that it fires,
# 152 times over a session with 143 combat ends. If this file ever stops producing
# `BC: eve fired`, the scope is the first suspect and nothing else in the mod is
# affected.
#
# on_combat_start is EMPTY in the loaded file - AGOT combat_on_actions.txt:3-4 is
# a bare `on_combat_start = { }`. An entirely unclaimed hook.
#
# THE TRAP THIS FILE EXISTS TO RECORD, AND THE ROUTE THAT REPLACED IT.
# `combat_attacker` and `combat_defender` DO NOT RESOLVE from inside this event.
# Both were probed as `combat_<side>.side_commander ?= { debug_log = ... }` across
# those same 152 fires and BOTH LOGGED ZERO. The `?=` swallowed the failure in
# silence - a `?=` block that never runs is indistinguishable from one that had
# nothing to do, which is why this needed a marker to detect at all.
#
# The eve therefore reaches the two sides through `every_combat_side` instead
# (effects.log: scope combat, target combat_side), and each side finds its own
# opposite through `enemy_side`, which bc_battle_is_notable_trigger already uses.
# THIS ROUTE IS ALSO UNPROVEN FROM `scope = combat` - it is unproven in exactly
# the way the two links that failed were unproven - so it ships behind its own
# markers. `BC: eve side` says the iteration ran; `BC: eve commander` says a man
# was reached. If the first appears without the second, the iteration works and
# side_commander is the failure; if neither appears, the iteration is the failure.
# Do not assume a scope link resolves because the API dump lists it - the dump is
# authoritative about existence and scope type, not about what resolves at a given
# call site.
#
# `events = { }` rather than `effect = { }` in the on_action, for the reason
# bc_on_actions.txt gives: an `effect = { }` in an on_action does not merge across
# files, and the winner is decided by alphabetical filename.

# ---------------------------------------------------------------------------
# The eve. Hidden, combat-scoped, raised by on_combat_start.
# ---------------------------------------------------------------------------
bc_chronicles.0010 = {
	hidden = yes
	scope = combat

	# THE CHEAP ORDER. Both rows are evaluated at combat scope, before anything
	# iterates or steps into a character. The size gate rejects the overwhelming
	# majority of the world's combats in one trigger.
	#
	# num_total_troops IS WRITTEN BARE HERE, and this is the one place in the mod
	# where that is right: it is scope `combat` and Root IS the combat.
	# bc_battle_is_notable_trigger has to write `combat = { num_total_troops }`
	# only because it starts from a combat SIDE.
	#
	# 5000, DELIBERATELY HALF THE NOTABILITY FLOOR OF 10000. A battle can be worth
	# an eve without turning out to be worth remembering, and the eve has to decide
	# before anyone knows which it is.
	#
	# NO OWN GAME RULE. bc_aftermath_off is the mod's "no combat-end and no
	# combat-start ceremony" switch and the eve is ceremony.
	trigger = {
		NOT = { has_game_rule = bc_aftermath_off }
		num_total_troops >= 5000
	}

	immediate = {
		debug_log = "BC: eve fired"

		# ONE PASS, NOT TWO. The plan this replaces needed two passes over
		# combat_attacker and combat_defender because each side's offer needed
		# the OTHER side's commander, and a single interleaved pass would have
		# offered to the attacker before the defender was saved. Reaching the
		# opposite side through enemy_side removes the ordering problem
		# entirely: each pass computes its own foe.
		#
		# BOTH SIDES ARE VISITED because either commander may be the player -
		# and at most one can be, which is also what makes the bc_eve_foe save
		# inside bc_eve_offer_effect safe under an iteration.
		#
		# THE SAVED SIDE SCOPE IS OVERWRITTEN EVERY PASS AND THAT IS FINE - it
		# is read inside the same pass that writes it, and nothing outside the
		# iteration reads it.
		every_combat_side = {
			debug_log = "BC: eve side"
			save_scope_as = bc_eve_side
			side_commander ?= {
				debug_log = "BC: eve commander"
				bc_eve_offer_effect = {
					FOE = scope:bc_eve_side.enemy_side.side_commander
				}
			}
		}
	}
}
