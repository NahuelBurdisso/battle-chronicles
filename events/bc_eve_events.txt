namespace = bc_chronicles

# Battle Chronicles - the opening of a battle.
#
# THIS FILE ONCE HELD A SECOND, HIDDEN `scope = combat` EVENT ON on_combat_start,
# AND IT IS GONE FOR A MEASURED REASON. Over 22 qualifying combats it fired 22
# times, its `every_combat_side` ran 44 times - exactly two sides per combat, so
# the scope and the iteration both worked - and `side_commander` produced a
# commander ZERO times. The same link reached a commander 227 times in the same
# session from mid-battle call sites. NO COMMANDER IS ASSIGNED TO A COMBAT SIDE
# AT on_combat_start. No routing fixes that, so the opening moved into the battle:
# it is raised from the end of bc_battle_open_effect, on the battle's first
# moment. See bc_eve_effects.txt.
#
# That also re-explains the older DIAG Q2 finding recorded in the README -
# `combat_attacker` / `combat_defender` were very likely never the problem; there
# was simply nobody there to find.
#
# GENERAL LESSON, KEPT BECAUSE IT COST TWO RUNS: a `?=` block that never runs is
# indistinguishable from one that had nothing to do. Every speculative scope link
# in this mod ships behind its own debug_log for exactly that reason.

# ---------------------------------------------------------------------------
# The opening, as the player sees it. A character_event on the commander.
#
# THE ONLY MOMENT IN THE MOD WHERE THE PLAYER COMMITS AHEAD OF KNOWING ANYTHING.
# Every other choice answers something that has already happened. This one is
# taken as the lines form - on the first moment of the battle, before any of it
# has resolved.
#
# FOUR OPTIONS APPLY A FAMILY-2 MODIFIER BY NAME, and these are the only call
# sites in the mod that do. Everywhere else the (KIND, SWING) table inside
# bc_record_moment_effect decides which modifier a moment implies. There is no
# moment here - nothing has happened yet - so there is no seam call to route
# through and the option names the posture directly. It still goes through
# bc_add_verb_effect rather than add_character_modifier, so the 20-day duration
# stays in one place per family.
#
# NO OPTION APPLIES MOMENTUM. The battle has not started. The eve is about shape,
# not advantage, and momentum is the tide.
#
# THE THIRD OPTION IS NOT FREE. Declining to commit is a choice, and a commander
# who never commits should not be indistinguishable from one who does.
# ---------------------------------------------------------------------------
bc_chronicles.0011 = {
	type = character_event
	title = bc_chronicles.0011.t
	theme = martial

	# THREE PARAGRAPHS, ASSEMBLED: what is happening, who is across the field,
	# what the field is. `desc` blocks concatenate in order; `first_valid` picks
	# exactly one of its children. Each chain ends in a trigger-less child for
	# the same reason the prose fragments have a fallback arm - an eve whose foe
	# or province did not resolve must still render a whole paragraph, not a gap.
	#
	# BOTH NEW CHAINS READ SAVED SCOPES, NOT SCRATCH. See bc_eve_effects.txt: the
	# pop-up renders when the player gets round to it, which can be after the
	# combat ended and bc_clear_scratch_effect ran.
	#
	# THE FOE CHAIN CARRIES THE SAME FIVE VALUES AS THE BOND LADDER BUT IN A
	# DIFFERENT ORDER - nemesis, kin, heathen, countryman, stranger, where the
	# ladder in bc_write_bond_effect runs nemesis, kin, countryman, heathen,
	# stranger. Heathen is lifted above countryman ON PURPOSE: a man of your own
	# people who prays to something else is the more interesting sentence, and
	# under the ladder's order he never gets one. The two orders disagree and
	# that is the intent - the ladder writes a fact, this chain picks a line.
	#
	# THE stranger ARM IS A TRUE FALLBACK AND HAS TO STAY REACHABLE ONLY BY
	# STRANGERS. Its text says "a banner nobody here can name", which is a lie
	# about a countryman. That is why the countryman arm is here and not dropped:
	# without it the largest match-up class on the map falls through to it.
	desc = {
		desc = bc_chronicles.0011.desc_intro
		first_valid = {
			triggered_desc = {
				trigger = { bc_eve_facing_nemesis_trigger = yes }
				desc = bc_chronicles.0011.desc_nemesis
			}
			triggered_desc = {
				trigger = {
					exists = scope:bc_eve_foe
					trigger_if = {
						limit = { exists = scope:bc_eve_foe }
						scope:bc_eve_foe = { is_close_family_of = root }
					}
				}
				desc = bc_chronicles.0011.desc_kin
			}
			triggered_desc = {
				trigger = {
					exists = scope:bc_eve_foe
					trigger_if = {
						limit = { exists = scope:bc_eve_foe }
						NOT = { scope:bc_eve_foe = { faith = root.faith } }
					}
				}
				desc = bc_chronicles.0011.desc_heathen
			}
			triggered_desc = {
				trigger = {
					exists = scope:bc_eve_foe
					trigger_if = {
						limit = { exists = scope:bc_eve_foe }
						scope:bc_eve_foe = { culture = root.culture }
					}
				}
				desc = bc_chronicles.0011.desc_countryman
			}
			desc = bc_chronicles.0011.desc_stranger
		}
		# THE GROUND CHAIN HAS FIVE NAMED BANDS PLUS A FALLBACK. The fallback
		# text ("The field is what it is. Nothing on it will decide anything for
		# you.") is the open-ground text, and `open` is the only band that
		# legitimately rides it — an unresolved province also reaches it, which
		# is the correct failure. Jungle was added after the terrain census at
		# README.md Q5 measured jungle at 8/137 and desert at 0/137: the
		# fallback sentence is false about a jungle in the same way the
		# `stranger` arm's text is false about a countryman, and a band measured
		# at 8/137 earns a bespoke line. Desert's zero is a sample result, not
		# a map truth — Dorne and Essos exist — so its arm stays.
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:bc_eve_ground
					trigger_if = {
						limit = { exists = scope:bc_eve_ground }
						scope:bc_eve_ground = { bc_ground_is_rough_trigger = yes }
					}
				}
				desc = bc_chronicles.0011.desc_ground_rough
			}
			triggered_desc = {
				trigger = {
					exists = scope:bc_eve_ground
					trigger_if = {
						limit = { exists = scope:bc_eve_ground }
						scope:bc_eve_ground = { bc_ground_is_forest_trigger = yes }
					}
				}
				desc = bc_chronicles.0011.desc_ground_forest
			}
			triggered_desc = {
				trigger = {
					exists = scope:bc_eve_ground
					trigger_if = {
						limit = { exists = scope:bc_eve_ground }
						scope:bc_eve_ground = { bc_ground_is_jungle_trigger = yes }
					}
				}
				desc = bc_chronicles.0011.desc_ground_jungle
			}
			triggered_desc = {
				trigger = {
					exists = scope:bc_eve_ground
					trigger_if = {
						limit = { exists = scope:bc_eve_ground }
						scope:bc_eve_ground = { bc_ground_is_winter_trigger = yes }
					}
				}
				desc = bc_chronicles.0011.desc_ground_winter
			}
			triggered_desc = {
				trigger = {
					exists = scope:bc_eve_ground
					trigger_if = {
						limit = { exists = scope:bc_eve_ground }
						scope:bc_eve_ground = { bc_ground_is_desert_trigger = yes }
					}
				}
				desc = bc_chronicles.0011.desc_ground_desert
			}
			desc = bc_chronicles.0011.desc_ground_open
		}
	}

	left_portrait = {
		character = root
		animation = personality_bold
	}

	immediate = {
		debug_log = "BC: eve popup 0011"
	}

	# NO `trigger` ON THIS EVENT. Both gates were taken in bc_eve_offer_effect
	# before the raise - re-testing them here would let a commander who passed
	# the gate lose the event to a race and take the 90-day cooldown for nothing.

	option = {
		name = bc_chronicles.0011.a

		bc_add_verb_effect = { MODIFIER = bc_bloodied }
		add_stress = 10
	}

	option = {
		name = bc_chronicles.0011.b

		bc_add_verb_effect = { MODIFIER = bc_shieldwall }
	}

	option = {
		name = bc_chronicles.0011.c

		add_prestige = -25
	}

	# TWO CONTEXTUAL OPTIONS. `trigger` on an event option HIDES it when false,
	# which is the behaviour wanted: an option greyed out on every plain would
	# advertise a mechanic the player cannot reach. On open ground against a
	# stranger this event is exactly what it was before.
	#
	# NEITHER APPLIES MOMENTUM. The file header states the rule: the battle has
	# not started, the eve is about shape, not advantage. Both go through
	# bc_add_verb_effect, which is family 2.
	#
	# BOTH TRIGGERS READ SCOPES, so a late-answered eve still renders the same
	# option set it was offered with. That is the whole reason bc_eve_ground is a
	# scope and not scratch.
	#
	# OPTION d IS DELIBERATELY NARROWER THAN THE GROUND DESC CHAIN. The chain
	# names five bands; this gate names two. They answer different questions:
	# the chain says what the ground IS, the option says whether the ground can
	# be HELD. bc_ground_held is +army_toughness / -army_pursuit, a line that
	# does not break and does not chase - which is rough and forest. Jungle
	# gained a desc arm without gaining this option on purpose: its paragraph
	# says men who go in there are gone from the battle, which is dispersal,
	# the opposite of a held line. Winter and desert are the same judgement.
	# If a band is ever added here, the modifier is the argument, not the
	# paragraph.
	#
	# DECISION TABLE for the two markers below. Their denominator is eves
	# ANSWERED, which is neither the moment count nor `BC: eve offered`:
	#   both 0 while `BC: eve offered` counts  the options are never reachable,
	#       or the eve is being dismissed. Not a trigger bug on its own.
	#   `eve option ground` 0 while `BC: eve ground missing` is 0  the ground
	#       scope resolves and the bands still never match: check the trigger.
	#   either counting                        the option was taken. There is no
	#       marker for "offered but not taken"; CK3 gives no hook for it.
	option = {
		name = bc_chronicles.0011.d
		trigger = {
			exists = scope:bc_eve_ground
			trigger_if = {
				limit = { exists = scope:bc_eve_ground }
				scope:bc_eve_ground = {
					OR = {
						bc_ground_is_rough_trigger = yes
						bc_ground_is_forest_trigger = yes
					}
				}
			}
		}
		bc_add_verb_effect = { MODIFIER = bc_ground_held }
		debug_log = "BC: eve option ground"
	}

	# bc_eve_facing_nemesis_trigger already guards its own reads - it is the
	# trigger the desc branch above uses - so this needs no wrapper of its own.
	option = {
		name = bc_chronicles.0011.e
		trigger = { bc_eve_facing_nemesis_trigger = yes }
		bc_add_verb_effect = { MODIFIER = bc_bloodied }
		add_stress = 5
		debug_log = "BC: eve option nemesis"
	}
}
