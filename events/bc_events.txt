# Battle Chronicles - the player-facing moments.
#
# THIS FILE HOLDS POP-UPS ONLY. The hidden combat-end graders live next door in
# bc_aftermath_events.txt. Keeping the player-facing events in a file of their
# own is what makes it obvious at a glance which of the mod's events the player
# can actually see and click.
#
# Five templates out of the sixty-two ever become a pop-up, and only when the
# player is personally in the moment and is not inside the 60-day pop-up
# cooldown. Everything else is a toast, because that is what vanilla and AGOT do
# and the mod should read as native.
#
# FOUR OF THE FIVE ROOT ON A COMMANDER. bc_chronicles.0005 is the exception and
# roots on a KNIGHT, which changes what its options are allowed to do: a knight
# carries none of the army modifiers, so every rung and every ledger write in
# that event is wrapped in `scope:bc_chronicle_subject = { }` - the commander,
# saved before the raise. See its own header.
#
# Nothing in here touches combat scopes. A triggered event runs on the character,
# and momentum is a character modifier, so the outcomes work without carrying
# scope:combat_side across the trigger_event boundary.
#
# The commander phase event applies NO momentum before firing these - the option
# is the outcome. Every option therefore has to leave the commander somewhere.

# ---------------------------------------------------------------------------
# The banner falls.
# ---------------------------------------------------------------------------
bc_chronicles.0001 = {
	type = character_event
	title = bc_chronicles.0001.t
	desc = bc_chronicles.0001.desc
	theme = martial

	left_portrait = {
		character = root
		animation = personality_bold
	}

	immediate = {
		debug_log = "BC: popup 0001 banner falls"
	}

	option = {
		name = bc_chronicles.0001.a
		trigger = { prowess >= 8 }

		bc_momentum_minor_effect = yes
		bc_record_moment_effect = { TIER = minor KIND = entrenched SWING = up }
		add_prestige = 100
		add_stress = 15
	}

	option = {
		name = bc_chronicles.0001.b

		bc_faltering_minor_effect = yes
		bc_record_moment_effect = { TIER = minor KIND = measured SWING = down }
	}

	option = {
		name = bc_chronicles.0001.c

		bc_faltering_major_effect = yes
		bc_record_moment_effect = { TIER = decisive KIND = wild SWING = down }
		add_prestige = 50
	}

	# ===================================================================
	# THE CONTEXTUAL OPTIONS, 2026-08-17. Same idea as the eve's f, g and h:
	# an option that only some men are offered, so two players meeting the
	# same pop-up do not meet the same choice.
	#
	# THEY CAN ONLY ASK ABOUT THE MAN, NEVER ABOUT THE BATTLE. This file's
	# header states the constraint and it is real: a pop-up is a
	# character_event reached by trigger_event, so scope:combat_side is not
	# carried across the boundary and there is nothing here to ask about the
	# ground, the odds or the foe. Traits, stats and purse are the whole
	# vocabulary available, which is why every contextual option in the mod's
	# three pop-ups is gated on one of those.
	#
	# NEITHER IS A CHEAPER `a`. Option a is the brave answer and costs
	# stress; d is the same answer bought with faith, and e is the same
	# answer bought with money. All three leave the commander with a rung,
	# which is the point - the pop-up is the outcome, so an option that led
	# nowhere would be a dead click.
	# ===================================================================

	# The zealot's answer. bc_the_zealot_will_not_stop is the trait's phase
	# event and gates on a bloody field; this is the same man on a bad day
	# with no field condition at all, because a pop-up cannot ask about one.
	#
	# `bloodied` AND NOT `entrenched`, unlike option a. Both grant a rung, but
	# a is a line that will not move and this is a line that is spending
	# itself; bc_bloodied trades toughness for damage and is the honest
	# reading of men who have been told the gods are watching.
	option = {
		name = bc_chronicles.0001.d
		trigger = { has_trait = zealous }

		bc_momentum_minor_effect = yes
		bc_record_moment_effect = { TIER = minor KIND = bloodied SWING = up }
		add_piety = 50
		add_stress = 10
		debug_log = "BC: popup 0001 option zealous"
	}

	# Bought. `remove_short_term_gold` and not `add_gold = -150`, for the
	# reason proven in game and recorded at bc_chronicles.0002.b: a negative
	# add_gold throws a script error and the option silently pays nothing.
	# This is the second gold option in the mod and it uses the same call.
	#
	# 150 AND NOT 100. The rally option charges 100 to climb out of faltering;
	# this charges more to avoid falling into it, because a man who still has
	# a line to hold is in the better position and should not get the cheaper
	# price.
	option = {
		name = bc_chronicles.0001.e
		trigger = { gold >= 150 }

		remove_short_term_gold = 150
		bc_momentum_minor_effect = yes
		bc_record_moment_effect = { TIER = minor KIND = entrenched SWING = up }
		debug_log = "BC: popup 0001 option paid"
	}
}

# ---------------------------------------------------------------------------
# Rally the line. Only reachable while faltering, so every option is a way out
# of a hole - they differ in what the way out costs.
# ---------------------------------------------------------------------------
bc_chronicles.0002 = {
	type = character_event
	title = bc_chronicles.0002.t
	desc = bc_chronicles.0002.desc
	theme = martial

	left_portrait = {
		character = root
		animation = personality_bold
	}

	immediate = {
		debug_log = "BC: popup 0002 rally the line"
	}

	option = {
		name = bc_chronicles.0002.a
		trigger = { prowess >= 10 }

		bc_rally_effect = yes
		bc_record_moment_effect = { TIER = major KIND = ordered SWING = up }
		add_stress = 10
	}

	option = {
		name = bc_chronicles.0002.b
		trigger = { gold >= 100 }

		# NOT `add_gold = -100`. add_gold rejects a negative value outright:
		#   Script system error! Error: add_gold effect [ Negative value in: {} ]
		# and the option silently does nothing but rally for free. Verified in game.
		remove_short_term_gold = 100
		bc_rally_effect = yes
		bc_record_moment_effect = { TIER = minor KIND = measured SWING = up }
	}

	option = {
		name = bc_chronicles.0002.c

		bc_clear_faltering_effect = yes
		bc_record_moment_effect = { TIER = minor KIND = entrenched SWING = up }
	}

	# The brave man's answer, and the only option in the mod that hands the
	# side a PENALTY on the way to a good outcome: `wild` maps to
	# bc_spent_horse, so this rallies the line and blows the horses doing it.
	#
	# IT IS GATED ON THE TRAIT AND NOT ON PROWESS, which is what separates it
	# from option a. a is the commander riding the line to be seen, and asks
	# whether he is good enough to be believed doing it; this is the commander
	# throwing whatever is still mounted at the men doing the pushing, and asks
	# only whether he is the sort to order it. A brave commander with prowess 6
	# has no other way out of this pop-up except paying for it or standing
	# still.
	option = {
		name = bc_chronicles.0002.d
		trigger = { has_trait = brave }

		bc_rally_effect = yes
		bc_record_moment_effect = { TIER = minor KIND = wild SWING = up }
		add_stress = 15
		debug_log = "BC: popup 0002 option brave"
	}
}

# The van breaks through.
# ---------------------------------------------------------------------------
bc_chronicles.0003 = {
	type = character_event
	title = bc_chronicles.0003.t
	desc = bc_chronicles.0003.desc
	theme = martial

	left_portrait = {
		character = root
		animation = war_attacker
	}

	immediate = {
		debug_log = "BC: popup 0003 van breaks through"
	}

	option = {
		name = bc_chronicles.0003.a

		bc_momentum_major_effect = yes
		bc_record_moment_effect = { TIER = major KIND = bloodied SWING = up }
		add_stress = 10
	}

	option = {
		name = bc_chronicles.0003.b

		bc_momentum_minor_effect = yes
		bc_record_moment_effect = { TIER = minor KIND = measured SWING = up }
		add_prestige = 50
	}

	# Let them have the baggage. THE ONLY OPTION IN THE MOD THAT PAYS THE
	# PLAYER AND CHARGES THE ARMY: gold in hand, `wild` on the record, and
	# therefore bc_spent_horse on the side. It is the commander-scale version
	# of bc_knight_strips_the_dead and it is deliberately the same trade.
	#
	# NO RUNG IN EITHER DIRECTION. Faltering would be too much - the side is
	# breaking through, and a sack does not undo that - and momentum would
	# pretend the disorder was free.
	option = {
		name = bc_chronicles.0003.c
		trigger = { has_trait = greedy }

		add_gold = 100
		bc_record_moment_effect = { TIER = minor KIND = wild SWING = down }
		debug_log = "BC: popup 0003 option plunder"
	}

	# The trained answer, gated on martial rather than on a trait so that it
	# is reachable by a man with no relevant personality at all - two of the
	# five contextual options in this file are trait-gated and a third purse-
	# gated, and a commander can be none of those things.
	#
	# `ordered`/`up` IS bc_pursuit, which is the one KIND the verb table
	# reads SWING on, and this is the only place in the mod where a player
	# CHOOSES it. Wheeling a broken van onto the enemy flank is what the
	# pursuit modifier is for.
	option = {
		name = bc_chronicles.0003.d
		trigger = { martial >= 14 }

		bc_momentum_minor_effect = yes
		bc_record_moment_effect = { TIER = minor KIND = ordered SWING = up }
		add_prestige = 25
		debug_log = "BC: popup 0003 option wheel"
	}
}

# ---------------------------------------------------------------------------
# THE FOURTH POP-UP, 2026-08-17. The flank goes.
#
# THREE OF THE MOD'S THIRTY-ONE COMMANDER TEMPLATES HAVE EVER ASKED THE PLAYER
# ANYTHING, and they have been the same three since plan 1: the banner, the
# rally and the van. bc_flank_collapses is the fourth-oldest template in the
# file and the best remaining candidate - it is a decisive setback with several
# real answers, and unlike bc_kin_across_the_line it needs to name nobody.
#
# THAT IS WHY IT IS THIS TEMPLATE AND NOT THE KIN ONE. A pop-up cannot see
# scope:combat_side, so it cannot reach the man across the field, and a kin
# pop-up whose text could not say who the kinsman was would be worse than the
# toast it replaced. The flank is a thing, not a person.
#
# THE PHASE EVENT APPLIES NO RUNG BEFORE RAISING THIS, exactly as the other
# three do. The option is the outcome; see this file's header. Its `else` arm
# still applies bc_faltering_major_effect, which is what the AI and any player
# inside the pop-up cooldown continue to get.
#
# ALL FOUR OPTIONS APPLY A FALTERING RUNG, AND THEY DIFFER IN WHICH ONE. This
# is the only pop-up in the mod where every answer is a bad answer - the flank
# has already gone by the time the player is asked - so the choice is between
# `minor` bought with something and `major` bought with nothing. There is no
# option here that comes out ahead, and inventing one would make the collapse
# free.
#
# a IS THE MOD'S SECOND `entrenched`/`down` CALL SITE. The pair census in
# bc_ledger_effects.txt had it at one, which is the thinnest cell that is not
# empty, and a line bending back onto itself and bracing is exactly what the
# pair was written to mean.
# ---------------------------------------------------------------------------
bc_chronicles.0004 = {
	type = character_event
	title = bc_chronicles.0004.t
	desc = bc_chronicles.0004.desc
	theme = martial

	left_portrait = {
		character = root
		animation = personality_bold
	}

	immediate = {
		debug_log = "BC: popup 0004 flank collapses"
	}

	# The controlled answer. It costs him the day and buys the cheaper rung.
	option = {
		name = bc_chronicles.0004.a

		bc_faltering_minor_effect = yes
		bc_record_moment_effect = { TIER = minor KIND = entrenched SWING = down }
		add_stress = 10
	}

	# The cheap answer, and the honest one: he pays nothing and the side does.
	# This is what the `else` arm of the phase event does for everybody who
	# never sees this pop-up, which is the right default to leave in the list.
	option = {
		name = bc_chronicles.0004.b

		bc_faltering_major_effect = yes
		bc_record_moment_effect = { TIER = major KIND = measured SWING = down }
	}

	# He goes himself. `wild`/`down` puts bc_spent_horse on the side, which is
	# the price of throwing everything still mounted at a hole, and the tier is
	# major because it is a decision the whole army sees him make.
	#
	# IT IS THE BEST OUTCOME AND THE MOST EXPENSIVE. Fifteen stress and a
	# blown mounted reserve against option b's nothing.
	option = {
		name = bc_chronicles.0004.c
		trigger = { prowess >= 10 }

		bc_faltering_minor_effect = yes
		bc_record_moment_effect = { TIER = major KIND = wild SWING = down }
		add_prestige = 100
		add_stress = 15
		debug_log = "BC: popup 0004 option rides"
	}

	# The trained answer, gated on martial for the reason bc_chronicles.0003.d
	# is: a commander can have no relevant trait and no prowess and still have
	# been taught this.
	#
	# `ordered`/`down` GRANTS NOTHING, and here that is the whole point - the
	# second line comes round to face the gap and no modifier lands on either
	# army. It is the only answer in this pop-up that costs him neither stress
	# nor horses.
	option = {
		name = bc_chronicles.0004.d
		trigger = { martial >= 14 }

		bc_faltering_minor_effect = yes
		bc_record_moment_effect = { TIER = minor KIND = ordered SWING = down }
		add_prestige = 25
		debug_log = "BC: popup 0004 option refuse"
	}
}

# ---------------------------------------------------------------------------
# The commander is down in the press, and the player is one of his knights.
#
# THE FIRST POP-UP IN THE MOD THAT DOES NOT ROOT ON A COMMANDER. Four pop-ups
# shipped over two content programmes and every one of them assumes the player
# is the man giving orders. In AGOT a player is very often somebody's sworn
# sword instead, and this mod has never once asked him anything.
#
# ROOT IS THE KNIGHT AND scope:bc_chronicle_subject IS THE COMMANDER. The
# second is saved by bc_battle_open_effect, which bc_knight_pulls_him_out now
# calls BEFORE the raise for exactly this purpose - a scope saved before
# trigger_event crosses the boundary where scope:combat_side does not.
#
# EVERY RUNG AND EVERY LEDGER WRITE IS WRAPPED IN THAT SCOPE. bc_momentum_*,
# bc_faltering_* and bc_record_moment_effect all act on the character they are
# called on, and on this side of the mod that character is the commander, never
# the knight - which is why the four commander pop-ups can write them bare and
# this one cannot. Getting this wrong would not error; it would hang the army
# modifier on a knight, where nothing reads it.
#
# THE MEMORY USES `root`, NOT scope:bc_knight. Both name the same man and the
# phase event saves the scope, but root cannot fail and needs no argument about
# what survives the raise.
#
# NO OPTION IS FREE AND NO OPTION IS SAFE. He is being asked whether to leave
# his place in a line that is already losing, so the two that go in cost stress
# and the one that stays costs him the man.
# ---------------------------------------------------------------------------
bc_chronicles.0005 = {
	type = character_event
	title = bc_chronicles.0005.t
	desc = bc_chronicles.0005.desc
	theme = martial

	left_portrait = {
		character = root
		animation = personality_bold
	}

	# THE FIRST right_portrait IN THE MOD, and the reason is the same as the
	# reason this event exists: it is the only pop-up about two men. `fear` is a
	# vanilla animation - vanilla writes it at events/activities/
	# coronation_activity/coronation_events.txt:3745 and AGOT at
	# agot_coronation_events.txt:3850.
	right_portrait = {
		character = scope:bc_chronicle_subject
		animation = fear
	}

	immediate = {
		debug_log = "BC: popup 0005 pulls him out"
	}

	# He goes. This is the phase event's `else` arm made into a choice, and it
	# is priced: the AI knight who never sees this pop-up pays nothing for it
	# because nobody asked him.
	option = {
		name = bc_chronicles.0005.a

		add_prestige = 100
		add_stress = 10

		scope:bc_chronicle_subject ?= {
			add_opinion = {
				target = root
				modifier = bc_lifesaver_opinion
				opinion = 40
			}

			create_character_memory = {
				type = bc_lifesaver_memory
				participants = {
					lifesaver = root
				}
			}

			bc_record_moment_effect = { TIER = major KIND = adapted SWING = up }
		}
		debug_log = "BC: popup 0005 option goes"
	}

	# He stays. THIS IS NOT THE COWARD'S ANSWER and the loc must not write it as
	# one - a knight who leaves his place in a losing line to go after one man
	# is making a hole, and the men either side of him know it.
	#
	# It is the only option that leaves no memory and no opinion, because
	# nothing happened between the two of them. The side gets bc_shieldwall out
	# of it, guarded like every other momentum grant.
	option = {
		name = bc_chronicles.0005.b

		scope:bc_chronicle_subject ?= {
			if = {
				limit = { NOT = { bc_has_momentum_trigger = yes } }
				bc_momentum_minor_effect = yes
			}

			bc_record_moment_effect = { TIER = minor KIND = entrenched SWING = up }
		}
		debug_log = "BC: popup 0005 option holds"
	}

	# The expensive one. He does not go alone - he takes the men either side of
	# him with him, which gets the commander out and blows the whole wedge open
	# doing it. `wild`/`up` is bc_spent_horse: the tide moves and the animals
	# pay for the moving.
	#
	# prowess >= 14 IS HIGHER THAN THE PHASE EVENT'S OWN GATE of 12. The option
	# is not "can he fight", it is "can he fight well enough that other men will
	# follow him into it".
	option = {
		name = bc_chronicles.0005.c
		trigger = { prowess >= 14 }

		add_prestige = 150
		add_stress = 20

		scope:bc_chronicle_subject ?= {
			add_opinion = {
				target = root
				modifier = bc_lifesaver_opinion
				opinion = 40
			}

			create_character_memory = {
				type = bc_lifesaver_memory
				participants = {
					lifesaver = root
				}
			}

			bc_record_moment_effect = { TIER = major KIND = wild SWING = up }
		}
		debug_log = "BC: popup 0005 option wedge"
	}

	# The trained answer, gated on martial for the reason bc_chronicles.0003.d
	# and .0004.d are: a man can have no relevant trait and middling prowess and
	# still have been taught how to get somebody out of a press without walking
	# into it himself.
	#
	# NO MEMORY AND NO OPINION. Somebody else's hands did the pulling, and the
	# memory belongs to the day one man went in after another. He is paid in
	# prestige, which is what organising it in front of witnesses is worth.
	option = {
		name = bc_chronicles.0005.d
		trigger = { martial >= 12 }

		add_prestige = 25

		scope:bc_chronicle_subject ?= {
			bc_record_moment_effect = { TIER = minor KIND = ordered SWING = up }
		}
		debug_log = "BC: popup 0005 option sends"
	}
}
