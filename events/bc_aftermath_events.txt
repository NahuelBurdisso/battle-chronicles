namespace = bc_chronicles

# Battle Chronicles - the combat-end graders.
#
# Reached from on_combat_end_winner / _loser via their `events = { }` list - see
# zz_bc_on_actions.txt for why an `effect = { }` there is discarded. Shape copied
# from vanilla combat_event.0001: hidden, scoped to the combat side, no `type =`.
#
# TWO events, one per hook. Not one: nothing in an event body can tell which of
# on_combat_end_winner / on_combat_end_loser invoked it. is_combat_side_pursuing
# is documented as "is this side the winner", but every use of it in vanilla and
# AGOT is DURING combat inside is_army_in_combat = yes, and its value at combat
# end is unproven. Two events makes the outcome structural instead of inferred.
#
# NEITHER CARRIES A `trigger`. The old cleanup event these two replace had none
# either, deliberately: it was the only caller of bc_cleanup_effect, and putting
# momentum stripping behind the aftermath off switch would leave `advantage +-10`
# modifiers bleeding twenty days into the next battle for any player who touched
# that setting. The game rule therefore lives on an INNER `if` around the grading
# branch only.
#
# The grade runs in `immediate`, not in an event trigger, even though the cited
# precedent historical_artifacts.0100 puts its calc_true_if in the trigger. The
# grading markers have to fire on EVERY combat end, including the ones that fail
# the bar, or they measure only the battles that already passed.
#
# Momentum must not survive the battle that earned it. Flags are deliberately NOT
# cleared here: they are the cooldowns, and they are supposed to outlive the
# battle.
#
# BODY ORDER IS LOAD-BEARING and is part of the design:
#   1. cleanup, OUTSIDE the gate
#   2. grading, INSIDE the gate
#   3. scratch clearing, OUTSIDE the gate, and LAST
# Step 3 must follow step 2 because grading reads scratch. Writing the two-column
# split the natural way — all ungated work together — is exactly wrong.
#
# Scratch clearing arrives in task 12. Until then step 3 is absent, which is
# correct: there is no scratch yet.

# ---------------------------------------------------------------------------
# Winning side.
#
# Everything the losing grader does, plus the two things that only make sense
# for a winner: the cumulative notable-win counter, and the name won on the
# field. The two events are NOT mirror images.
#
# The counter is incremented BEFORE the nickname check, so <N> counts the
# battle just fought.
# ---------------------------------------------------------------------------
bc_chronicles.0091 = {
	hidden = yes
	scope = combat_side

	immediate = {
		# --- measurement only ------------------------------------------------
		#
		# OUTSIDE THE NOTABILITY GATE ON PURPOSE, unlike the dead-knight block
		# below. This marker exists to answer two questions no static check can:
		# how often a wipe actually happens, and whether both sides agree it did.
		# Gating it on notability would only ever show wipes that were already
		# going to be notable, which is the population the row cannot affect.
		#
		# NO STATE. This writes nothing and reads nothing but scope:wipe. It is
		# the same kind of row as the calibration markers, and it stays.
		if = {
			limit = { bc_was_wipe_trigger = yes }
			debug_log = "BC: wipe"
		}

		debug_log = "BC: combat end cleanup"
		side_commander ?= {
			bc_cleanup_effect = yes
		}
		every_side_knight = {
			bc_cleanup_effect = yes
		}

		if = {
			limit = { NOT = { has_game_rule = bc_aftermath_off } }

			if = {
				limit = { bc_battle_is_notable_trigger = yes }
				debug_log = "BC: win notable"

				combat.location ?= {
					save_scope_as = bc_battle_province
				}

				# The province check is an enclosing if, not part of the limit -
				# it is a property of the battle, and re-asking it once per
				# knight is noise.
				if = {
					limit = { exists = scope:bc_battle_province }
					every_side_knight = {
						limit = {
							# Knights die in the combat that just ended. Vanilla
							# guards its own post-battle knight handling the same
							# way (AGOT combat_events.txt:854, :1028; the
							# vanilla equivalents are :813 and :877).
							is_alive = yes
							# See bc_chronicles.0092 for why this is trigger_if.
							trigger_if = {
								limit = { exists = root.side_commander }
								NOT = { this = root.side_commander }
							}
						}
						bc_battle_memory_effect = { TYPE = bc_battle_memory_won }
					}
				}

				# --- the knights who did not come back ----------------------
				#
				# A PRESENCE TEST, NOT AN ITERATION. One aggregate toast per
				# battle per liege. CK3 cannot group an iteration by an arbitrary
				# key, so there is no way to fan the dead out into one toast per
				# distinct liege - and no need, because on a combat side the
				# knights are the side commander's, so raising it once on him is
				# already once per liege.
				#
				# THE SAVE_SCOPE_AS IS NOT OPTIONAL. bc_addressed_toast_effect
				# writes left_icon = $SUBJECT$, a scripted effect does not rebind
				# ROOT, and its liege arm has already changed scope by the time
				# left_icon is evaluated - so SUBJECT must be a saved scope, never
				# `this`.
				#
				# event_toast_effect_bad. This and the rivalry toast are the
				# reason TYPE is a parameter of bc_addressed_toast_effect at all.
				#
				# OUTSIDE THE PROVINCE GATE, INSIDE THE NOTABILITY GATE. A battle
				# whose location does not resolve still killed knights, so the
				# province gate must not swallow this. The notability gate must
				# not be escaped: without it every skirmish in the world raises a
				# toast on somebody.
				#
				# NO is_alive GATE ON THE COMMANDER, unlike everything else in
				# this event. A commander who fell with his knights is exactly the
				# case worth telling, and bc_addressed_toast_effect routes a dead
				# man's toast to his liege because a dead character is never
				# human. Adding is_alive = yes here would silence the worst day
				# the side had.
				#
				# IT LOOKS ACROSS AT enemy_side, AND THAT IS THE WHOLE FIX.
				# Root is the WINNING side, and vanilla does not kill the winner's
				# knights at all - combat_event.1001 walks the winner at AGOT
				# combat_events.txt:846 only to write battle history, and does its
				# killing at :849 inside `enemy_side = { every_side_knight }`. The
				# dead knights of any battle are the LOSER's. This block read its
				# own side for two plans and logged zero over 112 grader runs.
				#
				# THE TOAST GOES TO THE MAN WHO LOST THEM, i.e. the loser's
				# commander, which is also who the loc string has always described.
				#
				# IT DEPENDS ON DISPATCH ORDER, WHICH IS WHY THE FILE IS NAMED
				# zz_bc_on_actions.txt. combat_event.1001 has to have run before
				# this, and both are `events = { }` entries on on_combat_end_winner
				# merged across files in ALPHABETICAL filename order.
				#
				# THREE MARKERS, ONE PER STEP, BECAUSE ONE WAS NOT ENOUGH - TWICE.
				# This block has now returned zero on two separate runs (274 then
				# 28 notable wins) with a single marker sitting inside the innermost
				# `?=`, which is the same trap that hid the eve's failure for two
				# runs before it. A zero from one marker cannot say WHICH step died,
				# and the three candidates need different fixes:
				#
				#   reached, not found  -> combat_event.1001 has not run yet. The
				#                          alphabetical-order assumption is wrong and
				#                          the fix is a different hook, not a rename.
				#   reached and found   -> enemy_side.side_commander is null at this
				#                          point. The `?=` is eating it, and the toast
				#                          needs a different addressee.
				#   nothing at all      -> enemy_side does not step from a combat side
				#                          on this hook, which would contradict
				#                          vanilla's own use at combat_events.txt:850.
				#
				# THESE ARE DIAGNOSTIC AND THEY COME OUT once the question is
				# settled. `BC: dead knights` is the permanent one.
				enemy_side = {
					debug_log = "BC: dk reached enemy side"
					if = {
						limit = { any_side_knight = { is_alive = no } }
						debug_log = "BC: dk found dead"
						side_commander ?= {
							save_scope_as = bc_bereaved_commander
							debug_log = "BC: dead knights"
							bc_addressed_toast_effect = {
								TYPE = event_toast_effect_bad
								TITLE = bc_dead_knights_toast
								DESC = bc_dead_knights_toast_desc
								SUBJECT = scope:bc_bereaved_commander
							}
						}
					}
				}

				side_commander ?= {
					if = {
						limit = { is_alive = yes }
						# NOTE, deviation from the spec: the spec puts is_alive in
						# the nickname limit only, which would let a commander who
						# died winning still take the tally. It is hoisted here
						# instead because the tally exists solely to gate the
						# nickname and a dead man will never command again, so the
						# difference is unobservable - and one gate is clearer than
						# two.
						bc_notable_battle_counter_effect = yes

						# The counter stays ABOVE the chain and stays here: the
						# chain's nickname arm reads `var:bc_notable_battles >= 3`,
						# so the counter must have already incremented for this
						# battle. The nickname gate itself moved INTO the chain
						# (bc_reputation_effects.txt) so that style, terrain and
						# nickname share one precedence ordering and award at most
						# one thing per battle.
						bc_reputation_effect = { OUTCOME = win }

						# Plan 4. AFTER B - the two share no variable, so the order
						# between them is free, but a fixed one is one less thing to
						# reason about later. It must stay BEFORE
						# bc_clear_scratch_effect, which removes bc_battle_foe.
						bc_rivalry_effect = { OUTCOME = win }
					}
				}
			}
			else = {
				debug_log = "BC: win not notable"
			}
		}

		# LAST, and OUTSIDE the gate above. Both properties are load-bearing.
		# Inside the gate, a player who turned aftermath off would accumulate
		# scratch forever. Before the grading branch, the grading would read
		# variables that had just been removed - from plan 3 onward the branch
		# reads bc_battle_style and bc_battle_foe - and would silently never
		# fire while logging a missing-variable line on every combat end in the
		# world. Grouping all the ungated work together is the natural way to
		# write this and is exactly wrong.
		side_commander ?= {
			bc_clear_scratch_effect = yes
		}
	}
}

# ---------------------------------------------------------------------------
# Losing side.
#
# Everything above except the two winner-only items: the cumulative notable-win
# counter and the nickname. The two events are NOT mirror images — the counter is
# wins only, deliberately.
# ---------------------------------------------------------------------------
bc_chronicles.0092 = {
	hidden = yes
	scope = combat_side

	immediate = {
		# --- measurement only ------------------------------------------------
		#
		# Identical to the block in bc_chronicles.0091, deliberately. A marker on
		# one side only would make every wipe look asymmetric and would falsify
		# the one thing the runtime checkpoint is looking for.
		if = {
			limit = { bc_was_wipe_trigger = yes }
			debug_log = "BC: wipe"
		}

		debug_log = "BC: combat end cleanup"
		side_commander ?= {
			bc_cleanup_effect = yes
		}
		every_side_knight = {
			bc_cleanup_effect = yes
		}

		if = {
			limit = { NOT = { has_game_rule = bc_aftermath_off } }

			if = {
				limit = { bc_battle_is_notable_trigger = yes }
				debug_log = "BC: lose notable"

				combat.location ?= {
					save_scope_as = bc_battle_province
				}

				# The province check is an enclosing if, not part of the limit -
				# it is a property of the battle, and re-asking it once per
				# knight is noise.
				if = {
					limit = { exists = scope:bc_battle_province }
					every_side_knight = {
						limit = {
							is_alive = yes
							# We never give a commander a battle memory - vanilla
							# already gives the loser's commander one, every time,
							# with no random gate. Nothing in the API says whether
							# every_side_knight includes him; this costs nothing if
							# it does not.
							#
							# NOTE, deviation from the spec: the spec writes this
							# as a bare `NOT = { this = root.side_commander }`.
							# trigger_if, not a bare NOT: a side can be leaderless,
							# which is why side_commander and combat.location are
							# reached with ?= everywhere else in this event. A bare
							# comparison against a missing scope is an error-log
							# line for every knight of every leaderless battle.
							trigger_if = {
								limit = { exists = root.side_commander }
								NOT = { this = root.side_commander }
							}
						}
						bc_battle_memory_effect = { TYPE = bc_battle_memory_lost }
					}
				}

				# --- NO DEAD-KNIGHT BLOCK HERE, AND IT IS NOT AN OVERSIGHT ---
				#
				# This event used to carry a copy of .0091's dead-knight toast, on
				# the reasoning that a loser told nothing about his dead is a gap.
				# It never fired once. Neither did .0091's copy: zero across 112
				# grader runs and 56 notable battles.
				#
				# VANILLA KILLS KNIGHTS ON ONE SIDE ONLY, AND NOT FROM HERE.
				# combat_event.1001 (AGOT combat_events.txt:849) does its killing
				# inside `enemy_side = { every_side_knight = { ... } }` and it is
				# raised from on_combat_end_winner. So the dead knights of a battle
				# are the LOSER's, and they are made dead on the WINNER's hook.
				# This event runs on on_combat_end_loser and cannot be ordered
				# after that; the winner's own knights are only walked at :846,
				# which writes battle history and kills nobody.
				#
				# The toast therefore lives once, in .0091, reaching across to
				# enemy_side. Do not restore a copy here.

				# Plan 3. `OUTCOME = lose` closes the nickname arm of the chain
				# and nothing else - style and terrain grade EITHER outcome,
				# because a man who loses three battles the same way has still
				# shown you who he is.
				#
				# NO bc_notable_battle_counter_effect HERE. That counter is wins
				# only, and this event's header says so.
				side_commander ?= {
					if = {
						limit = { is_alive = yes }
						bc_reputation_effect = { OUTCOME = lose }

						# Plan 4. `OUTCOME = lose` is the arm that CREATES or
						# ESCALATES a grudge; the winner's `win` arm only ever
						# decrements one that already names him.
						bc_rivalry_effect = { OUTCOME = lose }
					}
				}
			}
			else = {
				debug_log = "BC: lose not notable"
			}
		}

		# LAST, and OUTSIDE the gate above - see bc_chronicles.0091 for why both
		# properties matter.
		side_commander ?= {
			bc_clear_scratch_effect = yes
		}
	}
}
