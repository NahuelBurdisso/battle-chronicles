# Battle Chronicles - the combat-end graders.
#
# Reached from on_combat_end_winner / _loser via their `events = { }` list - see
# bc_on_actions.txt for why an `effect = { }` there is discarded. Shape copied
# from vanilla combat_event.0001: hidden, scoped to the combat side, no `type =`.
#
# TWO events, one per hook. Not one: nothing in an event body can tell which of
# on_combat_end_winner / on_combat_end_loser invoked it. is_combat_side_pursuing
# is documented as "is this side the winner", but every use of it in vanilla and
# AGOT is DURING combat inside is_army_in_combat = yes, and its value at combat
# end is unproven. Two events makes the outcome structural instead of inferred.
#
# NEITHER CARRIES A `trigger`. The old cleanup event these two replace had none
# either, deliberately: it was the only caller of bc_cleanup_effect, and putting
# momentum stripping behind the aftermath off switch would leave `advantage +-10`
# modifiers bleeding twenty days into the next battle for any player who touched
# that setting. The game rule therefore lives on an INNER `if` around the grading
# branch only.
#
# The grade runs in `immediate`, not in an event trigger, even though the cited
# precedent historical_artifacts.0100 puts its calc_true_if in the trigger. The
# grading markers have to fire on EVERY combat end, including the ones that fail
# the bar, or they measure only the battles that already passed.
#
# Momentum must not survive the battle that earned it. Flags are deliberately NOT
# cleared here: they are the cooldowns, and they are supposed to outlive the
# battle.
#
# BODY ORDER IS LOAD-BEARING and is part of the design:
#   1. cleanup, OUTSIDE the gate
#   2. grading, INSIDE the gate
#   3. scratch clearing, OUTSIDE the gate, and LAST
# Step 3 must follow step 2 because grading reads scratch. Writing the two-column
# split the natural way — all ungated work together — is exactly wrong.
#
# Scratch clearing arrives in task 12. Until then step 3 is absent, which is
# correct: there is no scratch yet.

# ---------------------------------------------------------------------------
# Winning side.
#
# Everything the losing grader does, plus the two things that only make sense
# for a winner: the cumulative notable-win counter, and the name won on the
# field. The two events are NOT mirror images.
#
# The counter is incremented BEFORE the nickname check, so <N> counts the
# battle just fought.
# ---------------------------------------------------------------------------
bc_chronicles.0091 = {
	hidden = yes
	scope = combat_side

	immediate = {
		debug_log = "BC: combat end cleanup"
		side_commander ?= {
			bc_cleanup_effect = yes
		}
		every_side_knight = {
			bc_cleanup_effect = yes
		}

		# DIAG. TEMPORARY, deleted at the end of stage 1. Outside every gate on
		# purpose: instrumentation that only runs on the battles which already
		# passed the bar measures nothing but the bar.
		side_commander ?= {
			if = {
				limit = { number_of_commander_traits >= 3 }
				debug_log = "BC: DIAG Q1 traits 3plus"
			}
			else_if = {
				limit = { number_of_commander_traits = 2 }
				debug_log = "BC: DIAG Q1 traits 2"
			}
			else_if = {
				limit = { number_of_commander_traits = 1 }
				debug_log = "BC: DIAG Q1 traits 1"
			}
			else = {
				debug_log = "BC: DIAG Q1 traits 0"
			}
		}

		# DIAG. TEMPORARY. Read from a trigger ladder, not from var:bc_battle_size
		# - it has to measure the world, not the mod, or a bug in
		# bc_battle_open_effect would make the measurement agree with itself.
		if = {
			limit = { combat = { num_total_troops >= 5000 } }
			debug_log = "BC: DIAG Q3 size 5k plus"
		}
		else_if = {
			limit = { combat = { num_total_troops >= 3000 } }
			debug_log = "BC: DIAG Q3 size 3k"
		}
		else_if = {
			limit = { combat = { num_total_troops >= 2000 } }
			debug_log = "BC: DIAG Q3 size 2k"
		}
		else_if = {
			limit = { combat = { num_total_troops >= 1000 } }
			debug_log = "BC: DIAG Q3 size 1k"
		}
		else = {
			debug_log = "BC: DIAG Q3 size under 1k"
		}

		# DIAG Q5. TEMPORARY. A census of where battles actually happen, so
		# plan 3 does not spend a trait arm on a bucket no war is fought in.
		# Resolved here rather than inside the exists = scope:bc_battle_province
		# block further down, which sits behind both the game rule and
		# notability - only ~13% of battles grade notable, which would leave six
		# buckets sharing a handful of samples.
		#
		# AGOT's helpers, never bare `terrain =` keys: AGOT makes road variants
		# distinct terrain types and adds seven bases vanilla has no key for, so
		# ~26% of the map is invisible to a bare check. A DIAG written that way
		# would measure the bug instead of the map.
		#
		# The six arms are exhaustive over all 23 helpers (2+5+2+2+7+5), so
		# NONE MATCHED is unreachable in principle. Any hit means AGOT changed
		# its terrain list and plan 3's ladder is stale the same way.
		combat.location ?= {
			if = {
				limit = {
					OR = {
						agot_is_jungle_terrain = yes
						agot_is_cloudforest_terrain = yes
					}
				}
				debug_log = "BC: DIAG Q5 terrain jungle"
			}
			else_if = {
				limit = {
					OR = {
						agot_is_drylands_terrain = yes
						agot_is_desert_terrain = yes
						agot_is_desert_mountains_terrain = yes
						agot_is_oasis_terrain = yes
						agot_is_canyon_terrain = yes
					}
				}
				debug_log = "BC: DIAG Q5 terrain desert"
			}
			else_if = {
				limit = {
					OR = {
						agot_is_frozen_flats_terrain = yes
						agot_is_glacier_terrain = yes
					}
				}
				debug_log = "BC: DIAG Q5 terrain winter"
			}
			else_if = {
				limit = {
					OR = {
						agot_is_forest_terrain = yes
						agot_is_taiga_terrain = yes
					}
				}
				debug_log = "BC: DIAG Q5 terrain forest"
			}
			else_if = {
				limit = {
					OR = {
						agot_is_hills_terrain = yes
						agot_is_mountains_terrain = yes
						agot_is_wetlands_terrain = yes
						agot_is_the_bog_terrain = yes
						agot_is_taiga_bog_terrain = yes
						agot_is_terraced_hills_terrain = yes
						agot_is_highlands_terrain = yes
					}
				}
				debug_log = "BC: DIAG Q5 terrain rough"
			}
			else_if = {
				limit = {
					OR = {
						agot_is_farmlands_terrain = yes
						agot_is_plains_terrain = yes
						agot_is_steppe_terrain = yes
						agot_is_floodplains_terrain = yes
						agot_is_urban_terrain = yes
					}
				}
				debug_log = "BC: DIAG Q5 terrain open"
			}
			else = {
				debug_log = "BC: DIAG Q5 terrain NONE MATCHED"
			}
		}

		if = {
			limit = { NOT = { has_game_rule = bc_aftermath_off } }

			if = {
				limit = { bc_battle_is_notable_trigger = yes }
				debug_log = "BC: win notable"

				combat.location ?= {
					save_scope_as = bc_battle_province
				}

				# The province check is an enclosing if, not part of the limit -
				# it is a property of the battle, and re-asking it once per
				# knight is noise.
				if = {
					limit = { exists = scope:bc_battle_province }
					every_side_knight = {
						limit = {
							# Knights die in the combat that just ended. Vanilla
							# guards its own post-battle knight handling the same
							# way (AGOT combat_events.txt:854, :1028; the
							# vanilla equivalents are :813 and :877).
							is_alive = yes
							# See bc_chronicles.0092 for why this is trigger_if.
							trigger_if = {
								limit = { exists = root.side_commander }
								NOT = { this = root.side_commander }
							}
						}
						bc_battle_memory_effect = { TYPE = bc_battle_memory_won }
					}
				}

				side_commander ?= {
					if = {
						limit = { is_alive = yes }
						# NOTE, deviation from the spec: the spec puts is_alive in
						# the nickname limit only, which would let a commander who
						# died winning still take the tally. It is hoisted here
						# instead because the tally exists solely to gate the
						# nickname and a dead man will never command again, so the
						# difference is unobservable - and one gate is clearer than
						# two.
						bc_notable_battle_counter_effect = yes

						# `exists` is belt and braces. What actually makes the
						# var: read below safe is bc_notable_battle_counter_effect
						# immediately above, which always writes the variable
						# before anything reads it. Do not rely on evaluation
						# order to make it safe - CK3 guarantees no order of
						# evaluation among siblings, in a limit or anywhere else.
						if = {
							limit = {
								has_any_nickname = no
								exists = var:bc_notable_battles
								var:bc_notable_battles >= 3
								# martial, not prowess. This names a man for
								# commanding, and it is the same threshold the mod
								# already uses for its high-martial phase-event
								# bonus. Deliberately NOT AGOT's heroic gate of
								# prowess >= 25 plus bold AI: a man can be the
								# finest field commander in Westeros and a
								# mediocre swordsman, and that is who this is for.
								martial >= 14
							}
							# The toast inside the effect needs a character for
							# left_icon, and a scripted effect does not rebind
							# ROOT - ROOT is still the combat side. Save him here.
							save_scope_as = bc_named_commander
							bc_aftermath_nickname_effect = yes
						}
					}
				}
			}
			else = {
				debug_log = "BC: win not notable"
			}
		}

		# LAST, and OUTSIDE the gate above. Both properties are load-bearing.
		# Inside the gate, a player who turned aftermath off would accumulate
		# scratch forever. Before the grading branch, the grading would read
		# variables that had just been removed - from plan 3 onward the branch
		# reads bc_battle_style and bc_battle_foe - and would silently never
		# fire while logging a missing-variable line on every combat end in the
		# world. Grouping all the ungated work together is the natural way to
		# write this and is exactly wrong.
		side_commander ?= {
			bc_clear_scratch_effect = yes
		}
	}
}

#

							# every_side_knight includes him; this costs nothing if
							# it does not.
							#
							# NOTE, deviation from the spec: the spec writes this
							# as a bare `NOT = { this = root.side_commander }`.

							# line for every knight of every leaderless battle.

								limit = { exists = root.side_commander }

							}
